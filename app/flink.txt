public class KafkaToIcebergJob {

    // DDL-колонки в том же порядке
    public static final String[] COLUMN_NAMES = {
        "ssohash", "user_id_hash", "product_id",
        "event_timestamp", "event_name", "is_promoted",
        "platform", "category_id",
        "category_level_1", "category_level_2", "category_level_3",

    };

    // Типы: все varchar кроме event_timestamp (timestamp) и dt (date)
    public static final LogicalType[] COLUMN_TYPES = buildTypes();

    private static LogicalType[] buildTypes() {
        LogicalType[] types = new LogicalType[COLUMN_NAMES.length];
        for (int i = 0; i < COLUMN_NAMES.length; i++) {
            String name = COLUMN_NAMES[i];
            if (name.equals("event_timestamp")) {
                types[i] = new TimestampType(6);
            } else if (name.equals("dt")) {
                types[i] = new DateType();
            } else {
                types[i] = VarCharType.STRING_TYPE;
            }
        }
        return types;
    }

    public static void main(String[] args) throws Exception {

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Checkpoint обязателен для Iceberg sink
        env.enableCheckpointing(60_000);
        env.getCheckpointConfig().setMinPauseBetweenCheckpoints(30_000);

        // --- Kafka Source ---
        KafkaSource<String> kafkaSource = KafkaSource.<String>builder()
            .setBootstrapServers("kafka-broker:9092")
            .setTopics("your_topic_name")
            .setGroupId("flink-iceberg-writer")
            .setStartingOffsets(OffsetsInitializer.earliest())
            .setValueOnlyDeserializer(new SimpleStringSchema())
            .build();

        // --- Parse JSON → RowData ---
        var stream = env
            .fromSource(kafkaSource, WatermarkStrategy.noWatermarks(), "Kafka Source")
            .map(new KafkaJsonToRowData())
            .filter(row -> row != null);  // дропаем битые сообщения

        // --- Iceberg Sink ---
        Configuration hadoopConf = new Configuration();
        // если S3 — добавь:
        // hadoopConf.set("fs.s3a.endpoint", "...");
        // hadoopConf.set("fs.s3a.access.key", "...");
        // hadoopConf.set("fs.s3a.secret.key", "...");

        TableLoader tableLoader = TableLoader.fromHadoopTable(
            "s3://starrocks-ods/warehouse/test_db/ANALOGS_BY_PROMOTED_AGG_DAILY",
            hadoopConf
        );

        FlinkSink.forRowData(stream)
            .tableLoader(tableLoader)
            .upsert(true)                    // upsert режим (equality delete)
            .equalityFieldColumns(            // ключ upsert — подбери под бизнес-логику
                java.util.List.of("user_id_hash", "product_id", "event_timestamp")
            )
            .writeParallelism(4)
            .build();

        env.execute("Kafka → Iceberg ANALOGS_BY_PROMOTED_AGG_DAILY");
    }

    // -------------------------------------------------------------------
    // Маппер JSON → RowData
    // -------------------------------------------------------------------
    public static class KafkaJsonToRowData implements MapFunction<String, RowData> {

        private static final ObjectMapper MAPPER = new ObjectMapper();
        private static final DateTimeFormatter TS_FMT =
            DateTimeFormatter.ISO_OFFSET_DATE_TIME;   // "2026-05-20T13:26:38.180+05:00"

        @Override
        public RowData map(String json) {
            try {
                JsonNode n = MAPPER.readTree(json);
                GenericRowData row = new GenericRowData(RowKind.INSERT, COLUMN_NAMES.length);

                // Индексы соответствуют COLUMN_NAMES
                setStr(row, 0,  n, "session_id");          // ssohash  ← маппинг из Kafka
                setStr(row, 1,  n, "user_id");
                setStr(row, 2,  n, "product_id");
                setTs (row, 3,  n, "event_timestamp");
                setStr(row, 4,  n, "event_name");
                setStr(row, 5,  n, "is_promoted");
                setStr(row, 6,  n, "platform");
                setStr(row, 7,  n, "category_id");
                setStr(row, 8,  n, "category_level_1");
                setStr(row, 9,  n, "category_level_2");
                setStr(row, 10, n, "category_level_3");
                setStr(row, 11, n, "category_level_4");   // нет в Kafka → null
                setStr(row, 12, n, "category_level_5");   // нет в Kafka → null
                setStr(row, 13, n, "brand");
                setStr(row, 14, n, "rating");
                setStr(row, 15, n, "product_name");
                setStr(row, 16, n, "shop_test_group");
                setStr(row, 17, n, "city");               // → shop_city
                setStr(row, 18, n, "item_price");         // → price
                setStr(row, 19, n, "block_name");         // нет в Kafka → null
                setStr(row, 20, n, "page_category");      // нет в Kafka → null
                setStr(row, 21, n, "ads_unique_id");      // нет в Kafka → null
                setStr(row, 22, n, "ads_supplier_id");    // нет в Kafka → null
                setStr(row, 23, n, "listitems_product_id"); // нет → null
                setStr(row, 24, n, "app_id");             // нет → null
                setStr(row, 25, n, "session_id");         // sso_hash ← дублируем session_id

                // dt: берём из load_dt или event_timestamp
                setDate(row, 26, n, "load_dt");

                return row;

            } catch (Exception e) {
                // логируем и пропускаем битое сообщение
                System.err.println("Parse error: " + e.getMessage() + " | json: " + json);
                return null;
            }
        }

        private void setStr(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            row.setField(idx, (v == null || v.isNull())
                ? null
                : BinaryStringData.fromString(v.asText()));
        }

        private void setTs(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            if (v == null || v.isNull()) { row.setField(idx, null); return; }
            try {
                OffsetDateTime odt = OffsetDateTime.parse(v.asText(), TS_FMT);
                LocalDateTime ldt = odt.toLocalDateTime();
                row.setField(idx, TimestampData.fromLocalDateTime(ldt));
            } catch (Exception e) {
                row.setField(idx, null);
            }
        }

        private void setDate(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            if (v == null || v.isNull()) { row.setField(idx, null); return; }
            try {
                // "2026-05-21T17:09:18.000+05:00" → берём только дату
                String raw = v.asText();
                LocalDate ld = raw.length() > 10
                    ? OffsetDateTime.parse(raw, TS_FMT).toLocalDate()
                    : LocalDate.parse(raw);
                row.setField(idx, (int) ld.toEpochDay());
            } catch (Exception e) {
                row.setField(idx, null);
            }
        }
    }
}


import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;

public class KafkaToIcebergSqlJob {

   public static void main(String[] args) throws Exception {

       StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
       env.enableCheckpointing(60_000);

       StreamTableEnvironment tEnv = StreamTableEnvironment.create(env);

       // ================================================================
       // 1. Kafka Source Table
       // ================================================================
       tEnv.executeSql("""
           CREATE TEMPORARY TABLE kafka_source (
               session_id          STRING,
               user_id             STRING,
               product_id          STRING,
               event_timestamp     STRING,       -- парсим руками в INSERT
               event_name          STRING,
               is_promoted         STRING,
               platform            STRING,
               category_id         STRING,
               category_level      STRING,
               category_level_1    STRING,
               category_level_2    STRING,
               category_level_3    STRING,
               brand               STRING,
               rating              STRING,
               product_name        STRING,
               shop_test_group     STRING,
               city                STRING,
               item_price          STRING,
               app_id              STRING,
               cnt_views           STRING,
               cnt_clicks          STRING,
               cnt_purchases_all   STRING,
               cnt_purchases_given STRING,
               load_dt             STRING        -- парсим руками в INSERT
           ) WITH (
               'connector'                     = 'kafka',
               'topic'                         = 'your_topic_name',
               'properties.bootstrap.servers'  = 'kafka-broker:9092',
               'properties.group.id'           = 'flink-iceberg-sql-writer',
               'scan.startup.mode'             = 'earliest-offset',
               'format'                        = 'json',
               'json.ignore-parse-errors'      = 'true',
               'json.timestamp-format.standard'= 'ISO-8601'
           )
       """);

       // ================================================================
       // 2. Iceberg Sink Table
       //    Повторяем DDL из скриншота, добавляем catalog
       // ================================================================
       tEnv.executeSql("""
           CREATE CATALOG iceberg_catalog WITH (
               'type'            = 'iceberg',
               'catalog-type'    = 'hadoop',
               'warehouse'       = 's3://starrocks-ods/warehouse/test_db',
               'property-version'= '1'
           )
       """);

       tEnv.executeSql("USE CATALOG iceberg_catalog");
       tEnv.executeSql("USE test_db");

       // Создаём таблицу если не существует
       // Если таблица уже создана в StarRocks — этот блок можно пропустить
       tEnv.executeSql("""
           CREATE TABLE IF NOT EXISTS ANALOGS_BY_PROMOTED_AGG_DAILY (
               ssohash                 STRING,
               user_id_hash            STRING,
               product_id              STRING,
               event_timestamp         TIMESTAMP(6),
               event_name              STRING,
               is_promoted             STRING,
               platform                STRING,
               category_id             STRING,
               category_level_1        STRING,
               category_level_2        STRING,
               category_level_3        STRING,
               category_level_4        STRING,
               category_level_5        STRING,
               brand                   STRING,
               rating                  STRING,
               product_name            STRING,
               shop_test_group         STRING,
               shop_city               STRING,
               price                   STRING,
               block_name              STRING,
               page_category           STRING,
               ads_unique_id           STRING,
               ads_supplier_id         STRING,
               listitems_product_id    STRING,
               app_id                  STRING,
               sso_hash                STRING,
               dt                      DATE,
               PRIMARY KEY (user_id_hash, product_id, event_timestamp) NOT ENFORCED
           ) PARTITIONED BY (dt)
           WITH (
               'write.upsert.enabled'              = 'true',
               'write.merge.mode'                  = 'merge-on-read',
               'write.format.default'              = 'parquet',
               'write.parquet.compression-codec'   = 'zstd',
               'write.target-file-size-bytes'      = '536870912'
           )
       """);

       // ================================================================
       // 3. INSERT: Kafka → Iceberg с маппингом полей
       // ================================================================
       tEnv.executeSql("""
           INSERT INTO ANALOGS_BY_PROMOTED_AGG_DAILY
           SELECT
               session_id                                          AS ssohash,
               user_id                                             AS user_id_hash,
               product_id,
               -- ISO-8601 с timezone → TIMESTAMP без timezone
               TO_TIMESTAMP(
                   REGEXP_REPLACE(event_timestamp, '\\+\\d{2}:\\d{2}$', ''),
                   'yyyy-MM-dd''T''HH:mm:ss.SSS'
               )                                                   AS event_timestamp,
               event_name,
               is_promoted,
               platform,
               category_id,
               category_level_1,
               category_level_2,
               category_level_3,
               CAST(NULL AS STRING)                                AS category_level_4,
               CAST(NULL AS STRING)                                AS category_level_5,
               brand,
               rating,
               product_name,
               shop_test_group,
               city                                                AS shop_city,
               item_price                                          AS price,
               CAST(NULL AS STRING)                                AS block_name,
               CAST(NULL AS STRING)                                AS page_category,
               CAST(NULL AS STRING)                                AS ads_unique_id,
               CAST(NULL AS STRING)                                AS ads_supplier_id,
               CAST(NULL AS STRING)                                AS listitems_product_id,
               app_id,
               session_id                                          AS sso_hash,
               -- load_dt: "2026-05-21T17:09:18.000+05:00" → DATE
               TO_DATE(SUBSTRING(load_dt, 1, 10))                  AS dt
           FROM kafka_source
       """);
   }
}

    // ================================================================
    // Mapper: JSON String → RowData
    // ================================================================
    public static class KafkaJsonMapper
        implements org.apache.flink.api.common.functions.MapFunction<String, RowData> {

        private static final ObjectMapper MAPPER = new ObjectMapper();
        private static final DateTimeFormatter ISO_FMT =
            DateTimeFormatter.ISO_OFFSET_DATE_TIME;

        @Override
        public RowData map(String json) {
            try {
                JsonNode n = MAPPER.readTree(json);
                GenericRowData row = new GenericRowData(RowKind.INSERT, COLUMNS.length);

                //  idx  Iceberg колонка       Kafka поле
                setStr(row,  0, n, "session_id");        // ssohash
                setStr(row,  1, n, "user_id");           // user_id_hash
                setStr(row,  2, n, "product_id");
                setTs (row,  3, n, "event_timestamp");
                setStr(row,  4, n, "event_name");
                setStr(row,  5, n, "is_promoted");
                setStr(row,  6, n, "platform");
                setStr(row,  7, n, "category_id");
                setStr(row,  8, n, "category_level_1");
                setStr(row,  9, n, "category_level_2");
                setStr(row, 10, n, "category_level_3");
                row.setField(11, null);                  // category_level_4 — нет в Kafka
                row.setField(12, null);                  // category_level_5 — нет в Kafka
                setStr(row, 13, n, "brand");
                setStr(row, 14, n, "rating");
                setStr(row, 15, n, "product_name");
                setStr(row, 16, n, "shop_test_group");
                setStr(row, 17, n, "city");              // shop_city
                setStr(row, 18, n, "item_price");        // price
                row.setField(19, null);                  // block_name
                row.setField(20, null);                  // page_category
                row.setField(21, null);                  // ads_unique_id
                row.setField(22, null);                  // ads_supplier_id
                row.setField(23, null);                  // listitems_product_id
                setStr(row, 24, n, "app_id");
                setStr(row, 25, n, "session_id");        // sso_hash
                setDate(row, 26, n, "load_dt");          // dt

                return row;

            } catch (Exception e) {
                System.err.println("[SKIP] parse error: " + e.getMessage()
                    + "\njson: " + json);
                return null;
            }
        }

        private void setStr(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            row.setField(idx, (v == null || v.isNull())
                ? null
                : BinaryStringData.fromString(v.asText()));
        }

        private void setTs(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            if (v == null || v.isNull()) { row.setField(idx, null); return; }
            try {
                var ldt = OffsetDateTime.parse(v.asText(), ISO_FMT).toLocalDateTime();
                row.setField(idx, TimestampData.fromLocalDateTime(ldt));
            } catch (Exception e) {
                row.setField(idx, null);
            }
        }

        private void setDate(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            if (v == null || v.isNull()) { row.setField(idx, null); return; }
            try {
                String raw = v.asText();
                LocalDate ld = raw.length() > 10
                    ? OffsetDateTime.parse(raw, ISO_FMT).toLocalDate()
                    : LocalDate.parse(raw);
                row.setField(idx, (int) ld.toEpochDay());
            } catch (Exception e) {
                row.setField(idx, null);
            }
        }
    }
