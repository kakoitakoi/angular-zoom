import org.apache.iceberg.*;
import org.apache.iceberg.catalog.*;
import org.apache.iceberg.jdbc.JdbcCatalog;
import org.apache.iceberg.types.Types;
import org.apache.hadoop.conf.Configuration;

import java.util.HashMap;
import java.util.Map;

public class CreateIcebergTable {

    public static void main(String[] args) {

        // 1. Настройка каталога
        Map<String, String> catalogProps = new HashMap<>();
        catalogProps.put("uri", "jdbc:postgresql://your-pg-host:5432/iceberg_catalog");
        catalogProps.put("jdbc.user", "iceberg");
        catalogProps.put("jdbc.password", "password");
        catalogProps.put("warehouse", "s3://your-bucket/iceberg");

        // S3/MinIO настройки
        catalogProps.put("io-impl", "org.apache.iceberg.aws.s3.S3FileIO");
        catalogProps.put("s3.endpoint", "http://your-minio-host:9000");
        catalogProps.put("s3.access-key-id", "YOUR_ACCESS_KEY");
        catalogProps.put("s3.secret-access-key", "YOUR_SECRET_KEY");
        catalogProps.put("s3.path-style-access", "true"); // для MinIO

        // 2. Инициализация каталога
        JdbcCatalog catalog = new JdbcCatalog();
        catalog.initialize("iceberg_catalog", catalogProps, new Configuration());

        // 3. Создать базу (namespace)
        Namespace namespace = Namespace.of("test_db");
        if (!catalog.namespaceExists(namespace)) {
            catalog.createNamespace(namespace);
            System.out.println("Namespace создан: test_db");
        }

        // 4. Определить схему
        Schema schema = new Schema(
            Types.NestedField.optional(1,  "ssohash",              Types.StringType.get()),
            Types.NestedField.optional(2,  "user_id_hash",         Types.StringType.get()),
            Types.NestedField.optional(3,  "product_id",           Types.StringType.get()),
            Types.NestedField.required(4,  "event_name",           Types.StringType.get()),
            Types.NestedField.optional(5,  "is_promoted",          Types.StringType.get()),
            Types.NestedField.optional(6,  "platform",             Types.StringType.get()),
            Types.NestedField.optional(7,  "category_id",          Types.StringType.get()),
            Types.NestedField.optional(8,  "category_level_1",     Types.StringType.get()),
            Types.NestedField.optional(9,  "category_level_2",     Types.StringType.get()),
            Types.NestedField.optional(10, "category_level_3",     Types.StringType.get()),
            Types.NestedField.optional(11, "category_level_4",     Types.StringType.get()),
            Types.NestedField.optional(12, "category_level_5",     Types.StringType.get()),
            Types.NestedField.optional(13, "brand",                Types.StringType.get()),
            Types.NestedField.optional(14, "rating",               Types.StringType.get()),
            Types.NestedField.optional(15, "product_name",         Types.StringType.get()),
            Types.NestedField.optional(16, "shop_test_group",      Types.StringType.get()),
            Types.NestedField.optional(17, "shop_city",            Types.StringType.get()),
            Types.NestedField.optional(18, "price",                Types.StringType.get()),
            Types.NestedField.required(19, "event_timestamp",      Types.TimestampType.withoutZone()),
            Types.NestedField.optional(20, "block_name",           Types.StringType.get()),
            Types.NestedField.optional(21, "page_category",        Types.StringType.get()),
            Types.NestedField.optional(22, "ads_unique_id",        Types.StringType.get()),
            Types.NestedField.optional(23, "ads_supplier_id",      Types.StringType.get()),
            Types.NestedField.optional(24, "listitems_product_id", Types.StringType.get()),
            Types.NestedField.optional(25, "app_id",               Types.StringType.get()),
            Types.NestedField.optional(26, "sso_hash",             Types.StringType.get()),
            Types.NestedField.required(27, "dt",                   Types.DateType.get())
        );

        // 5. Партиционирование по dt
        PartitionSpec spec = PartitionSpec.builderFor(schema)
            .day("dt")
            .build();

        // 6. Свойства таблицы
        Map<String, String> tableProps = new HashMap<>();
        tableProps.put("format-version", "2");
        tableProps.put("write.upsert.enabled", "true");
        tableProps.put("write.merge.mode", "merge-on-read");
        tableProps.put("write.target-file-size-bytes", "536870912");

        // 7. Создать таблицу
        TableIdentifier tableId = TableIdentifier.of(namespace, "aws_events");

        if (!catalog.tableExists(tableId)) {
            catalog.createTable(tableId, schema, spec, tableProps);
            System.out.println("Таблица создана: test_db.aws_events");
        } else {
            System.out.println("Таблица уже существует");
        }

        catalog.close();
    }
}

// pom.xml dependencies:
// flink-connector-kafka 1.17

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.serialization.SimpleStringSchema;
import org.apache.flink.connector.kafka.source.KafkaSource;
import org.apache.flink.connector.kafka.source.enumerator.initializer.OffsetsInitializer;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.table.data.*;
import org.apache.flink.table.data.binary.BinaryStringData;
import org.apache.flink.table.types.logical.*;
import org.apache.flink.types.RowKind;
import org.apache.iceberg.flink.FlinkSchemaUtil;
import org.apache.iceberg.flink.TableLoader;
import org.apache.iceberg.flink.sink.FlinkSink;
import org.apache.hadoop.conf.Configuration;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.OffsetDateTime;
import java.time.format.DateTimeFormatter;

public class KafkaToIcebergJob {

    // DDL-колонки в том же порядке
    public static final String[] COLUMN_NAMES = {
        "ssohash", "user_id_hash", "product_id",
        "event_timestamp", "event_name", "is_promoted",
        "platform", "category_id",
        "category_level_1", "category_level_2", "category_level_3",
        "category_level_4", "category_level_5",
        "brand", "rating", "product_name",
        "shop_test_group", "shop_city", "price",
        "block_name", "page_category", "ads_unique_id",
        "ads_supplier_id", "listitems_product_id",
        "app_id", "sso_hash", "dt"
    };

    // Типы: все varchar кроме event_timestamp (timestamp) и dt (date)
    public static final LogicalType[] COLUMN_TYPES = buildTypes();

    private static LogicalType[] buildTypes() {
        LogicalType[] types = new LogicalType[COLUMN_NAMES.length];
        for (int i = 0; i < COLUMN_NAMES.length; i++) {
            String name = COLUMN_NAMES[i];
            if (name.equals("event_timestamp")) {
                types[i] = new TimestampType(6);
            } else if (name.equals("dt")) {
                types[i] = new DateType();
            } else {
                types[i] = VarCharType.STRING_TYPE;
            }
        }
        return types;
    }

    public static void main(String[] args) throws Exception {

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Checkpoint обязателен для Iceberg sink
        env.enableCheckpointing(60_000);
        env.getCheckpointConfig().setMinPauseBetweenCheckpoints(30_000);

        // --- Kafka Source ---
        KafkaSource<String> kafkaSource = KafkaSource.<String>builder()
            .setBootstrapServers("kafka-broker:9092")
            .setTopics("your_topic_name")
            .setGroupId("flink-iceberg-writer")
            .setStartingOffsets(OffsetsInitializer.earliest())
            .setValueOnlyDeserializer(new SimpleStringSchema())
            .build();

        // --- Parse JSON → RowData ---
        var stream = env
            .fromSource(kafkaSource, WatermarkStrategy.noWatermarks(), "Kafka Source")
            .map(new KafkaJsonToRowData())
            .filter(row -> row != null);  // дропаем битые сообщения

        // --- Iceberg Sink ---
        Configuration hadoopConf = new Configuration();
        // если S3 — добавь:
        // hadoopConf.set("fs.s3a.endpoint", "...");
        // hadoopConf.set("fs.s3a.access.key", "...");
        // hadoopConf.set("fs.s3a.secret.key", "...");

        TableLoader tableLoader = TableLoader.fromHadoopTable(
            "s3://starrocks-ods/warehouse/test_db/ANALOGS_BY_PROMOTED_AGG_DAILY",
            hadoopConf
        );

        FlinkSink.forRowData(stream)
            .tableLoader(tableLoader)
            .upsert(true)                    // upsert режим (equality delete)
            .equalityFieldColumns(            // ключ upsert — подбери под бизнес-логику
                java.util.List.of("user_id_hash", "product_id", "event_timestamp")
            )
            .writeParallelism(4)
            .build();

        env.execute("Kafka → Iceberg ANALOGS_BY_PROMOTED_AGG_DAILY");
    }

    // -------------------------------------------------------------------
    // Маппер JSON → RowData
    // -------------------------------------------------------------------
    public static class KafkaJsonToRowData implements MapFunction<String, RowData> {

        private static final ObjectMapper MAPPER = new ObjectMapper();
        private static final DateTimeFormatter TS_FMT =
            DateTimeFormatter.ISO_OFFSET_DATE_TIME;   // "2026-05-20T13:26:38.180+05:00"

        @Override
        public RowData map(String json) {
            try {
                JsonNode n = MAPPER.readTree(json);
                GenericRowData row = new GenericRowData(RowKind.INSERT, COLUMN_NAMES.length);

                // Индексы соответствуют COLUMN_NAMES
                setStr(row, 0,  n, "session_id");          // ssohash  ← маппинг из Kafka
                setStr(row, 1,  n, "user_id");
                setStr(row, 2,  n, "product_id");
                setTs (row, 3,  n, "event_timestamp");
                setStr(row, 4,  n, "event_name");
                setStr(row, 5,  n, "is_promoted");
                setStr(row, 6,  n, "platform");
                setStr(row, 7,  n, "category_id");
                setStr(row, 8,  n, "category_level_1");
                setStr(row, 9,  n, "category_level_2");
                setStr(row, 10, n, "category_level_3");
                setStr(row, 11, n, "category_level_4");   // нет в Kafka → null
                setStr(row, 12, n, "category_level_5");   // нет в Kafka → null
                setStr(row, 13, n, "brand");
                setStr(row, 14, n, "rating");
                setStr(row, 15, n, "product_name");
                setStr(row, 16, n, "shop_test_group");
                setStr(row, 17, n, "city");               // → shop_city
                setStr(row, 18, n, "item_price");         // → price
                setStr(row, 19, n, "block_name");         // нет в Kafka → null
                setStr(row, 20, n, "page_category");      // нет в Kafka → null
                setStr(row, 21, n, "ads_unique_id");      // нет в Kafka → null
                setStr(row, 22, n, "ads_supplier_id");    // нет в Kafka → null
                setStr(row, 23, n, "listitems_product_id"); // нет → null
                setStr(row, 24, n, "app_id");             // нет → null
                setStr(row, 25, n, "session_id");         // sso_hash ← дублируем session_id

                // dt: берём из load_dt или event_timestamp
                setDate(row, 26, n, "load_dt");

                return row;

            } catch (Exception e) {
                // логируем и пропускаем битое сообщение
                System.err.println("Parse error: " + e.getMessage() + " | json: " + json);
                return null;
            }
        }

        private void setStr(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            row.setField(idx, (v == null || v.isNull())
                ? null
                : BinaryStringData.fromString(v.asText()));
        }

        private void setTs(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            if (v == null || v.isNull()) { row.setField(idx, null); return; }
            try {
                OffsetDateTime odt = OffsetDateTime.parse(v.asText(), TS_FMT);
                LocalDateTime ldt = odt.toLocalDateTime();
                row.setField(idx, TimestampData.fromLocalDateTime(ldt));
            } catch (Exception e) {
                row.setField(idx, null);
            }
        }

        private void setDate(GenericRowData row, int idx, JsonNode n, String field) {
            JsonNode v = n.get(field);
            if (v == null || v.isNull()) { row.setField(idx, null); return; }
            try {
                // "2026-05-21T17:09:18.000+05:00" → берём только дату
                String raw = v.asText();
                LocalDate ld = raw.length() > 10
                    ? OffsetDateTime.parse(raw, TS_FMT).toLocalDate()
                    : LocalDate.parse(raw);
                row.setField(idx, (int) ld.toEpochDay());
            } catch (Exception e) {
                row.setField(idx, null);
            }
        }
    }
}
