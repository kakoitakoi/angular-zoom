# маппинг типов Oracle -> StarRocks
TYPE_MAP = {
    'NUMBER': lambda p, s: f'DECIMAL({p or 18},{s or 0})' if p else 'BIGINT',
    'VARCHAR2': lambda p, s: f'VARCHAR({p})',
    'DATE': lambda p, s: 'DATETIME',
    'TIMESTAMP': lambda p, s: 'DATETIME',
    'CLOB': lambda p, s: 'STRING',
    'FLOAT': lambda p, s: 'DOUBLE',
}
