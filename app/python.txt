"""
GitLab MCP Server на базе FastMCP (streamable-HTTP transport).
Полностью асинхронный: async def инструменты + httpx.AsyncClient.

Авторизация — per-request через HTTP-заголовки:
    X-GitLab-Url   — базовый URL корпоративного GitLab (без trailing slash)
    X-GitLab-Token — Personal Access Token пользователя (scope: api)

Запуск:
    pip install fastmcp httpx
    python gitlab_mcp_server.py

Опциональные переменные окружения:
    MCP_HOST — хост HTTP-сервера (default: 0.0.0.0)
    MCP_PORT — порт           (default: 8000)

Инструменты (11):
    1.  get_file_content       — содержимое файла или листинг директории
    2.  get_repository_tree    — дерево файлов/папок (рекурсивно, все страницы)
    3.  create_or_update_file  — создать/обновить один файл (один коммит)
    4.  commit_multiple_files  — атомарный коммит нескольких файлов (create/update/delete/move)
    5.  create_merge_request   — открыть MR
    6.  merge_merge_request    — принять (смёрджить) MR
    7.  list_commits           — список коммитов ветки
    8.  get_commit_details     — метаданные коммита
    9.  get_commit_diff        — diff коммита
    10. list_branches          — список веток
    11. search_code            — полнотекстовый поиск по коду
"""

import base64
import os
from typing import Any

import httpx
from fastmcp import FastMCP, Context
from fastmcp.exceptions import ToolError

# ---------------------------------------------------------------------------
# Конфигурация сервера
# ---------------------------------------------------------------------------

MCP_HOST = os.environ.get("MCP_HOST", "0.0.0.0")
MCP_PORT = int(os.environ.get("MCP_PORT", "8000"))

# ---------------------------------------------------------------------------
# Извлечение учётных данных из заголовков входящего запроса
# ---------------------------------------------------------------------------

def _credentials(ctx: Context) -> tuple[str, str]:
    """
    Читает X-GitLab-Url и X-GitLab-Token из HTTP-заголовков текущего запроса.

    Returns:
        (api_base, token)

    Raises:
        ToolError — если заголовок отсутствует.
    """
    headers = ctx.request_context.request.headers

    url   = headers.get("x-gitlab-url", "").rstrip("/")
    token = headers.get("x-gitlab-token", "")

    if not url:
        raise ToolError("Заголовок X-GitLab-Url обязателен.")
    if not token:
        raise ToolError("Заголовок X-GitLab-Token обязателен.")

    return f"{url}/api/v4", token


def _http_headers(token: str) -> dict:
    return {"PRIVATE-TOKEN": token, "Content-Type": "application/json"}


# ---------------------------------------------------------------------------
# Асинхронные HTTP-хелперы
# ---------------------------------------------------------------------------

async def _get(api_base: str, token: str, path: str, params: dict | None = None) -> Any:
    async with httpx.AsyncClient(headers=_http_headers(token), timeout=30.0) as c:
        r = await c.get(f"{api_base}{path}", params=params)
        r.raise_for_status()
        return r.json()


async def _post(api_base: str, token: str, path: str, json: dict) -> Any:
    async with httpx.AsyncClient(headers=_http_headers(token), timeout=30.0) as c:
        r = await c.post(f"{api_base}{path}", json=json)
        r.raise_for_status()
        return r.json()


async def _put(api_base: str, token: str, path: str, json: dict) -> Any:
    async with httpx.AsyncClient(headers=_http_headers(token), timeout=30.0) as c:
        r = await c.put(f"{api_base}{path}", json=json)
        r.raise_for_status()
        return r.json()


def _encode_id(project_id: str) -> str:
    """namespace/repo → namespace%2Frepo."""
    return project_id.replace("/", "%2F")


# ---------------------------------------------------------------------------
# FastMCP приложение
# ---------------------------------------------------------------------------

mcp = FastMCP(
    name="gitlab",
    description=(
        "Асинхронные инструменты для работы с корпоративным GitLab (API v4). "
        "Требует заголовки X-GitLab-Url и X-GitLab-Token в каждом запросе."
    ),
)

# ---------------------------------------------------------------------------
# Tool 1 — Содержимое файла или листинг директории
# ---------------------------------------------------------------------------

@mcp.tool()
async def get_file_content(
    project_id: str,
    path: str,
    ref: str = "main",
    ctx: Context = None,
) -> dict:
    """
    Возвращает содержимое файла (декодированное из base64) или список
    элементов директории для указанного пути.

    Args:
        project_id: ID или «namespace/repo» проекта.
        path:       Путь к файлу или директории.
        ref:        Ветка, тег или SHA (default: main).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    # Пробуем как файл
    try:
        encoded_path = path.replace("/", "%2F")
        data = await _get(api_base, token,
                          f"/projects/{pid}/repository/files/{encoded_path}",
                          {"ref": ref})
        content = base64.b64decode(data["content"]).decode("utf-8", errors="replace")
        return {
            "type": "file",
            "file_name": data["file_name"],
            "file_path": data["file_path"],
            "size": data["size"],
            "ref": data["ref"],
            "last_commit_id": data["last_commit_id"],
            "content": content,
        }
    except httpx.HTTPStatusError as e:
        if e.response.status_code != 404:
            raise

    # Fallback: директория
    items = await _get(api_base, token,
                       f"/projects/{pid}/repository/tree",
                       {"path": path, "ref": ref, "per_page": 100})
    return {"type": "directory", "path": path, "ref": ref, "items": items}


# ---------------------------------------------------------------------------
# Tool 2 — Полное дерево файлов и папок
# ---------------------------------------------------------------------------

@mcp.tool()
async def get_repository_tree(
    project_id: str,
    path: str = "",
    ref: str = "main",
    recursive: bool = True,
    ctx: Context = None,
) -> list[dict]:
    """
    Возвращает дерево файлов и директорий репозитория (все страницы пагинации).

    Args:
        project_id: ID или «namespace/repo» проекта.
        path:       Стартовый путь (пустая строка = корень).
        ref:        Ветка/тег/SHA (default: main).
        recursive:  Обходить подпапки рекурсивно (default: True).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    params: dict = {"ref": ref, "recursive": recursive, "per_page": 100}
    if path:
        params["path"] = path

    results: list[dict] = []
    page = 1
    while True:
        params["page"] = page
        batch = await _get(api_base, token, f"/projects/{pid}/repository/tree", params)
        if not batch:
            break
        results.extend(batch)
        if len(batch) < 100:
            break
        page += 1

    return results


# ---------------------------------------------------------------------------
# Tool 3 — Создать или обновить файл (один коммит)
# ---------------------------------------------------------------------------

@mcp.tool()
async def create_or_update_file(
    project_id: str,
    file_path: str,
    content: str,
    commit_message: str,
    branch: str = "main",
    start_branch: str | None = None,
    ctx: Context = None,
) -> dict:
    """
    Создаёт новый файл или обновляет существующий одним коммитом.
    Метод (POST/PUT) определяется автоматически.

    Args:
        project_id:     ID или «namespace/repo» проекта.
        file_path:      Путь к файлу в репозитории.
        content:        Новое содержимое файла (plain text, UTF-8).
        commit_message: Сообщение коммита.
        branch:         Целевая ветка (default: main).
        start_branch:   Создать новую ветку от этой базы (опционально).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)
    encoded_fp = file_path.replace("/", "%2F")

    payload: dict = {
        "branch": branch,
        "content": content,
        "commit_message": commit_message,
        "encoding": "text",
    }
    if start_branch:
        payload["start_branch"] = start_branch

    # Проверяем существование → выбираем метод
    try:
        await _get(api_base, token,
                   f"/projects/{pid}/repository/files/{encoded_fp}",
                   {"ref": branch})
        method = "update"
        result = await _put(api_base, token,
                            f"/projects/{pid}/repository/files/{encoded_fp}",
                            payload)
    except httpx.HTTPStatusError as e:
        if e.response.status_code != 404:
            raise
        method = "create"
        result = await _post(api_base, token,
                             f"/projects/{pid}/repository/files/{encoded_fp}",
                             payload)

    return {"action": method, **result}


# ---------------------------------------------------------------------------
# Tool 4 — Создать Merge Request
# ---------------------------------------------------------------------------

@mcp.tool()
async def create_merge_request(
    project_id: str,
    source_branch: str,
    target_branch: str,
    title: str,
    description: str = "",
    remove_source_branch: bool = False,
    squash: bool = False,
    ctx: Context = None,
) -> dict:
    """
    Открывает новый Merge Request из source_branch в target_branch.

    Args:
        project_id:           ID или «namespace/repo» проекта.
        source_branch:        Ветка с изменениями.
        target_branch:        Целевая ветка.
        title:                Заголовок MR.
        description:          Описание MR (markdown).
        remove_source_branch: Удалить source_branch после мёрджа.
        squash:               Схлопнуть коммиты в один.
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    return await _post(api_base, token, f"/projects/{pid}/merge_requests", {
        "source_branch": source_branch,
        "target_branch": target_branch,
        "title": title,
        "description": description,
        "remove_source_branch": remove_source_branch,
        "squash": squash,
    })


# ---------------------------------------------------------------------------
# Tool 5 — Принять (смёрджить) Merge Request
# ---------------------------------------------------------------------------

@mcp.tool()
async def merge_merge_request(
    project_id: str,
    merge_request_iid: int,
    merge_commit_message: str | None = None,
    squash: bool = False,
    should_remove_source_branch: bool = False,
    ctx: Context = None,
) -> dict:
    """
    Принимает открытый Merge Request.

    Args:
        project_id:                  ID или «namespace/repo» проекта.
        merge_request_iid:           Внутренний IID MR (не глобальный ID).
        merge_commit_message:        Кастомное сообщение merge-коммита.
        squash:                      Схлопнуть коммиты.
        should_remove_source_branch: Удалить source-ветку после мёрджа.
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    payload: dict = {
        "squash": squash,
        "should_remove_source_branch": should_remove_source_branch,
    }
    if merge_commit_message:
        payload["merge_commit_message"] = merge_commit_message

    return await _put(api_base, token,
                      f"/projects/{pid}/merge_requests/{merge_request_iid}/merge",
                      payload)


# ---------------------------------------------------------------------------
# Tool 6 — Список коммитов
# ---------------------------------------------------------------------------

@mcp.tool()
async def list_commits(
    project_id: str,
    ref_name: str = "main",
    since: str | None = None,
    until: str | None = None,
    path: str | None = None,
    per_page: int = 20,
    ctx: Context = None,
) -> list[dict]:
    """
    Возвращает список коммитов репозитория.

    Args:
        project_id: ID или «namespace/repo» проекта.
        ref_name:   Ветка/тег/SHA (default: main).
        since:      ISO 8601 дата начала (опционально).
        until:      ISO 8601 дата конца (опционально).
        path:       Фильтр по пути файла/директории (опционально).
        per_page:   Кол-во коммитов (max 100, default 20).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    params: dict = {"ref_name": ref_name, "per_page": min(per_page, 100)}
    if since: params["since"] = since
    if until: params["until"] = until
    if path:  params["path"]  = path

    return await _get(api_base, token, f"/projects/{pid}/repository/commits", params)


# ---------------------------------------------------------------------------
# Tool 7 — Детали коммита
# ---------------------------------------------------------------------------

@mcp.tool()
async def get_commit_details(
    project_id: str,
    sha: str,
    stats: bool = True,
    ctx: Context = None,
) -> dict:
    """
    Возвращает метаданные коммита: автор, дата, сообщение, статистика строк.

    Args:
        project_id: ID или «namespace/repo» проекта.
        sha:        Полный или сокращённый SHA коммита.
        stats:      Включить статистику добавленных/удалённых строк.
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)
    return await _get(api_base, token,
                      f"/projects/{pid}/repository/commits/{sha}",
                      {"stats": stats})


# ---------------------------------------------------------------------------
# Tool 8 — Diff коммита
# ---------------------------------------------------------------------------

@mcp.tool()
async def get_commit_diff(
    project_id: str,
    sha: str,
    ctx: Context = None,
) -> list[dict]:
    """
    Возвращает список изменённых файлов с unified diff для коммита.
    Каждый элемент: old_path, new_path, diff, флаги new_file/deleted_file/renamed_file.

    Args:
        project_id: ID или «namespace/repo» проекта.
        sha:        SHA коммита.
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)
    return await _get(api_base, token,
                      f"/projects/{pid}/repository/commits/{sha}/diff")


# ---------------------------------------------------------------------------
# Tool 9 — Список веток
# ---------------------------------------------------------------------------

@mcp.tool()
async def list_branches(
    project_id: str,
    search: str | None = None,
    ctx: Context = None,
) -> list[dict]:
    """
    Возвращает список веток с информацией о последнем коммите каждой.

    Args:
        project_id: ID или «namespace/repo» проекта.
        search:     Строка поиска по имени ветки (опционально).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    params: dict = {"per_page": 100}
    if search:
        params["search"] = search

    return await _get(api_base, token, f"/projects/{pid}/repository/branches", params)


# ---------------------------------------------------------------------------
# Tool 10 — Поиск по коду
# ---------------------------------------------------------------------------

@mcp.tool()
async def search_code(
    project_id: str,
    query: str,
    ref: str = "main",
    ctx: Context = None,
) -> list[dict]:
    """
    Полнотекстовый поиск по содержимому файлов репозитория (scope: blobs).

    Args:
        project_id: ID или «namespace/repo» проекта.
        query:      Поисковый запрос.
        ref:        Ветка/тег для поиска (default: main).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    return await _get(api_base, token, f"/projects/{pid}/search", {
        "scope": "blobs",
        "search": query,
        "ref": ref,
        "per_page": 20,
    })


# ---------------------------------------------------------------------------
# Точка входа
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    mcp.run(transport="streamable-http", host=MCP_HOST, port=MCP_PORT)
