
"""
MCP HTTP Server — async, FastMCP, транспорт streamable-http
Запуск: python mcp_doris_server.py
Зависимости: pip install "mcp[cli]" aiomysql
Эндпоинт: http://localhost:8000/mcp
"""

import os
import aiomysql
from mcp.server.fastmcp import FastMCP

# ── Конфигурация Doris ────────────────────────────────────────────────────────
DORIS_HOST     = os.getenv("DORIS_HOST",     "localhost")
DORIS_PORT     = int(os.getenv("DORIS_PORT", "9030"))
DORIS_USER     = os.getenv("DORIS_USER",     "root")
DORIS_PASSWORD = os.getenv("DORIS_PASSWORD", "")
DORIS_DB       = os.getenv("DORIS_DB",       "test_db")
# ─────────────────────────────────────────────────────────────────────────────

mcp = FastMCP(
    name="doris-query-server",
    # stateless_http=True  # раскомментировать если нужен stateless режим
)


async def _execute(sql: str) -> dict:
    """Открывает соединение, выполняет запрос, возвращает результат."""
    conn = await aiomysql.connect(
        host=DORIS_HOST,
        port=DORIS_PORT,
        user=DORIS_USER,
        password=DORIS_PASSWORD,
        db=DORIS_DB,
        charset="utf8mb4",
        connect_timeout=10,
        cursorclass=aiomysql.DictCursor,
    )
    try:
        async with conn.cursor() as cursor:
            await cursor.execute(sql)
            rows = await cursor.fetchall()          # list[dict]
            columns = [d[0] for d in cursor.description] if cursor.description else []
    finally:
        conn.close()

    # Приводим не-JSON-сериализуемые типы (Decimal, date, …) к str
    safe_rows = [
        {k: (str(v) if v is not None else None) for k, v in row.items()}
        for row in rows
    ]
    return {"columns": columns, "rows": safe_rows, "row_count": len(safe_rows)}


@mcp.tool()
async def query_doris(sql: str) -> dict:
    """
    Выполняет SELECT-запрос в Apache Doris и возвращает результат.

    Args:
        sql: SQL-запрос (только SELECT; остальные операторы отклоняются).

    Returns:
        columns   — список названий колонок
        rows      — список строк (каждая строка — dict {колонка: значение})
        row_count — количество строк
    """
    if not sql.strip().upper().startswith("SELECT"):
        return {"error": "Разрешены только SELECT-запросы."}

    try:
        return await _execute(sql)
    except aiomysql.MySQLError as e:
        return {"error": f"Ошибка Doris: {e}"}
    except Exception as e:
        return {"error": f"Неожиданная ошибка: {e}"}


if __name__ == "__main__":
    # transport="streamable-http" → эндпоинт http://localhost:8000/mcp
    mcp.run(transport="streamable-http")


"""
MCP HTTP Client — тестирует инструмент query_doris через streamable-http
Запуск: python mcp_doris_client.py
Зависимости: pip install "mcp[cli]"
"""

import asyncio
import json
from mcp.client.streamable_http import streamablehttp_client
from mcp import ClientSession

SERVER_URL = "http://localhost:8000/mcp"

TEST_QUERIES = [
    "SELECT 1 AS ping",
    "SELECT NOW() AS server_time",
    "SELECT DATABASE() AS current_db",
    "SELECT * FROM information_schema.tables LIMIT 5",
]


async def run_client() -> None:
    print(f"Подключаемся к MCP-серверу: {SERVER_URL}")
    print("=" * 60)

    async with streamablehttp_client(SERVER_URL) as (read, write, _):
        async with ClientSession(read, write) as session:

            await session.initialize()
            print("✅ Сессия инициализирована\n")

            # Список доступных инструментов
            tools = (await session.list_tools()).tools
            print("Доступные инструменты:")
            for t in tools:
                print(f"  • {t.name}: {t.description}")
            print()

            # Прогоняем тестовые запросы
            for sql in TEST_QUERIES:
                print(f"─── SQL: {sql}")
                result = await session.call_tool("query_doris", {"sql": sql})

                for block in result.content:
                    if block.type == "text":
                        try:
                            data = json.loads(block.text)
                        except json.JSONDecodeError:
                            print(f"  Raw: {block.text}")
                            continue

                        if "error" in data:
                            print(f"  ⚠️  {data['error']}")
                        else:
                            print(f"  Колонки : {data['columns']}")
                            print(f"  Строк   : {data['row_count']}")
                            for row in data["rows"]:
                                print(f"            {row}")
                print()

    print("=" * 60)
    print("Клиент завершил работу.")


if __name__ == "__main__":
    asyncio.run(run_client())
