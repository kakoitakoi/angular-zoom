# --- Маппинг типов Oracle -> StarRocks ---
def map_type(data_type, length, precision, scale):
    dt = data_type.upper()

    if dt == 'NUMBER':
        if precision is None:
            return 'BIGINT'
        if scale and scale > 0:
            return f'DECIMAL({precision}, {scale})'
        if precision <= 3:
            return 'TINYINT'
        if precision <= 5:
            return 'SMALLINT'
        if precision <= 10:
            return 'INT'
        if precision <= 18:
            return 'BIGINT'
        return f'DECIMAL({precision}, {scale or 0})'

    if dt in ('VARCHAR2', 'NVARCHAR2', 'CHAR', 'NCHAR'):
        return f'VARCHAR({min(length or 255, 65533)})'

    if dt in ('DATE', 'TIMESTAMP') or dt.startswith('TIMESTAMP'):
        return 'DATETIME'

    if dt in ('CLOB', 'NCLOB', 'LONG', 'XMLTYPE'):
        return 'STRING'

    if dt in ('BLOB', 'RAW', 'LONG RAW'):
        return 'STRING'  # или VARBINARY — зависит от задачи

    if dt in ('FLOAT', 'BINARY_FLOAT'):
        return 'FLOAT'

    if dt == 'BINARY_DOUBLE':
        return 'DOUBLE'

    # fallback
    return 'STRING'

# --- Генерация CREATE TABLE для StarRocks ---
def generate_starrocks_ddl(table_name, columns, pk_columns):
    col_defs = []

    for col_name, data_type, length, precision, scale, nullable in columns:
        sr_type = map_type(data_type, length, precision, scale)
        null_clause = "" if nullable == 'Y' else " NOT NULL"
        col_defs.append(f"    {col_name} {sr_type}{null_clause}")

    # Если PK есть — используем его, иначе берём первую колонку
    key_cols = pk_columns if pk_columns else [columns[0][0]]
    key_type = "PRIMARY KEY" if pk_columns else "DUPLICATE KEY"

    ddl = f"CREATE TABLE {table_name} (\n"
    ddl += ",\n".join(col_defs)
    ddl += f"\n)\n"
    ddl += f"{key_type} ({', '.join(key_cols)})\n"
    ddl += f"DISTRIBUTED BY HASH({key_cols[0]}) BUCKETS 10\n"
    ddl += f"PROPERTIES (\"replication_num\" = \"1\");"

    return ddl

def search(query: str, source_filter: str = None, top_k: int = 5) -> list[dict]:
    # Эмбеддинг запроса
    q_vec = embed([query])[0]
    q_arr = json.dumps(q_vec)

    # Doris не поддерживает ANN нативно через SQL в ARRAY<FLOAT>,
    # поэтому считаем cosine similarity через встроенную функцию
    where = "WHERE 1=1"
    params = []

    if source_filter:
        where += " AND source = %s"
        params.append(source_filter)

    sql = f"""
        SELECT
            id,
            doc_id,
            source,
            chunk_text,
            cosine_distance(embedding, CAST({q_arr} AS ARRAY<FLOAT>)) AS score
        FROM chunks
        {where}
        ORDER BY score ASC          -- cosine_distance: меньше = ближе
        LIMIT %s
    """
    params.append(top_k)

    with conn.cursor(pymysql.cursors.DictCursor) as cur:
        cur.execute(sql, params)
        return cur.fetchall()

def build_cql_exact(term: str) -> str:
    # строгое совпадение через кавычки
    return f'type IN (page, comment) AND text ~ "\\"{term}\\"" ORDER BY lastmodified DESC'


def fetch_all_cql(cql: str):
    url = f"{BASE_URL}/rest/api/content/search"
    params = {
        "cql": cql,
        "limit": 100,
        "expand": "body.storage"
    }

    results = []

    while url:
        resp = requests.get(url, params=params, auth=AUTH, headers=HEADERS)
        resp.raise_for_status()
        data = resp.json()

        results.extend(data.get("results", []))

        next_link = data.get("_links", {}).get("next")
        if next_link:
            url = urljoin(BASE_URL, next_link)
            params = None  # дальше параметры уже внутри next
        else:
            url = None

    return results

def html_to_structured_text(html: str) -> str:
    soup = BeautifulSoup(html, "html.parser")

    # удаляем мусор
    for tag in soup(["script", "style"]):
        tag.decompose()

    # удаляем confluence макросы (ac:, ri:)
    for tag in soup.find_all():
        if ":" in tag.name:
            tag.decompose()

    lines = []

    def process(node, indent=0):
        if node.name in ["h1", "h2", "h3", "h4"]:
            lines.append("\n" + node.get_text(strip=True).upper())

        elif node.name == "p":
            lines.append("\n" + node.get_text(strip=True))

        elif node.name == "li":
            prefix = "  " * indent + "- "
            lines.append(prefix + node.get_text(strip=True))

        elif node.name in ["ul", "ol"]:
            for li in node.find_all("li", recursive=False):
                process(li, indent + 1)

        elif node.name == "pre":
            lines.append("\n[CODE]")
            lines.append(node.get_text())
            lines.append("[/CODE]")

        elif node.name == "code":
            lines.append(f"`{node.get_text(strip=True)}`")

    for elem in soup.find_all(recursive=False):
        process(elem)

    # fallback если структура не поймалась
    if not lines:
        return soup.get_text(separator="\n").strip()

    # чистим пустые строки
    return "\n".join([l for l in lines if l.strip()])

def search_and_extract(term: str):
    cql = build_cql_exact(term)
    results = fetch_all_cql(cql)

    output = []

    for item in results:
        content_type = item["type"]
        content_id = item["id"]
        title = item.get("title", "")

        html = item.get("body", {}).get("storage", {}).get("value", "")
        text = html_to_structured_text(html)

        output.append({
            "type": content_type,
            "id": content_id,
            "title": title,
            "text": text
        })

        # если это page → подтянуть комментарии
        if content_type == "page":
            comments = fetch_comments(content_id)

            for c in comments:
                c_html = c.get("body", {}).get("storage", {}).get("value", "")
                c_text = html_to_structured_text(c_html)

                output.append({
                    "type": "comment",
                    "parent_page_id": content_id,
                    "id": c["id"],
                    "text": c_text
                })

    return output

from bs4 import BeautifulSoup

def find_rows_by_keyword(html: str, keyword: str) -> list[dict]:
    """
    Ищет строки в таблицах HTML по ключевому слову.
    Возвращает список найденных строк с контекстом.
    """
    soup = BeautifulSoup(html, "html.parser")
    tables = soup.find_all("table")

    if not tables:
        print("Таблицы в документе не найдены.")
        return []

    print(f"Найдено таблиц: {len(tables)}")
    results = []

    for table_idx, table in enumerate(tables):
        rows = table.find_all("tr")

        # Извлекаем заголовок (первая строка)
        headers = []
        if rows:
            headers = [th.get_text(strip=True) for th in rows[0].find_all(["th", "td"])]

        for row_idx, row in enumerate(rows):
            cells = row.find_all(["th", "td"])
            cell_texts = [cell.get_text(strip=True) for cell in cells]
            row_text = " ".join(cell_texts)

            if keyword.lower() in row_text.lower():
                results.append({
                    "table_index": table_idx + 1,
                    "row_index": row_idx,
                    "headers": headers,
                    "cells": cell_texts,
                })

    return results


def print_results(results: list[dict], keyword: str):
    if not results:
        print(f'Строки с "{keyword}" не найдены.')
        return

    print(f'\nНайдено строк с "{keyword}": {len(results)}\n')
    print("=" * 60)

    for r in results:
        print(f"Таблица #{r['table_index']}, строка #{r['row_index']}")

        if r["headers"]:
            # Выводим как пары заголовок: значение
            pairs = zip(r["headers"], r["cells"])
            for header, value in pairs:
                marker = " ◄" if keyword.lower() in value.lower() else ""
                print(f"  {header or '—'}: {value}{marker}")
            # Если колонок больше чем заголовков
            for extra in r["cells"][len(r["headers"]):]:
                print(f"  —: {extra}")
        else:
            print("  " + " | ".join(r["cells"]))

        print("-" * 60)

from bs4 import BeautifulSoup, NavigableString, Tag


def split_into_sections(soup: BeautifulSoup) -> list[dict]:
    """
    Разбивает HTML на секции по заголовкам h1-h6.
    Каждая секция содержит заголовок и все элементы до следующего заголовка.
    """
    sections = []
    current_section = {"heading": None, "heading_level": 0, "elements": []}

    for element in soup.body.children if soup.body else soup.children:
        if not isinstance(element, Tag):
            continue

        if element.name in ("h1", "h2", "h3", "h4", "h5", "h6"):
            # Сохраняем предыдущую секцию если она не пустая
            if current_section["elements"] or current_section["heading"]:
                sections.append(current_section)

            current_section = {
                "heading": element.get_text(strip=True),
                "heading_level": int(element.name[1]),
                "elements": [],
            }
        else:
            current_section["elements"].append(element)

    # Последняя секция
    if current_section["elements"] or current_section["heading"]:
        sections.append(current_section)

    return sections


def extract_text_from_element(element: Tag) -> str:
    return element.get_text(separator=" ", strip=True)


def search_in_table(table: Tag, keyword: str, headers: list[str]) -> list[dict]:
    """Ищет keyword в строках таблицы."""
    matches = []
    rows = table.find_all("tr")

    # Определяем заголовки таблицы
    table_headers = headers
    if rows:
        first_row_cells = rows[0].find_all(["th", "td"])
        candidate_headers = [c.get_text(strip=True) for c in first_row_cells]
        # Считаем строку заголовком если есть <th> теги
        if rows[0].find("th"):
            table_headers = candidate_headers
            rows = rows[1:]

    for row_idx, row in enumerate(rows):
        cells = [td.get_text(strip=True) for td in row.find_all(["th", "td"])]
        row_text = " ".join(cells)

        if keyword.lower() in row_text.lower():
            if table_headers:
                row_data = dict(zip(table_headers, cells))
            else:
                row_data = {f"col_{i}": v for i, v in enumerate(cells)}

            matches.append({"row_index": row_idx, "data": row_data, "raw_cells": cells})

    return matches


def search_in_section(section: dict, keyword: str) -> list[dict]:
    """Ищет keyword во всех элементах секции."""
    results = []

    for element in section["elements"]:
        # Случай 1: таблица
        if element.name == "table":
            table_matches = search_in_table(element, keyword, headers=[])
            if table_matches:
                results.append({
                    "type": "table",
                    "matches": table_matches,
                })

        # Случай 2: вложенные таблицы (div/section внутри)
        elif element.name in ("div", "section", "article"):
            for table in element.find_all("table"):
                table_matches = search_in_table(table, keyword, headers=[])
                if table_matches:
                    results.append({
                        "type": "table",
                        "matches": table_matches,
                    })
            # И текст вне таблиц внутри div
            for child in element.children:
                if isinstance(child, Tag) and child.name != "table":
                    text = extract_text_from_element(child)
                    if keyword.lower() in text.lower():
                        results.append({
                            "type": "text",
                            "element": child.name,
                            "text": text,
                        })

        # Случай 3: обычный текстовый элемент
        else:
            text = extract_text_from_element(element)
            if keyword.lower() in text.lower():
                results.append({
                    "type": "text",
                    "element": element.name,
                    "text": text,
                })

    return results


def find_by_keyword(html: str, keyword: str) -> list[dict]:
    soup = BeautifulSoup(html, "html.parser")
    sections = split_into_sections(soup)

    all_results = []

    for section in sections:
        matches = search_in_section(section, keyword)
        if matches:
            all_results.append({
                "section_heading": section["heading"] or "(без заголовка)",
                "heading_level": section["heading_level"],
                "matches": matches,
            })

    return all_results


def print_results(results: list[dict], keyword: str):
    if not results:
        print(f'"{keyword}" не найдено.')
        return

    total = sum(len(r["matches"]) for r in results)
    print(f'Найдено вхождений "{keyword}": {total} в {len(results)} секции(ях)\n')

    for section_result in results:
        heading = section_result["section_heading"]
        level = section_result["heading_level"]
        prefix = "#" * level if level else "~"
        print(f"{prefix} {heading}")
        print("=" * 60)

        for match in section_result["matches"]:
            if match["type"] == "table":
                print("  [ТАБЛИЦА]")
                for row in match["matches"]:
                    print(f"  Строка {row['row_index'] + 1}:")
                    for col, val in row["data"].items():
                        marker = " ◄" if keyword.lower() in val.lower() else ""
                        print(f"    {col}: {val}{marker}")
            else:
                tag = match.get("element", "?")
                text = match["text"]
                # Подсвечиваем вхождение
                highlighted = text.replace(
                    keyword, f"[{keyword.upper()}]"
                )
                print(f"  [{tag.upper()}] {highlighted}")

        print()

"""
SQL Rules Generator for Text2SQL
Парсит SQL-запросы из логов StarRocks, обогащает семантическими описаниями,
кластеризует по доменам и генерирует rules через LLM.
"""

import json
import re
import hashlib
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Optional

import sqlglot
from sqlglot import exp
import anthropic

# ─────────────────────────────────────────────
# 1. ТИПЫ ДАННЫХ
# ─────────────────────────────────────────────

@dataclass
class ColumnMeta:
    name: str
    data_type: str
    description: str = ""
    sample_values: list[str] = field(default_factory=list)

@dataclass
class TableMeta:
    name: str
    description: str = ""
    columns: dict[str, ColumnMeta] = field(default_factory=dict)


@dataclass
class ParsedQuery:
    raw_sql: str
    fingerprint: str                        # хэш нормализованного SQL — для дедупликации
    tables: list[str] = field(default_factory=list)
    joins: list[dict] = field(default_factory=list)       # {left, right, condition}
    filters: list[str] = field(default_factory=list)      # выражения из WHERE/HAVING
    aggregations: list[str] = field(default_factory=list) # COUNT, SUM, AVG …
    window_funcs: list[str] = field(default_factory=list) # OVER (…)
    group_by_cols: list[str] = field(default_factory=list)
    order_by_cols: list[str] = field(default_factory=list)
    selected_cols: list[str] = field(default_factory=list)
    subqueries: int = 0
    ctes: list[str] = field(default_factory=list)
    parse_error: Optional[str] = None


# ─────────────────────────────────────────────
# 2. ПАРСИНГ
# ─────────────────────────────────────────────

def _fingerprint(sql: str) -> str:
    """Нормализуем SQL и считаем хэш для дедупликации."""
    try:
        normalized = sqlglot.transpile(sql, read="starrocks", write="starrocks",
                                       pretty=False, normalize=True)[0]
    except Exception:
        normalized = re.sub(r"\s+", " ", sql.upper().strip())
    return hashlib.md5(normalized.encode()).hexdigest()


def parse_query(sql: str) -> ParsedQuery:
    pq = ParsedQuery(raw_sql=sql, fingerprint=_fingerprint(sql))
    try:
        tree = sqlglot.parse_one(sql, read="starrocks", error_level=sqlglot.ErrorLevel.WARN)
    except Exception as e:
        pq.parse_error = str(e)
        return pq

    # Таблицы
    pq.tables = sorted({
        t.name.lower()
        for t in tree.find_all(exp.Table)
        if t.name
    })

    # Колонки в SELECT
    pq.selected_cols = [
        col.sql(dialect="starrocks")
        for col in tree.find_all(exp.Column)
        if isinstance(col.parent, (exp.Select, exp.Alias))
    ]

    # JOIN-условия
    for join in tree.find_all(exp.Join):
        pq.joins.append({
            "table": join.this.name if hasattr(join.this, "name") else str(join.this),
            "kind":  join.args.get("kind", ""),
            "on":    join.args["on"].sql(dialect="starrocks") if join.args.get("on") else "",
        })

    # WHERE / HAVING
    for node_type in (exp.Where, exp.Having):
        node = tree.find(node_type)
        if node:
            pq.filters.append(node.this.sql(dialect="starrocks"))

    # Агрегатные функции
    AGG_TYPES = (exp.Count, exp.Sum, exp.Avg, exp.Max, exp.Min,
                 exp.ApproxDistinct, exp.GroupConcat)
    pq.aggregations = list({
        a.sql(dialect="starrocks") for a in tree.find_all(*AGG_TYPES)
    })

    # Оконные функции
    pq.window_funcs = list({
        w.sql(dialect="starrocks") for w in tree.find_all(exp.Window)
    })

    # GROUP BY
    gb = tree.find(exp.Group)
    if gb:
        pq.group_by_cols = [e.sql(dialect="starrocks") for e in gb.expressions]

    # ORDER BY
    ob = tree.find(exp.Order)
    if ob:
        pq.order_by_cols = [e.sql(dialect="starrocks") for e in ob.expressions]

    # Подзапросы
    pq.subqueries = sum(1 for _ in tree.find_all(exp.Subquery))

    # CTE
    with_node = tree.find(exp.With)
    if with_node:
        pq.ctes = [cte.alias for cte in with_node.find_all(exp.CTE)]

    return pq


# ─────────────────────────────────────────────
# 3. ОБОГАЩЕНИЕ МЕТАДАННЫМИ
# ─────────────────────────────────────────────

def enrich(pq: ParsedQuery, schema: dict[str, TableMeta]) -> dict:
    """Добавляем к ParsedQuery семантику из схемы."""
    enriched_tables = {}
    for tname in pq.tables:
        meta = schema.get(tname)
        if not meta:
            enriched_tables[tname] = {"description": "unknown table"}
            continue
        enriched_tables[tname] = {
            "description": meta.description,
            "columns_used": {},
        }
        # Какие колонки таблицы реально используются в запросе
        all_sql = " ".join(
            pq.selected_cols + pq.filters + pq.aggregations +
            pq.group_by_cols + pq.order_by_cols +
            [j["on"] for j in pq.joins]
        ).lower()
        for cname, cmeta in meta.columns.items():
            if cname.lower() in all_sql:
                enriched_tables[tname]["columns_used"][cname] = {
                    "type":        cmeta.data_type,
                    "description": cmeta.description,
                    "samples":     cmeta.sample_values,
                }

    return {
        "tables":       enriched_tables,
        "joins":        pq.joins,
        "filters":      pq.filters,
        "aggregations": pq.aggregations,
        "window_funcs": pq.window_funcs,
        "group_by":     pq.group_by_cols,
        "order_by":     pq.order_by_cols,
        "subqueries":   pq.subqueries,
        "ctes":         pq.ctes,
    }

def cluster_by_domain(queries: list[ParsedQuery]) -> dict[str, list[ParsedQuery]]:
    """
    Группируем запросы по набору таблиц (доменный ключ).
    Запросы без таблиц — в кластер '_unparsed'.

    clusters: dict[str, list[ParsedQuery]] = defaultdict(list)
    for pq in queries:

        clusters[key].append(pq)
    return dict(clusters)


def deduplicate(queries: list[ParsedQuery]) -> list[ParsedQuery]:

    result = []
    for pq in queries:
        if pq.fingerprint not in seen:
            seen.add(pq.fingerprint)
            result.append(pq)
    return result

"""
SQL Rules Generator for Text2SQL
Парсит SQL-запросы из логов StarRocks, обогащает семантическими описаниями,
кластеризует по доменам и генерирует rules через LLM.
"""

import json
import re
import hashlib
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Optional

import sqlglot
from sqlglot import exp
import anthropic

# ─────────────────────────────────────────────
# 1. ТИПЫ ДАННЫХ
# ─────────────────────────────────────────────

@dataclass
class ColumnMeta:
    name: str
    data_type: str
    description: str = ""
    sample_values: list[str] = field(default_factory=list)

@dataclass
class TableMeta:
    name: str
    description: str = ""
    columns: dict[str, ColumnMeta] = field(default_factory=dict)


@dataclass
class ParsedQuery:
    raw_sql: str
    fingerprint: str                        # хэш нормализованного SQL — для дедупликации
    tables: list[str] = field(default_factory=list)
    joins: list[dict] = field(default_factory=list)       # {left, right, condition}
    filters: list[str] = field(default_factory=list)      # выражения из WHERE/HAVING
    aggregations: list[str] = field(default_factory=list) # COUNT, SUM, AVG …
    window_funcs: list[str] = field(default_factory=list) # OVER (…)
    group_by_cols: list[str] = field(default_factory=list)
    order_by_cols: list[str] = field(default_factory=list)
    selected_cols: list[str] = field(default_factory=list)
    subqueries: int = 0
    ctes: list[str] = field(default_factory=list)
    parse_error: Optional[str] = None


# ─────────────────────────────────────────────
# 2. ПАРСИНГ
# ─────────────────────────────────────────────

def _fingerprint(sql: str) -> str:
    """Нормализуем SQL и считаем хэш для дедупликации."""
    try:
        normalized = sqlglot.transpile(sql, read="starrocks", write="starrocks",
                                       pretty=False, normalize=True)[0]
    except Exception:
        normalized = re.sub(r"\s+", " ", sql.upper().strip())
    return hashlib.md5(normalized.encode()).hexdigest()


def parse_query(sql: str) -> ParsedQuery:
    pq = ParsedQuery(raw_sql=sql, fingerprint=_fingerprint(sql))
    try:
        tree = sqlglot.parse_one(sql, read="starrocks", error_level=sqlglot.ErrorLevel.WARN)
    except Exception as e:
        pq.parse_error = str(e)
        return pq

    # Таблицы
    pq.tables = sorted({
        t.name.lower()
        for t in tree.find_all(exp.Table)
        if t.name
    })

    # Колонки в SELECT
    pq.selected_cols = [
        col.sql(dialect="starrocks")
        for col in tree.find_all(exp.Column)
        if isinstance(col.parent, (exp.Select, exp.Alias))
    ]

    # JOIN-условия
    for join in tree.find_all(exp.Join):
        pq.joins.append({
            "table": join.this.name if hasattr(join.this, "name") else str(join.this),
            "kind":  join.args.get("kind", ""),
            "on":    join.args["on"].sql(dialect="starrocks") if join.args.get("on") else "",
        })

    # WHERE / HAVING
    for node_type in (exp.Where, exp.Having):
        node = tree.find(node_type)
        if node:
            pq.filters.append(node.this.sql(dialect="starrocks"))

    # Агрегатные функции
    AGG_TYPES = (exp.Count, exp.Sum, exp.Avg, exp.Max, exp.Min,
                 exp.ApproxDistinct, exp.GroupConcat)
    pq.aggregations = list({
        a.sql(dialect="starrocks") for a in tree.find_all(*AGG_TYPES)
    })

    # Оконные функции
    pq.window_funcs = list({
        w.sql(dialect="starrocks") for w in tree.find_all(exp.Window)
    })

    # GROUP BY
    gb = tree.find(exp.Group)
    if gb:
        pq.group_by_cols = [e.sql(dialect="starrocks") for e in gb.expressions]

    # ORDER BY
    ob = tree.find(exp.Order)
    if ob:
        pq.order_by_cols = [e.sql(dialect="starrocks") for e in ob.expressions]

    # Подзапросы
    pq.subqueries = sum(1 for _ in tree.find_all(exp.Subquery))

    # CTE
    with_node = tree.find(exp.With)
    if with_node:
        pq.ctes = [cte.alias for cte in with_node.find_all(exp.CTE)]

    return pq


# ─────────────────────────────────────────────
# 3. ОБОГАЩЕНИЕ МЕТАДАННЫМИ
# ─────────────────────────────────────────────

def enrich(pq: ParsedQuery, schema: dict[str, TableMeta]) -> dict:
    """Добавляем к ParsedQuery семантику из схемы."""
    enriched_tables = {}
    for tname in pq.tables:
        meta = schema.get(tname)
        if not meta:
            enriched_tables[tname] = {"description": "unknown table"}
            continue
        enriched_tables[tname] = {
            "description": meta.description,
            "columns_used": {},
        }
        # Какие колонки таблицы реально используются в запросе
        all_sql = " ".join(
            pq.selected_cols + pq.filters + pq.aggregations +
            pq.group_by_cols + pq.order_by_cols +
            [j["on"] for j in pq.joins]
        ).lower()
        for cname, cmeta in meta.columns.items():
            if cname.lower() in all_sql:
                enriched_tables[tname]["columns_used"][cname] = {
                    "type":        cmeta.data_type,
                    "description": cmeta.description,
                    "samples":     cmeta.sample_values,
                }

    return {
        "tables":       enriched_tables,
        "joins":        pq.joins,
        "filters":      pq.filters,
        "aggregations": pq.aggregations,
        "window_funcs": pq.window_funcs,
        "group_by":     pq.group_by_cols,
        "order_by":     pq.order_by_cols,
        "subqueries":   pq.subqueries,
        "ctes":         pq.ctes,
    }


# ─────────────────────────────────────────────
# 4. КЛАСТЕРИЗАЦИЯ ПО ДОМЕНАМ
# ─────────────────────────────────────────────

def cluster_by_domain(queries: list[ParsedQuery]) -> dict[str, list[ParsedQuery]]:
    """
    Группируем запросы по набору таблиц (доменный ключ).
    Запросы без таблиц — в кластер '_unparsed'.
    """
    clusters: dict[str, list[ParsedQuery]] = defaultdict(list)
    for pq in queries:
        key = "+".join(sorted(pq.tables)) if pq.tables else "_unparsed"
        clusters[key].append(pq)
    return dict(clusters)


def deduplicate(queries: list[ParsedQuery]) -> list[ParsedQuery]:
    seen: set[str] = set()
    result = []
    for pq in queries:
        if pq.fingerprint not in seen:
            seen.add(pq.fingerprint)
            result.append(pq)
    return result

# 5. ГЕНЕРАЦИЯ RULES ЧЕРЕЗ LLM
# ─────────────────────────────────────────────

SYSTEM_PROMPT = """\
Ты — эксперт по SQL и проектированию Text2SQL систем.
Тебе дают кластер реальных SQL-запросов из production-логов StarRocks
вместе с обогащёнными метаданными (описания таблиц, колонок, типы данных).

Твоя задача — сформулировать "SQL rules" для этого кластера таблиц.
Rules — это короткие, однозначные правила, которые LLM будет использовать
при генерации SQL для данных таблиц.

Формат ответа — строго JSON-массив объектов:
[
  {
    "rule_id": "R001",
    "category": "join|filter|aggregation|naming|performance|business_logic",
    "tables":   ["table1", "table2"],
    "rule":     "Краткое правило на русском языке",
    "rationale": "Почему это правило важно (1-2 предложения)",
    "example_sql": "Короткий пример SQL (необязательно)"
  }
]

Правила:
- Выводи только то, что подтверждается паттернами из запросов
- Не придумывай правила без доказательств в данных
- Категория business_logic — для правил о бизнес-смысле фильтров/агрегаций
- Минимум 5, максимум 20 rules на кластер
- Только JSON, без markdown-обёртки
"""


def build_user_message(
    cluster_key: str,
    queries: list[ParsedQuery],
    schema: dict[str, TableMeta],
    max_queries: int = 30,
) -> str:
    samples = queries[:max_queries]
    enriched = [enrich(pq, schema) for pq in samples]

    payload = {
        "domain": cluster_key,
        "total_queries_in_cluster": len(queries),
        "sample_size": len(samples),
        "queries": [
            {
                "sql":      pq.raw_sql[:2000],  # обрезаем монстров
                "metadata": enriched[i],
            }
            for i, pq in enumerate(samples)
        ],
    }
    return json.dumps(payload, ensure_ascii=False, indent=2)


def generate_rules_for_cluster(
    cluster_key: str,
    queries: list[ParsedQuery],
    schema: dict[str, TableMeta],
    client: anthropic.Anthropic,
    model: str = "claude-opus-4-5",
) -> list[dict]:
    user_msg = build_user_message(cluster_key, queries, schema)
    response = client.messages.create(
        model=model,
        max_tokens=4096,
        system=SYSTEM_PROMPT,
        messages=[{"role": "user", "content": user_msg}],
    )
    raw = response.content[0].text.strip()
    # На случай если модель всё же обернула в ```json
    raw = re.sub(r"^```(?:json)?\s*", "", raw)
    raw = re.sub(r"\s*```$", "", raw)
    try:
        return json.loads(raw)
    except json.JSONDecodeError as e:
        print(f"  [!] JSON parse error for cluster '{cluster_key}': {e}")
        return [{"rule_id": "ERR", "raw_response": raw}]

def generate_all_rules(
    sql_queries: list[str],
    schema: dict[str, TableMeta],
    api_key: str,
    model: str = "claude-opus-4-5",
    min_cluster_size: int = 2,
    output_path: str = "sql_rules.json",
) -> dict[str, list[dict]]:
    """
    sql_queries      — список сырых SQL-строк из логов
    schema           — словарь {table_name: TableMeta}
    api_key          — Anthropic API key
    min_cluster_size — кластеры меньше этого размера пропускаем
    output_path      — куда сохранить итоговый JSON
    """
    print(f"[1/4] Парсим {len(sql_queries)} запросов...")
    parsed = [parse_query(sql) for sql in sql_queries]
    errors = [p for p in parsed if p.parse_error]
    print(f"      Ошибок парсинга: {len(errors)}")

    print("[2/4] Дедуплицируем...")
    parsed = deduplicate(parsed)
    print(f"      Уникальных запросов: {len(parsed)}")

    print("[3/4] Кластеризуем по доменам...")
    clusters = cluster_by_domain(parsed)
    for key, qs in clusters.items():
        print(f"      {key}: {len(qs)} запросов")

    print("[4/4] Генерируем rules через LLM...")
    client = anthropic.Anthropic(api_key=api_key)
    all_rules: dict[str, list[dict]] = {}

    for cluster_key, qs in clusters.items():
        if cluster_key == "_unparsed":
            print(f"  [skip] {cluster_key}")
            continue
        if len(qs) < min_cluster_size:
            print(f"  [skip] {cluster_key} (слишком мало запросов: {len(qs)})")
            continue
        print(f"  Обрабатываем '{cluster_key}' ({len(qs)} запросов)...")
        rules = generate_rules_for_cluster(cluster_key, qs, schema, client, model)
        all_rules[cluster_key] = rules
        print(f"    → сгенерировано {len(rules)} rules")

    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(all_rules, f, ensure_ascii=False, indent=2)
    print(f"\nГотово! Результат сохранён в {output_path}")
    return all_rules

from sqlglot import parse_one, exp


# -------------------------------
# helpers
# -------------------------------

def extract_tables(tree):
    """
    alias -> {table, db, type}
    """
    tables = {}

    # --- CTE ---
    with_expr = tree.find(exp.With)
    if with_expr:
        for cte in with_expr.expressions:
            alias = cte.alias
            tables[alias] = {
                "table": alias,
                "db": None,
                "type": "CTE"
            }

    # --- TABLES + SUBQUERY ---
    for node in tree.walk():
        if isinstance(node, exp.Table):
            alias = node.alias_or_name
            tables[alias] = {
                "table": node.name,
                "db": node.db,
                "type": "TABLE"
            }

        elif isinstance(node, exp.Subquery):
            alias = node.alias
            if alias:
                tables[alias] = {
                    "table": None,
                    "db": None,
                    "type": "SUBQUERY"
                }

    return tables


def is_column(node):
    return isinstance(node, exp.Column)


def extract_column(node):
    return node.table, node.name


def extract_conditions(expr):
    """
    рекурсивно достаём только COLUMN = COLUMN
    """
    result = []

    if expr is None:
        return result

    if isinstance(expr, exp.EQ):
        l, r = expr.left, expr.right

        if is_column(l) and is_column(r):
            result.append((extract_column(l), extract_column(r)))

    elif isinstance(expr, (exp.And, exp.Or)):
        result += extract_conditions(expr.left)
        result += extract_conditions(expr.right)

    else:
        for arg in expr.args.values():
            if isinstance(arg, exp.Expression):
                result += extract_conditions(arg)
            elif isinstance(arg, list):
                for item in arg:
                    if isinstance(item, exp.Expression):
                        result += extract_conditions(item)

    return result

# MAIN

def extract_joins(sql: str):
    tree = parse_one(sql, read="mysql")

    alias_map = extract_tables(tree)
    results = []

    for join in tree.find_all(exp.Join):
        on_expr = join.args.get("on")

        conditions = extract_conditions(on_expr)

        if not conditions:
            continue

        # определяем алиасы таблиц из условий
        aliases = set()
        for (l, r) in conditions:
            aliases.add(l[0])
            aliases.add(r[0])

        if len(aliases) != 2:
            # пропускаем сложные случаи (3+ таблицы)
            continue

        a1, a2 = list(aliases)

        t1 = alias_map.get(a1, {})
        t2 = alias_map.get(a2, {})

        # --- формируем связи ---
        links = []
        for i, (l, r) in enumerate(conditions, start=1):
            if l[0] == a1:
                left_col = l[1]
                right_col = r[1]
            else:
                left_col = r[1]
                right_col = l[1]

            links.append({
                f"связь{i}": {
                    "поле1": left_col,
                    "поле2": right_col,
                    "предикат": "ON"
                }
            })

        # --- isOnlyTables ---
        is_only_tables = (
            t1.get("type") == "TABLE" and
            t2.get("type") == "TABLE"
        )

        result = {
            "таблица1": t1.get("table"),
            "алиас1": a1,
            "таблица2": t2.get("table"),
            "алиас2": a2,
            "кол-во связей": len(links),
            "join_side": join.side,
            "join_kind": join.kind,
            "isOnlyTables": is_only_tables
        }

        # добавляем связи внутрь
        for link in links:
            result.update(link)

        results.append(result)

    return results








def analyze_joins(sql: str):
    tree = sqlglot.parse_one(sql)

    # 1. Собираем все алиасы
    sources = [tree.find(exp.From).this] + [j.this for j in tree.find_all(exp.Join)]
    aliases = {}
    for src in sources:
        alias = src.alias or src.name
        full  = f"{src.args.get('db', '')}.{src.name}".lstrip(".")
        aliases[alias] = full

    # 2. Собираем связи
    links = []
    for join in tree.find_all(exp.Join):
        on_clause = join.args.get("on")
        if not on_clause:
            continue
        for eq in on_clause.find_all(exp.EQ):
            links.append((
                f"{eq.left.table}.{eq.left.name}",
                f"{eq.right.table}.{eq.right.name}",
            ))

    print("=== Таблицы ===")

import sqlglot
from sqlglot import exp

sql = """
SELECT *
FROM dwh.fact_sales   AS fs
JOIN dwh.dim_product  AS dp ON fs.product_id = dp.id
JOIN dwh.dim_customer AS dc ON get_json_string(dp.json, "customer_id") = dc.id
"""

tree = sqlglot.parse_one(sql, dialect="starrocks")

def extract_columns_from_expr(node) -> list[exp.Column]:
    """Рекурсивно достаём все exp.Column из выражения (включая внутри функций)."""
    return list(node.find_all(exp.Column))

def analyze_joins(tree):
    links = []

    for join in tree.find_all(exp.Join):
        on_clause = join.args.get("on")
        if not on_clause:
            continue

        for eq in on_clause.find_all(exp.EQ):
            left_cols  = extract_columns_from_expr(eq.left)
            right_cols = extract_columns_from_expr(eq.right)

            # Определяем "сырое" представление каждой стороны
            left_repr  = eq.left.sql()   # читаемо: get_json_string(dp.json, 'customer_id')
            right_repr = eq.right.sql()  # dc.id

            # Уникальные алиасы таблиц, задействованных в каждой стороне
            left_tables  = list(dict.fromkeys(c.table for c in left_cols  if c.table))
            right_tables = list(dict.fromkeys(c.table for c in right_cols if c.table))

            links.append({
                "left_expr":   left_repr,
                "left_tables": left_tables,
                "right_expr":  right_repr,
                "right_tables": right_tables,
                # Пара для графа связей (все комбинации)
                "pairs": [(lt, rt) for lt in left_tables for rt in right_tables],
            })

    return links

for link in analyze_joins(tree):

    result = {}

    """Заменяет алиасы таблиц на полные имена во всём выражении."""
    node = node.copy()

    return node.sql()

import sqlglot
from sqlglot import exp

sql = """
WITH cte_sales AS (
    SELECT * FROM dwh.fact_sales
)
SELECT *
FROM cte_sales AS fs
JOIN dwh.dim_product  AS dp ON fs.product_id = dp.id
JOIN dwh.dim_customer AS dc ON get_json_string(dp.json, "customer_id") = dc.id
JOIN (SELECT id FROM dwh.dim_region) AS dr ON dc.region_id = dr.id
"""

tree = sqlglot.parse_one(sql, dialect="starrocks")

def build_alias_map(tree) -> dict[str, str]:
    # Имена CTE — исключаем
    cte_names = {
        cte.alias for cte in tree.find_all(exp.CTE)
    }

    sources = [tree.find(exp.From).this] + [j.this for j in tree.find_all(exp.Join)]
    result = {}

    for src in sources:
        alias = src.alias or src.name

        # Пропускаем подзапросы (Subquery внутри FROM/JOIN)
        if isinstance(src, exp.Subquery):
            continue

        # Пропускаем если имя таблицы совпадает с CTE
        if src.name in cte_names:
            continue

        db   = src.args.get("db") or ""
        full = f"{db}.{src.name}" if db else src.name
        result[alias] = full

    return result

def resolve_expr(node, alias_map: dict) -> str:
    node = node.copy()
    for col in node.find_all(exp.Column):
        if col.table and col.table in alias_map:
            col.set("table", exp.Identifier(this=alias_map[col.table], quoted=False))
    return node.sql()

def analyze_joins(tree):
    alias_map = build_alias_map(tree)
    links = []

    for join in tree.find_all(exp.Join):
        on_clause = join.args.get("on")
        if not on_clause:
            continue

        for eq in on_clause.find_all(exp.EQ):
            left_cols  = list(eq.left.find_all(exp.Column))
            right_cols = list(eq.right.find_all(exp.Column))

            # Фильтруем алиасы, которых нет в alias_map (CTE / Subquery)
            left_tables = list(dict.fromkeys(
                alias_map[c.table]
                for c in left_cols
                if c.table and c.table in alias_map
            ))
            right_tables = list(dict.fromkeys(
                alias_map[c.table]
                for c in right_cols
                if c.table and c.table in alias_map
            ))

            # Пропускаем связь если обе стороны не разрешились в реальные таблицы
            if not left_tables and not right_tables:
                continue

            links.append({
                "left_expr":   resolve_expr(eq.left,  alias_map),
                "right_expr":  resolve_expr(eq.right, alias_map),
                "left_tables":  left_tables  or ["<cte_or_subquery>"],
                "right_tables": right_tables or ["<cte_or_subquery>"],
                "pairs":        [(lt, rt) for lt in (left_tables  or ["<cte_or_subquery>"])
                                          for rt in (right_tables or ["<cte_or_subquery>"])],
            })

    return links

for link in analyze_joins(tree):
    print(f"Выражение: {link['left_expr']}  =  {link['right_expr']}")
    print(f"Таблицы:   {link['left_tables']} <-> {link['right_tables']}")
    print(f"Связи:     {link['pairs']}")
    print()

CTE-фильтр — собираем все имена через tree.find_all(exp.CTE) и исключаем таблицы, чьё src.name попало в этот список.
Subquery-фильтр — isinstance(src, exp.Subquery) отсекает инлайн-подзапросы в FROM/JOIN.
В analyze_joins — дополнительная проверка c.table in alias_map при формировании left_tables/right_tables: алиасы CTE и Subquery просто не попадают в результат, заменяясь на <cte_or_subquery> для наглядности (можно заменить на None или пропускать совсем).​​​​​​​​​​​​​​​​

# ── 1. Получить одну задачу ──────────────────────────────────────────
def get_issue(issue_key: str) -> dict:
    """
    Возвращает поля задачи, включая description в rendered (HTML) виде.
    """
    url = f"{JIRA_URL}/rest/api/3/issue/{issue_key}"
    params = {
        "fields": "summary,description,status,assignee,comment",
        "expand": "renderedFields",   # ← даёт description как HTML
    }
    resp = requests.get(url, headers=headers, auth=auth, params=params)
    resp.raise_for_status()
    return resp.json()


# ── 2. Поиск задач по JQL ────────────────────────────────────────────
def search_issues(jql: str, fields: list[str] = None, max_results: int = 50) -> list[dict]:
    """
    Поиск через актуальный эндпоинт /rest/api/3/search/jql
    с поддержкой пагинации через nextPageToken.
    """
    url = f"{JIRA_URL}/rest/api/3/search/jql"
    fields = fields or ["summary", "description", "status", "assignee"]

    issues = []
    next_page_token = None

    while True:
        payload = {
            "jql": jql,
            "maxResults": max_results,
            "fields": fields,
            "expand": ["renderedFields"],
        }
        if next_page_token:
            payload["nextPageToken"] = next_page_token

        resp = requests.post(url, headers={**headers, "Content-Type": "application/json"},
                             auth=auth, json=payload)
        resp.raise_for_status()
        data = resp.json()

        issues.extend(data.get("issues", []))
        next_page_token = data.get("nextPageToken")

        if not next_page_token:
            break

    return issues


# ── 3. Извлечь HTML-описание и прогнать через наш парсер ─────────────
def get_issue_html(issue: dict) -> str | None:
    """
    renderedFields.description содержит HTML (если expand=renderedFields).
    """
    rendered = issue.get("renderedFields", {})
    return rendered.get("description")  # HTML-строка или None

def search_issues(keyword: str, max_results: int = 50) -> list[dict]:
    """
    text~ ищет по summary, description, comments, environment
    и всем текстовым custom fields одновременно.
    """
    url = f"{JIRA_URL}/rest/api/2/search"
    issues = []
    start_at = 0

    while True:
        payload = {
            "jql": f'text ~ "{keyword}" ORDER BY updated DESC',
            "startAt": start_at,
            "maxResults": max_results,
            "fields": [
                "summary", "description", "status", "assignee",
                "reporter", "project", "issuetype", "comment",
                "attachment", "labels", "environment",
            ],
            "expand": ["renderedFields"],
        }

        resp = requests.post(url, headers=headers, auth=auth, json=payload)
        resp.raise_for_status()
        data = resp.json()
        batch = data.get("issues", [])
        issues.extend(batch)

        # Пагинация
        start_at += len(batch)
        if start_at >= data.get("total", 0) or not batch:
            break

    return issues


# ═══════════════════════════════════════════════════════════

def search_in_worklogs(issue_key: str, keyword: str) -> list[dict]:

    resp = requests.get(url, headers=headers, auth=auth)
    resp.raise_for_status()

    matches = []
    for wl in resp.json().get("worklogs", []):
        comment = wl.get("comment", "")
        if keyword.lower() in comment.lower():

                "started":    wl.get("started", "")[:10],
                "time_spent": wl.get("timeSpent"),

# 4. ИЗВЛЕЧЕНИЕ ВХОЖДЕНИЙ ИЗ HTML-ПОЛЕЙ
#    (description, environment, comment — приходят как HTML)
# ═══════════════════════════════════════════════════════════

def extract_context_from_html(html: str, keyword: str) -> list[str]:
    """Возвращает список абзацев/ячеек содержащих keyword."""
    if not html:
        return []
    soup = BeautifulSoup(html, "html.parser")
    matches = []

    for el in soup.find_all(["p", "li", "td", "th", "h1", "h2", "h3", "pre"]):
        text = el.get_text(separator=" ", strip=True)
        if keyword.lower() in text.lower():
            highlighted = text.replace(keyword, f"[{keyword.upper()}]")
            tag = el.name.upper()
            matches.append(f"[{tag}] {highlighted}")

    return matches

def analyze_issue(issue: dict, keyword: str, include_worklogs: bool = True) -> dict | None:
    key     = issue["key"]
    fields  = issue["fields"]
    rendered = issue.get("renderedFields", {})

    result = {
        "key":       key,
        "summary":   fields.get("summary", ""),
        "project":   fields.get("project", {}).get("name"),
        "status":    fields.get("status", {}).get("name"),
        "assignee":  (fields.get("assignee") or {}).get("displayName", "—"),
        "reporter":  (fields.get("reporter") or {}).get("displayName", "—"),
        "areas":     {},
    }

    # ── Summary ──────────────────────────────────────────
    if keyword.lower() in result["summary"].lower():
        result["areas"]["summary"] = result["summary"]

    # ── Description ──────────────────────────────────────
    desc_html = rendered.get("description", "")
    desc_hits = extract_context_from_html(desc_html, keyword)
    if desc_hits:
        result["areas"]["description"] = desc_hits

    # ── Environment ──────────────────────────────────────
    env_html = rendered.get("environment", "")
    env_hits = extract_context_from_html(env_html, keyword)
    if env_hits:
        result["areas"]["environment"] = env_hits

    # ── Labels ───────────────────────────────────────────
    label_hits = [l for l in fields.get("labels", []) if keyword.lower() in l.lower()]
    if label_hits:
        result["areas"]["labels"] = label_hits

    # ── Comments ─────────────────────────────────────────
    comment_hits = []
    comments = rendered.get("comment", {}).get("comments", [])
    for c in comments:
        hits = extract_context_from_html(c.get("renderedBody", ""), keyword)
        if hits:
            comment_hits.append({
                "author":  c.get("author", {}).get("displayName"),
                "created": c.get("created", "")[:10],
                "hits":    hits,
            })
    if comment_hits:
        result["areas"]["comments"] = comment_hits

    # ── Attachments ──────────────────────────────────────
    att_hits = search_in_attachments(issue, keyword)
    if att_hits:
        result["areas"]["attachments"] = att_hits

    # ── Worklogs ─────────────────────────────────────────
    if include_worklogs:
        wl_hits = search_in_worklogs(key, keyword)
        if wl_hits:
            result["areas"]["worklogs"] = wl_hits

    # Если нет ни одного совпадения — пропускаем
    if not result["areas"]:
        return None

    return result

import re

def is_exact_match(text: str, keyword: str) -> bool:
    """
    Проверяет что keyword встречается как отдельное слово/идентификатор,
    а не как часть другого слова.
    Граница: пробел, пунктуация, начало/конец строки.
    """
    # \b не работает для snake_case идентификаторов с _ на границе,
    # поэтому используем явные разделители
    pattern = r'(?<![a-zA-Z0-9_])' + re.escape(keyword) + r'(?![a-zA-Z0-9_])'
    return bool(re.search(pattern, text, re.IGNORECASE))

def extract_context_from_html(html: str, keyword: str) -> list[str]:
    if not html:
        return []
    soup = BeautifulSoup(html, "html.parser")
    matches = []

    for el in soup.find_all(["p", "li", "td", "th", "h1", "h2", "h3", "pre"]):
        text = el.get_text(separator=" ", strip=True)
        if is_exact_match(text, keyword):       # ← было: keyword.lower() in text.lower()
            highlighted = re.sub(
                r'(?<![a-zA-Z0-9_])' + re.escape(keyword) + r'(?![a-zA-Z0-9_])',
                f"[{keyword.upper()}]",
                text,
                flags=re.IGNORECASE,
            )
            tag = el.name.upper()
            matches.append(f"[{tag}] {highlighted}")

    return matches

import sqlglot

from collections import Counter
from typing import List

def extract_tables(sql: str, dialect: str = "starrocks") -> List[str]:
    """Извлекает все таблицы из SQL запроса в формате db.table"""
    try:
        statements = sqlglot.parse(sql, dialect=dialect)
    except Exception:
        return []

    tables = []
    for statement in statements:
        if statement is None:
            continue
        for table in statement.find_all(exp.Table):
            # Пропускаем CTE-алиасы (сами определения WITH x AS (...))
            if isinstance(table.parent, exp.CTE):
                continue

            db = table.args.get("db")
            name = table.name

            if not name:
                continue

            if db:
                tables.append(f"{db.name}.{name}")
            else:
                tables.append(name)

    return tables


def count_tables(queries: List[str], dialect: str = "starrocks") -> Counter:
    """Считает сколько раз каждая таблица встречается во всех запросах"""
    counter = Counter()
    for sql in queries:
        tables = extract_tables(sql, dialect=dialect)
        counter.update(tables)
    return counter

def flush_to_doris(conn: pymysql.Connection, counter: Counter, batch_size: int):
    """Сбрасывает Counter в Doris батчами через INSERT ... ON DUPLICATE KEY UPDATE."""
    rows    = [(db, name, cnt) for (db, name), cnt in counter.items()]
    total   = len(rows)
    written = 0

    with conn.cursor() as cur:
        for i in range(0, total, batch_size):
            batch = rows[i : i + batch_size]
            placeholders = ", ".join(["(%s, %s, %s)"] * len(batch))
            flat_values  = [v for row in batch for v in row]

            sql = f"""
                INSERT INTO {TARGET_TABLE} (db, name, cnt)
                VALUES {placeholders}
                ON DUPLICATE KEY UPDATE cnt = cnt + VALUES(cnt)
            """
            cur.execute(sql, flat_values)
            written += len(batch)

    conn.commit()
    log.info(f"Записано в Doris: {written:,} / {total:,} уникальных таблиц")

# CREATE TABLE query_quality_stats (
#     query_id        VARCHAR(64),
#     query_time      DATETIME,
#     user            VARCHAR(128),
#     db              VARCHAR(128),
#     sql_clean       TEXT          REPLACE,      -- без комментариев
#     tables          TEXT          REPLACE,      -- JSON-список ["db.tbl", ...]
#     has_cte         BOOLEAN       REPLACE,
#     has_group_by    BOOLEAN       REPLACE,
#     has_limit       BOOLEAN       REPLACE,
#     has_window      BOOLEAN       REPLACE,
#     scan_bytes      BIGINT        REPLACE,
#     scan_rows       BIGINT        REPLACE,
#     return_rows     BIGINT        REPLACE,
#     cpu_cost_ns     BIGINT        REPLACE,
#     mem_cost_bytes  BIGINT        REPLACE,
#     score           INT           REPLACE
#  ) ENGINE=OLAP
#  UNIQUE KEY(query_id)
#  DISTRIBUTED BY HASH(query_id) BUCKETS 8
#  PROPERTIES ("replication_num" = "1");

# ─── Fetch ───────────────────────────────────────────────────────────────────

def get_total(config: dict) -> int:
    conn = pymysql.connect(**config)
    try:
        with conn.cursor() as cur:
            cur.execute(f"""
                SELECT COUNT(*)
                FROM {SOURCE_TABLE}
                WHERE stmt LIKE 'SELECT%'
                  AND state = 'EOF'
                  AND stmt IS NOT NULL
            """)
            return cur.fetchone()[0]
    finally:
        conn.close()


def fetch_page(config: dict, offset: int, limit: int) -> list:
    conn = pymysql.connect(**config)
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(f"""
                SELECT
                    queryId,
                    timestamp,
                    user,
                    db,
                    stmt,
                    scanBytes,
                    scanRows,
                    returnRows,
                    cpuCostNs,
                    memCostBytes
                FROM {SOURCE_TABLE}
                WHERE stmt LIKE 'SELECT%'
                  AND state = 'EOF'
                  AND stmt IS NOT NULL
                ORDER BY queryId
                LIMIT %s OFFSET %s
            """, (limit, offset))
            return cur.fetchall()
    finally:
        conn.close()


def iter_pages(config: dict, total: int, page_size: int) -> Iterator[list]:
    for offset in range(0, total, page_size):
        rows = fetch_page(config, offset, page_size)
        if not rows:
            break
        yield rows

# ─── Score ───────────────────────────────────────────────────────────────────
# Логика оценки:
#   Хороший эталонный запрос — тот который:
#     - читает много данных (высокий scanRows/scanBytes) → сложный, показательный
#     - возвращает мало строк (низкий returnRows/scanRows) → есть фильтрация/агрегация
#     - эффективен по CPU и памяти относительно объёма скана
#
#   Штрафуем за:
#     - огромный mem_cost (неэффективное использование памяти)
#     - returnRows ≈ scanRows (нет фильтрации — скорее всего SELECT *)
#     - нулевой скан (пустой/тривиальный запрос)

def log_scale(value: float, base: float = 1_000_000) -> float:
    """Нормализует большие числа в [0, 1] через логарифм."""
    if value <= 0:
        return 0.0
    return min(1.0, math.log1p(value) / math.log1p(base))


def compute_score(
    scan_bytes: int,
    scan_rows: int,
    return_rows: int,
    cpu_cost_ns: int,
    mem_cost_bytes: int,
) -> int:
    """
    Возвращает score [0, 100].

    Компоненты (веса):
      40% — complexity:   насколько много данных читает запрос (log-шкала по scanRows)
      30% — selectivity:  насколько хорошо фильтрует (1 - returnRows/scanRows)
      20% — cpu_eff:      CPU на строку скана (меньше — лучше, штраф за дорогие запросы)
      10% — mem_eff:      память на строку скана (меньше — лучше)
    """
    if scan_rows == 0:
        return 0  # тривиальный запрос — не эталон

    # 1. Complexity — хотим запросы с реальным сканом (эталоны должны быть нетривиальны)
    complexity = log_scale(scan_rows, base=10_000_000)  # 10M строк → score ~1.0

    # 2. Selectivity — хороший запрос возвращает << чем читает
    selectivity = 1.0 - min(1.0, return_rows / scan_rows)

    # 3. CPU efficiency — нормируем cpu_cost на scanRow, штрафуем дорогие
    cpu_per_row = cpu_cost_ns / scan_rows  # ns на строку
    # считаем что 1000 ns/row — граница нормы; выше — штраф
    cpu_eff = 1.0 - log_scale(cpu_per_row, base=100_000)

    # 4. Memory efficiency
    mem_per_row = mem_cost_bytes / scan_rows
    mem_eff = 1.0 - log_scale(mem_per_row, base=10_000)

    raw = (
        0.40 * complexity  +
        0.30 * selectivity +
        0.20 * cpu_eff     +
        0.10 * mem_eff
    )

    return max(0, min(100, round(raw * 100)))

# ─── SQL parsing ─────────────────────────────────────────────────────────────

def remove_comments(sql: str) -> str:
    """Убирает -- и /* */ комментарии через sqlglot tokenizer."""
    try:
        tokens = sqlglot.tokens.Tokenizer().tokenize(sql)
        clean_parts = []
        prev_end = 0
        for tok in tokens:
            if tok.token_type in (
                sqlglot.tokens.TokenType.COMMENT,
                sqlglot.tokens.TokenType.BLOCK_COMMENT,
            ):
                continue
            clean_parts.append(tok.text)
        return " ".join(clean_parts)
    except Exception:
        return sql


def parse_sql(sql: str) -> dict:
    """
    Возвращает:
      tables     : list[str]  — только реальные таблицы db.name (не CTE, не подзапрос-алиас)
      has_cte    : bool
      has_group_by: bool
      has_limit  : bool
      has_window : bool
    """
    result = {
        "tables":       [],
        "has_cte":      False,
        "has_group_by": False,
        "has_limit":    False,
        "has_window":   False,
    }
    try:
        stmts = sqlglot.parse(sql, dialect="starrocks", error_level=sqlglot.ErrorLevel.IGNORE)
    except Exception:
        return result

    for stmt in stmts:
        if stmt is None:
            continue

        # ── CTE ──────────────────────────────────────────────────────────────
        cte_names = set()
        for cte in stmt.find_all(exp.CTE):
            result["has_cte"] = True
            cte_names.add(cte.alias)  # имя CTE — не считаем реальной таблицей

        # ── GROUP BY ─────────────────────────────────────────────────────────
        if stmt.find(exp.Group):
            result["has_group_by"] = True

        # ── LIMIT ────────────────────────────────────────────────────────────
        if stmt.find(exp.Limit):
            result["has_limit"] = True

        # ── Window functions ─────────────────────────────────────────────────
        if stmt.find(exp.Window):
            result["has_window"] = True

        # ── Tables (только реальные) ─────────────────────────────────────────
        for table in stmt.find_all(exp.Table):
            # пропускаем объявление CTE
            if isinstance(table.parent, exp.CTE):
                continue
            name = table.name
            if not name:
                continue
            # пропускаем ссылки на CTE по имени
            if name in cte_names:
                continue
            db_node = table.args.get("db")
            full    = f"{db_node.name}.{name}" if db_node else name
            result["tables"].append(full)

    # дедупликация с сохранением порядка
    seen = set()
    result["tables"] = [t for t in result["tables"] if not (t in seen or seen.add(t))]
    return result


# ─── Worker ──────────────────────────────────────────────────────────────────

def process_batch(rows: list) -> list[dict]:
    """Обрабатывает батч строк из audit-таблицы, возвращает список готовых записей."""
    results = []
    for row in rows:
        sql_raw = row.get("stmt") or ""

        sql_clean   = remove_comments(sql_raw).strip()
        parsed      = parse_sql(sql_clean)

        scan_bytes     = int(row.get("scanBytes")     or 0)
        scan_rows      = int(row.get("scanRows")      or 0)
        return_rows    = int(row.get("returnRows")    or 0)
        cpu_cost_ns    = int(row.get("cpuCostNs")     or 0)
        mem_cost_bytes = int(row.get("memCostBytes")  or 0)

        score = compute_score(scan_bytes, scan_rows, return_rows, cpu_cost_ns, mem_cost_bytes)

        results.append({
            "query_id":       row.get("queryId", ""),
            "query_time":     str(row.get("timestamp", "")),
            "user":           row.get("user", ""),
            "db":             row.get("db", ""),
            "sql_clean":      sql_clean[:65_000],   # TEXT лимит Doris
            "tables":         json.dumps(parsed["tables"], ensure_ascii=False),
            "has_cte":        int(parsed["has_cte"]),
            "has_group_by":   int(parsed["has_group_by"]),
            "has_limit":      int(parsed["has_limit"]),
            "has_window":     int(parsed["has_window"]),
            "scan_bytes":     scan_bytes,
            "scan_rows":      scan_rows,
            "return_rows":    return_rows,
            "cpu_cost_ns":    cpu_cost_ns,
            "mem_cost_bytes": mem_cost_bytes,
            "score":          score,
        })
    return results

# ─── Write ───────────────────────────────────────────────────────────────────

def flush_to_doris(conn: pymysql.Connection, records: list[dict], batch_size: int):
    if not records:
        return

    columns = [
        "query_id", "query_time", "user", "db", "sql_clean",
        "tables", "has_cte", "has_group_by", "has_limit", "has_window",
        "scan_bytes", "scan_rows", "return_rows", "cpu_cost_ns", "mem_cost_bytes",
        "score",
    ]
    col_str = ", ".join(columns)

    with conn.cursor() as cur:
        for i in range(0, len(records), batch_size):
            batch        = records[i : i + batch_size]
            placeholders = ", ".join([f"({', '.join(['%s'] * len(columns))})"] * len(batch))
            flat_values  = [row[c] for row in batch for c in columns]
            cur.execute(f"INSERT INTO {TARGET_TABLE} ({col_str}) VALUES {placeholders}", flat_values)

    conn.commit()

# ─── Main ────────────────────────────────────────────────────────────────────

def main():
    log.info("Считаю строки в audit-таблице StarRocks...")
    total = get_total(STARROCKS_CONFIG)
    log.info(f"Всего SELECT запросов: {total:,} | воркеров: {N_WORKERS}")

    processed  = 0
    write_conn = pymysql.connect(**DORIS_CONFIG)

    try:
        with mp.Pool(processes=N_WORKERS) as pool:
            for batch_records in pool.imap_unordered(
                process_batch,
                iter_pages(STARROCKS_CONFIG, total, PAGE_SIZE),
                chunksize=1,
            ):
                flush_to_doris(write_conn, batch_records, WRITE_BATCH)
                processed += len(batch_records)

                if processed % LOG_EVERY == 0:
                    log.info(f"Обработано и записано: {processed:,} / {total:,}")
    finally:
        write_conn.close()

    log.info(f"Готово. Всего записано: {processed:,}")


if __name__ == "__main__":
    main()








import asyncio
import aiohttp
import json
import logging
from dataclasses import dataclass, field
from typing import Any
from datetime import datetime

logger = logging.getLogger(__name__)


# ─── Конфигурация ────────────────────────────────────────────────────────────

@dataclass
class LLMConfig:
    url: str
    api_key: str
    max_concurrent: int = 20      # параллельных запросов к LLM
    max_retries: int = 3
    retry_delays: list = field(default_factory=lambda: [5, 15, 30])  # сек
    timeout_sec: int = 120
    retry_on_status: set = field(default_factory=lambda: {429, 500, 502, 503, 504})


# ─── Результат обработки одной записи ────────────────────────────────────────

@dataclass
class RecordResult:
    record_id: Any
    success: bool
    data: dict | None = None
    error: str | None = None
    attempts: int = 0


# ─── Вызов LLM с retry ───────────────────────────────────────────────────────

async def call_llm_with_retry(
    session: aiohttp.ClientSession,
    record: dict,
    config: LLMConfig,
) -> RecordResult:
    record_id = record.get("id")
    last_error = None

    for attempt in range(1, config.max_retries + 1):
        try:
            payload = build_payload(record)  # твоя функция сборки промпта

            async with session.post(
                config.url,
                json=payload,
                timeout=aiohttp.ClientTimeout(total=config.timeout_sec),
            ) as resp:

                # HTTP-ошибки с retry
                if resp.status in config.retry_on_status:
                    last_error = f"HTTP {resp.status}"
                    logger.warning(f"[{record_id}] attempt {attempt}: {last_error}")
                    await _maybe_wait(attempt, config)
                    continue

                # Другие HTTP-ошибки — не ретраим
                if resp.status != 200:
                    return RecordResult(
                        record_id=record_id,
                        success=False,
                        error=f"HTTP {resp.status} (no retry)",
                        attempts=attempt,
                    )

                raw = await resp.json()

        except (aiohttp.ClientError, asyncio.TimeoutError) as e:
            last_error = str(e)
            logger.warning(f"[{record_id}] attempt {attempt}: network error: {e}")
            await _maybe_wait(attempt, config)
            continue

        # Парсим ответ — изолируем ошибки структуры
        try:
            parsed = parse_llm_response(raw)  # твоя функция парсинга
            logger.info(f"[{record_id}] OK after {attempt} attempt(s)")
            return RecordResult(
                record_id=record_id,
                success=True,
                data=parsed,
                attempts=attempt,
            )
        except (KeyError, ValueError, json.JSONDecodeError) as e:
            last_error = f"parse error: {e}"
            logger.warning(f"[{record_id}] attempt {attempt}: {last_error}, raw={str(raw)[:200]}")
            await _maybe_wait(attempt, config)
            continue

    return RecordResult(
        record_id=record_id,
        success=False,
        error=last_error,
        attempts=config.max_retries,
    )


async def _maybe_wait(attempt: int, config: LLMConfig):
    if attempt <= len(config.retry_delays):
        delay = config.retry_delays[attempt - 1]
        logger.info(f"  waiting {delay}s before retry...")
        await asyncio.sleep(delay)


# ─── Оркестратор с семафором ──────────────────────────────────────────────────

async def process_records(
    records: list[dict],
    config: LLMConfig,
) -> list[RecordResult]:
    semaphore = asyncio.Semaphore(config.max_concurrent)
    results: list[RecordResult] = []
    failed: list[RecordResult] = []

    async def bounded_call(record: dict) -> RecordResult:
        async with semaphore:
            return await call_llm_with_retry(session, record, config)

    connector = aiohttp.TCPConnector(limit=config.max_concurrent + 5)
    async with aiohttp.ClientSession(
        headers={"Authorization": f"Bearer {config.api_key}"},
        connector=connector,
    ) as session:
        tasks = [bounded_call(r) for r in records]

        # tqdm-прогресс (опционально: pip install tqdm)
        from tqdm.asyncio import tqdm
        for coro in tqdm.as_completed(tasks, total=len(tasks)):
            result = await coro
            results.append(result)
            if not result.success:
                failed.append(result)

    # Итог
    logger.info(f"Done: {len(results) - len(failed)}/{len(results)} OK, "
                f"{len(failed)} failed")
    for f in failed:
        logger.error(f"FAILED id={f.record_id}: {f.error} ({f.attempts} attempts)")

    return results


# ─── Точка входа ─────────────────────────────────────────────────────────────

def run_pipeline(records: list[dict], config: LLMConfig) -> list[RecordResult]:
    return asyncio.run(process_records(records, config))
