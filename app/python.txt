# --- Маппинг типов Oracle -> StarRocks ---
def map_type(data_type, length, precision, scale):
    dt = data_type.upper()

    if dt == 'NUMBER':
        if precision is None:
            return 'BIGINT'
        if scale and scale > 0:
            return f'DECIMAL({precision}, {scale})'
        if precision <= 3:
            return 'TINYINT'
        if precision <= 5:
            return 'SMALLINT'
        if precision <= 10:
            return 'INT'
        if precision <= 18:
            return 'BIGINT'
        return f'DECIMAL({precision}, {scale or 0})'

    if dt in ('VARCHAR2', 'NVARCHAR2', 'CHAR', 'NCHAR'):
        return f'VARCHAR({min(length or 255, 65533)})'

    if dt in ('DATE', 'TIMESTAMP') or dt.startswith('TIMESTAMP'):
        return 'DATETIME'

    if dt in ('CLOB', 'NCLOB', 'LONG', 'XMLTYPE'):
        return 'STRING'

    if dt in ('BLOB', 'RAW', 'LONG RAW'):
        return 'STRING'  # или VARBINARY — зависит от задачи

    if dt in ('FLOAT', 'BINARY_FLOAT'):
        return 'FLOAT'

    if dt == 'BINARY_DOUBLE':
        return 'DOUBLE'

    # fallback
    return 'STRING'

# --- Генерация CREATE TABLE для StarRocks ---
def generate_starrocks_ddl(table_name, columns, pk_columns):
    col_defs = []

    for col_name, data_type, length, precision, scale, nullable in columns:
        sr_type = map_type(data_type, length, precision, scale)
        null_clause = "" if nullable == 'Y' else " NOT NULL"
        col_defs.append(f"    {col_name} {sr_type}{null_clause}")

    # Если PK есть — используем его, иначе берём первую колонку
    key_cols = pk_columns if pk_columns else [columns[0][0]]
    key_type = "PRIMARY KEY" if pk_columns else "DUPLICATE KEY"

    ddl = f"CREATE TABLE {table_name} (\n"
    ddl += ",\n".join(col_defs)
    ddl += f"\n)\n"
    ddl += f"{key_type} ({', '.join(key_cols)})\n"
    ddl += f"DISTRIBUTED BY HASH({key_cols[0]}) BUCKETS 10\n"
    ddl += f"PROPERTIES (\"replication_num\" = \"1\");"

    return ddl

# --- Запуск ---

def embed(texts: list[str]) -> list[list[float]]:
    enc = tokenizer(texts, padding=True, truncation=True,
                    max_length=512, return_tensors="pt")
    with torch.no_grad():
        out = model(**enc)
        vecs = out.last_hidden_state[:, 0]
        vecs = torch.nn.functional.normalize(vecs, dim=-1)
    return vecs.tolist()

def insert_chunks(chunks: list[dict]):
    """
    chunk = {
        'id': 1,
        'doc_id': 'schema_orders',
        'source': 'orders.sql',
        'chunk_text': 'Таблица orders содержит...'

        for c, emb in zip(chunks, embeddings)
    ]

    with conn.cursor() as cur:
        cur.executemany(sql, rows)
    conn.commit()



import numpy as np

def search(query: str, source_filter: str = None, top_k: int = 5) -> list[dict]:
    # Эмбеддинг запроса
    q_vec = embed([query])[0]
    q_arr = json.dumps(q_vec)

    # Doris не поддерживает ANN нативно через SQL в ARRAY<FLOAT>,
    # поэтому считаем cosine similarity через встроенную функцию
    where = "WHERE 1=1"
    params = []

    if source_filter:
        where += " AND source = %s"
        params.append(source_filter)

    sql = f"""
        SELECT
            id,
            doc_id,
            source,
            chunk_text,
            cosine_distance(embedding, CAST({q_arr} AS ARRAY<FLOAT>)) AS score
        FROM chunks
        {where}
        ORDER BY score ASC          -- cosine_distance: меньше = ближе
        LIMIT %s
    """
    params.append(top_k)

    with conn.cursor(pymysql.cursors.DictCursor) as cur:
        cur.execute(sql, params)
        return cur.fetchall()

import requests

from urllib.parse import urljoin

BASE_URL = "https://your-confluence.local"
AUTH = ("username", "password")  # или token
HEADERS = {"Content-Type": "application/json"}


# =========================
# 1. CQL поиск (строгое совпадение)
# =========================
def build_cql_exact(term: str) -> str:
    # строгое совпадение через кавычки
    return f'type IN (page, comment) AND text ~ "\\"{term}\\"" ORDER BY lastmodified DESC'


def fetch_all_cql(cql: str):
    url = f"{BASE_URL}/rest/api/content/search"
    params = {
        "cql": cql,
        "limit": 100,
        "expand": "body.storage"
    }

    results = []

    while url:
        resp = requests.get(url, params=params, auth=AUTH, headers=HEADERS)
        resp.raise_for_status()
        data = resp.json()

        results.extend(data.get("results", []))

        next_link = data.get("_links", {}).get("next")
        if next_link:
            url = urljoin(BASE_URL, next_link)
            params = None  # дальше параметры уже внутри next
        else:
            url = None

    return results


# =========================
# 2. Получение комментариев страницы
# =========================
def fetch_comments(page_id: str):
    url = f"{BASE_URL}/rest/api/content/{page_id}/child/comment"
    params = {
        "limit": 100,
        "expand": "body.storage"
    }

    comments = []

    while url:
        resp = requests.get(url, params=params, auth=AUTH, headers=HEADERS)
        resp.raise_for_status()
        data = resp.json()

        comments.extend(data.get("results", []))

        next_link = data.get("_links", {}).get("next")
        if next_link:
            url = urljoin(BASE_URL, next_link)
            params = None
        else:
            url = None

    return comments


# =========================
# 3. HTML → структурированный текст
# =========================
def html_to_structured_text(html: str) -> str:
    soup = BeautifulSoup(html, "html.parser")

    # удаляем мусор
    for tag in soup(["script", "style"]):
        tag.decompose()

    # удаляем confluence макросы (ac:, ri:)
    for tag in soup.find_all():
        if ":" in tag.name:
            tag.decompose()

    lines = []

    def process(node, indent=0):
        if node.name in ["h1", "h2", "h3", "h4"]:
            lines.append("\n" + node.get_text(strip=True).upper())

        elif node.name == "p":
            lines.append("\n" + node.get_text(strip=True))

        elif node.name == "li":
            prefix = "  " * indent + "- "
            lines.append(prefix + node.get_text(strip=True))

        elif node.name in ["ul", "ol"]:
            for li in node.find_all("li", recursive=False):
                process(li, indent + 1)

        elif node.name == "pre":
            lines.append("\n[CODE]")
            lines.append(node.get_text())
            lines.append("[/CODE]")

        elif node.name == "code":
            lines.append(f"`{node.get_text(strip=True)}`")

    for elem in soup.find_all(recursive=False):
        process(elem)

    # fallback если структура не поймалась
    if not lines:
        return soup.get_text(separator="\n").strip()

    # чистим пустые строки
    return "\n".join([l for l in lines if l.strip()])


# =========================
# 4. Основной pipeline
# =========================
def search_and_extract(term: str):
    cql = build_cql_exact(term)
    results = fetch_all_cql(cql)

    output = []

    for item in results:
        content_type = item["type"]
        content_id = item["id"]
        title = item.get("title", "")

        html = item.get("body", {}).get("storage", {}).get("value", "")
        text = html_to_structured_text(html)

        output.append({
            "type": content_type,
            "id": content_id,
            "title": title,
            "text": text
        })

        # если это page → подтянуть комментарии
        if content_type == "page":
            comments = fetch_comments(content_id)

            for c in comments:
                c_html = c.get("body", {}).get("storage", {}).get("value", "")
                c_text = html_to_structured_text(c_html)

                output.append({
                    "type": "comment",
                    "parent_page_id": content_id,
                    "id": c["id"],
                    "text": c_text
                })

    return output

from bs4 import BeautifulSoup

def find_rows_by_keyword(html: str, keyword: str) -> list[dict]:
    """
    Ищет строки в таблицах HTML по ключевому слову.
    Возвращает список найденных строк с контекстом.
    """
    soup = BeautifulSoup(html, "html.parser")
    tables = soup.find_all("table")

    if not tables:
        print("Таблицы в документе не найдены.")
        return []

    print(f"Найдено таблиц: {len(tables)}")
    results = []

    for table_idx, table in enumerate(tables):
        rows = table.find_all("tr")

        # Извлекаем заголовок (первая строка)
        headers = []
        if rows:
            headers = [th.get_text(strip=True) for th in rows[0].find_all(["th", "td"])]

        for row_idx, row in enumerate(rows):
            cells = row.find_all(["th", "td"])
            cell_texts = [cell.get_text(strip=True) for cell in cells]
            row_text = " ".join(cell_texts)

            if keyword.lower() in row_text.lower():
                results.append({
                    "table_index": table_idx + 1,
                    "row_index": row_idx,
                    "headers": headers,
                    "cells": cell_texts,
                })

    return results


def print_results(results: list[dict], keyword: str):
    if not results:
        print(f'Строки с "{keyword}" не найдены.')
        return

    print(f'\nНайдено строк с "{keyword}": {len(results)}\n')
    print("=" * 60)

    for r in results:
        print(f"Таблица #{r['table_index']}, строка #{r['row_index']}")

        if r["headers"]:
            # Выводим как пары заголовок: значение
            pairs = zip(r["headers"], r["cells"])
            for header, value in pairs:
                marker = " ◄" if keyword.lower() in value.lower() else ""
                print(f"  {header or '—'}: {value}{marker}")
            # Если колонок больше чем заголовков
            for extra in r["cells"][len(r["headers"]):]:
                print(f"  —: {extra}")
        else:
            print("  " + " | ".join(r["cells"]))

        print("-" * 60)

from bs4 import BeautifulSoup, NavigableString, Tag


def split_into_sections(soup: BeautifulSoup) -> list[dict]:
    """
    Разбивает HTML на секции по заголовкам h1-h6.
    Каждая секция содержит заголовок и все элементы до следующего заголовка.
    """
    sections = []
    current_section = {"heading": None, "heading_level": 0, "elements": []}

    for element in soup.body.children if soup.body else soup.children:
        if not isinstance(element, Tag):
            continue

        if element.name in ("h1", "h2", "h3", "h4", "h5", "h6"):
            # Сохраняем предыдущую секцию если она не пустая
            if current_section["elements"] or current_section["heading"]:
                sections.append(current_section)

            current_section = {
                "heading": element.get_text(strip=True),
                "heading_level": int(element.name[1]),
                "elements": [],
            }
        else:
            current_section["elements"].append(element)

    # Последняя секция
    if current_section["elements"] or current_section["heading"]:
        sections.append(current_section)

    return sections


def extract_text_from_element(element: Tag) -> str:
    return element.get_text(separator=" ", strip=True)


def search_in_table(table: Tag, keyword: str, headers: list[str]) -> list[dict]:
    """Ищет keyword в строках таблицы."""
    matches = []
    rows = table.find_all("tr")

    # Определяем заголовки таблицы
    table_headers = headers
    if rows:
        first_row_cells = rows[0].find_all(["th", "td"])
        candidate_headers = [c.get_text(strip=True) for c in first_row_cells]
        # Считаем строку заголовком если есть <th> теги
        if rows[0].find("th"):
            table_headers = candidate_headers
            rows = rows[1:]

    for row_idx, row in enumerate(rows):
        cells = [td.get_text(strip=True) for td in row.find_all(["th", "td"])]
        row_text = " ".join(cells)

        if keyword.lower() in row_text.lower():
            if table_headers:
                row_data = dict(zip(table_headers, cells))
            else:
                row_data = {f"col_{i}": v for i, v in enumerate(cells)}

            matches.append({"row_index": row_idx, "data": row_data, "raw_cells": cells})

    return matches


def search_in_section(section: dict, keyword: str) -> list[dict]:
    """Ищет keyword во всех элементах секции."""
    results = []

    for element in section["elements"]:
        # Случай 1: таблица
        if element.name == "table":
            table_matches = search_in_table(element, keyword, headers=[])
            if table_matches:
                results.append({
                    "type": "table",
                    "matches": table_matches,
                })

        # Случай 2: вложенные таблицы (div/section внутри)
        elif element.name in ("div", "section", "article"):
            for table in element.find_all("table"):
                table_matches = search_in_table(table, keyword, headers=[])
                if table_matches:
                    results.append({
                        "type": "table",
                        "matches": table_matches,
                    })
            # И текст вне таблиц внутри div
            for child in element.children:
                if isinstance(child, Tag) and child.name != "table":
                    text = extract_text_from_element(child)
                    if keyword.lower() in text.lower():
                        results.append({
                            "type": "text",
                            "element": child.name,
                            "text": text,
                        })

        # Случай 3: обычный текстовый элемент
        else:
            text = extract_text_from_element(element)
            if keyword.lower() in text.lower():
                results.append({
                    "type": "text",
                    "element": element.name,
                    "text": text,
                })

    return results


def find_by_keyword(html: str, keyword: str) -> list[dict]:
    soup = BeautifulSoup(html, "html.parser")
    sections = split_into_sections(soup)

    all_results = []

    for section in sections:
        matches = search_in_section(section, keyword)
        if matches:
            all_results.append({
                "section_heading": section["heading"] or "(без заголовка)",
                "heading_level": section["heading_level"],
                "matches": matches,
            })

    return all_results


def print_results(results: list[dict], keyword: str):
    if not results:
        print(f'"{keyword}" не найдено.')
        return

    total = sum(len(r["matches"]) for r in results)
    print(f'Найдено вхождений "{keyword}": {total} в {len(results)} секции(ях)\n')

    for section_result in results:
        heading = section_result["section_heading"]
        level = section_result["heading_level"]
        prefix = "#" * level if level else "~"
        print(f"{prefix} {heading}")
        print("=" * 60)

        for match in section_result["matches"]:
            if match["type"] == "table":
                print("  [ТАБЛИЦА]")
                for row in match["matches"]:
                    print(f"  Строка {row['row_index'] + 1}:")
                    for col, val in row["data"].items():
                        marker = " ◄" if keyword.lower() in val.lower() else ""
                        print(f"    {col}: {val}{marker}")
            else:
                tag = match.get("element", "?")
                text = match["text"]
                # Подсвечиваем вхождение
                highlighted = text.replace(
                    keyword, f"[{keyword.upper()}]"
                )
                print(f"  [{tag.upper()}] {highlighted}")

        print()
