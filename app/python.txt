# ── Конфигурация Doris ────────────────────────────────────────────────────────
DORIS_HOST     = os.getenv("DORIS_HOST",     "localhost")

DORIS_DB       = os.getenv("DORIS_DB",       "test_db")
# ─────────────────────────────────────────────────────────────────────────────

mcp = FastMCP(
    name="doris-query-server",
    # stateless_http=True  # раскомментировать если нужен stateless режим
)


async def _execute(sql: str) -> dict:
    """Открывает соединение, выполняет запрос, возвращает результат."""
    conn = await aiomysql.connect(
        host=DORIS_HOST,
        port=DORIS_PORT,
        user=DORIS_USER,
        password=DORIS_PASSWORD,
        db=DORIS_DB,
        charset="utf8mb4",
        connect_timeout=10,
        cursorclass=aiomysql.DictCursor,
    )
    try:
        async with conn.cursor() as cursor:
            await cursor.execute(sql)
            rows = await cursor.fetchall()          # list[dict]
            columns = [d[0] for d in cursor.description] if cursor.description else []
    finally:
        conn.close()

    # Приводим не-JSON-сериализуемые типы (Decimal, date, …) к str
    safe_rows = [
        {k: (str(v) if v is not None else None) for k, v in row.items()}
        for row in rows
    ]
    return {"columns": columns, "rows": safe_rows, "row_count": len(safe_rows)}


@mcp.tool()
async def query_doris(sql: str) -> dict:
    """
    Выполняет SELECT-запрос в Apache Doris и возвращает результат.

    Args:
        sql: SQL-запрос (только SELECT; остальные операторы отклоняются).

    Returns:
        columns   — список названий колонок
        rows      — список строк (каждая строка — dict {колонка: значение})
        row_count — количество строк
    """
    if not sql.strip().upper().startswith("SELECT"):
        return {"error": "Разрешены только SELECT-запросы."}

    try:
        return await _execute(sql)
    except aiomysql.MySQLError as e:
        return {"error": f"Ошибка Doris: {e}"}
    except Exception as e:
        return {"error": f"Неожиданная ошибка: {e}"}


if __name__ == "__main__":
    # transport="streamable-http" → эндпоинт http://localhost:8000/mcp
    mcp.run(transport="streamable-http")


"""
MCP HTTP Client — тестирует инструмент query_doris через streamable-http
Запуск: python mcp_doris_client.py
Зависимости: pip install "mcp[cli]"
"""

import asyncio
import json
from mcp.client.streamable_http import streamablehttp_client
from mcp import ClientSession

SERVER_URL = "http://localhost:8000/mcp"

TEST_QUERIES = [
    "SELECT 1 AS ping",
    "SELECT NOW() AS server_time",
    "SELECT DATABASE() AS current_db",
    "SELECT * FROM information_schema.tables LIMIT 5",
]


async def run_client() -> None:
    print(f"Подключаемся к MCP-серверу: {SERVER_URL}")
    print("=" * 60)

    async with streamablehttp_client(SERVER_URL) as (read, write, _):
        async with ClientSession(read, write) as session:

            await session.initialize()
            print("✅ Сессия инициализирована\n")

            # Список доступных инструментов
            tools = (await session.list_tools()).tools
            print("Доступные инструменты:")
            for t in tools:
                print(f"  • {t.name}: {t.description}")
            print()

            # Прогоняем тестовые запросы
            for sql in TEST_QUERIES:
                print(f"─── SQL: {sql}")
                result = await session.call_tool("query_doris", {"sql": sql})

                for block in result.content:
                    if block.type == "text":
                        try:
                            data = json.loads(block.text)
                        except json.JSONDecodeError:
                            print(f"  Raw: {block.text}")
                            continue

                        if "error" in data:
                            print(f"  ⚠️  {data['error']}")
                        else:
                            print(f"  Колонки : {data['columns']}")
                            print(f"  Строк   : {data['row_count']}")
                            for row in data["rows"]:
                                print(f"            {row}")
                print()

    print("=" * 60)
    print("Клиент завершил работу.")


if __name__ == "__main__":
    asyncio.run(run_client())

# collector.py
import psycopg2
import pymysql
import pandas as pd
import time
import os

PG = psycopg2.connect("postgresql://user:pass@localhost/bench")
SR = dict(host="localhost", port=9030, user="root", password="")

PHASE = os.getenv("BENCH_PHASE", "unknown")
CATALOG = os.getenv("ICEBERG_CATALOG", "iceberg_ods")
DB      = os.getenv("ICEBERG_DB",      "test_db")
TABLE   = os.getenv("ICEBERG_TABLE",   "ANALOGS_BY_PROMOTED_AGG_DAILY")

def q(sql):
   conn = pymysql.connect(**SR)
   df = pd.read_sql(sql, conn)
   conn.close()
   return df

def collect():
   sr_conn = pymysql.connect(**SR)
   full_table = f"{CATALOG}.{DB}.{TABLE}"

   # ----------------------------------------------------------------
   # 1. Латентность запросов из audit log
   # ----------------------------------------------------------------
   lat = pd.read_sql(f"""
       SELECT
           PERCENTILE_APPROX(query_time_ms, 0.95) AS p95,
           PERCENTILE_APPROX(query_time_ms, 0.99) AS p99,
           AVG(query_time_ms)                      AS avg_ms,
           COUNT(*)                                AS qps
       FROM starrocks_audit_db__.starrocks_audit_tbl__
       WHERE query_time > NOW() - INTERVAL 2 MINUTE
         AND stmt LIKE '%{TABLE}%'
   """, sr_conn)

   # ----------------------------------------------------------------
   # 2. Файлы: data / delete / общее
   # ----------------------------------------------------------------
   files = pd.read_sql(f"""
       SELECT
           file_content,
           COUNT(*)                        AS cnt,
           SUM(file_size_in_bytes)         AS total_bytes,
           AVG(file_size_in_bytes)         AS avg_bytes,
           SUM(record_count)               AS total_records
       FROM {full_table}$files
       GROUP BY file_content
   """, sr_conn)
   # file_content: 0=data, 1=position_deletes, 2=equality_deletes

   data_row  = files[files.file_content == 0].iloc[0] if not files[files.file_content == 0].empty else None
   del_row   = files[files.file_content == 1].iloc[0] if not files[files.file_content == 1].empty else None
   eq_row    = files[files.file_content == 2].iloc[0] if not files[files.file_content == 2].empty else None

   total_files      = int(files.cnt.sum())
   total_size_bytes = int(files.total_bytes.sum())

   data_files       = int(data_row.cnt)          if data_row is not None else 0
   data_size_bytes  = int(data_row.total_bytes)  if data_row is not None else 0
   data_avg_bytes   = int(data_row.avg_bytes)    if data_row is not None else 0
   data_records     = int(data_row.total_records)if data_row is not None else 0
   del_files        = int(del_row.cnt)           if del_row  is not None else 0
   eq_del_files     = int(eq_row.cnt)            if eq_row   is not None else 0

   # ----------------------------------------------------------------
   # 3. Партиции
   # ----------------------------------------------------------------
   parts = pd.read_sql(f"""
       SELECT COUNT(DISTINCT partition) AS partition_cnt
       FROM {full_table}$partitions
   """, sr_conn)
   partition_cnt = int(parts.partition_cnt.iloc[0])

   # ----------------------------------------------------------------
   # 4. Snapshots
   # ----------------------------------------------------------------
   snaps = pd.read_sql(f"""
       SELECT COUNT(*) AS cnt
       FROM {full_table}$snapshots
   """, sr_conn)
   snapshot_cnt = int(snaps.cnt.iloc[0])

   # ----------------------------------------------------------------
   # 5. Манифесты (metadata)
   # ----------------------------------------------------------------
   manifests = pd.read_sql(f"""
       SELECT
           COUNT(*)                AS manifest_cnt,
           SUM(length)             AS manifest_total_bytes,
           SUM(added_files_count)  AS added_files,
           SUM(deleted_files_count)AS deleted_files,
           SUM(existing_files_count) AS existing_files
       FROM {full_table}$manifests
   """, sr_conn)
   manifest_cnt        = int(manifests.manifest_cnt.iloc[0])
   manifest_bytes      = int(manifests.manifest_total_bytes.iloc[0] or 0)

   # ----------------------------------------------------------------
   # 6. Metadata JSON файлы — считаем через $history
   #    (каждый snapshot = один metadata JSON)
   # ----------------------------------------------------------------
   meta_json = pd.read_sql(f"""
       SELECT COUNT(DISTINCT snapshot_id) AS meta_json_cnt
       FROM {full_table}$history
   """, sr_conn)
   meta_json_cnt = int(meta_json.meta_json_cnt.iloc[0])

   sr_conn.close()

   # ----------------------------------------------------------------
   # 7. Пишем в Postgres
   # ----------------------------------------------------------------
   with PG.cursor() as cur:
       cur.execute("""
           INSERT INTO bench_metrics (
               phase,
               -- latency
               p95_ms, p99_ms, avg_ms, qps,
               -- files summary
               total_files, total_size_bytes,
               -- data files
               data_files, data_size_bytes, data_avg_bytes, data_records,
               -- delete files
               del_files, eq_del_files,
               -- partitions
               partition_cnt,
               -- snapshots
               snapshot_cnt,
               -- metadata
               manifest_cnt, manifest_bytes, meta_json_cnt
           ) VALUES (
               %s,
               %s, %s, %s, %s,
               %s, %s,
               %s, %s, %s, %s,
               %s, %s,
               %s,
               %s,
               %s, %s, %s
           )
       """, (
           PHASE,
           float(lat.p95.iloc[0]  or 0),
           float(lat.p99.iloc[0]  or 0),
           float(lat.avg_ms.iloc[0] or 0),
           int(lat.qps.iloc[0]    or 0),
           total_files, total_size_bytes,
           data_files, data_size_bytes, data_avg_bytes, data_records,
           del_files, eq_del_files,
           partition_cnt,
           snapshot_cnt,
           manifest_cnt, manifest_bytes, meta_json_cnt,
       ))
       PG.commit()
   print(f"[OK] {PHASE} | P95={lat.p95.iloc[0]:.0f}ms"
         f" | files={total_files} | del={del_files}"
         f" | parts={partition_cnt} | snaps={snapshot_cnt}")

if __name__ == "__main__":
   while True:
       try:
           collect()
       except Exception as e:
           print(f"[ERROR] {e}")
       time.sleep(30)

--------------------------
Читает метаданные Iceberg напрямую из S3 (без Spark/catalog),

# ─────────────────────────── Avro minimal reader ────────────────────────────
# Iceberg manifest files — это Avro. Тащить fastavro не хочется,
# поэтому реализуем минимальный reader для нужных полей.

AVRO_MAGIC = b"Obj\x01"


def _read_long(buf: io.BytesIO) -> int:
    """ZigZag + VarInt decode."""
    b, n, shift = 0, 0, 0
    while True:
        byte = buf.read(1)
        if not byte:
            raise EOFError("unexpected EOF in avro long")
        byte = byte[0]
        n |= (byte & 0x7F) << shift
        shift += 7
        if not (byte & 0x80):
            break
    return (n >> 1) ^ -(n & 1)


def _read_bytes(buf: io.BytesIO) -> bytes:
    n = _read_long(buf)
    return buf.read(n)


def _read_string(buf: io.BytesIO) -> str:
    return _read_bytes(buf).decode("utf-8", errors="replace")


def _skip_value(buf: io.BytesIO, schema, named_types: dict):
    """Пропускаем значение по схеме Avro."""
    if isinstance(schema, list):
        idx = _read_long(buf)
        _skip_value(buf, schema[idx], named_types)
        return
    if isinstance(schema, dict):
        t = schema.get("type")
        if t == "record":
            for f in schema["fields"]:
                _skip_value(buf, f["type"], named_types)
            return
        if t == "array":
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    _skip_value(buf, schema["items"], named_types)
            return
        if t == "map":
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    _read_string(buf)
                    _skip_value(buf, schema["values"], named_types)
            return
        if t in ("bytes", "fixed"):
            if t == "fixed":
                buf.read(schema["size"])
            else:
                _read_bytes(buf)
            return
        if t == "string":
            _read_string(buf)
            return
        if t in ("int", "long"):
            _read_long(buf)
            return
        if t in ("float",):
            buf.read(4)
            return
        if t in ("double",):
            buf.read(8)
            return
        if t in ("boolean",):
            buf.read(1)
            return
        if t == "null":
            return
        if t == "enum":
            _read_long(buf)
            return
        # named type ref
        if isinstance(t, str) and t in named_types:
            _skip_value(buf, named_types[t], named_types)
            return
    if isinstance(schema, str):
        if schema in named_types:
            _skip_value(buf, named_types[schema], named_types)
            return
        if schema == "string":
            _read_string(buf)
        elif schema in ("int", "long"):
            _read_long(buf)
        elif schema == "float":
            buf.read(4)
        elif schema == "double":
            buf.read(8)
        elif schema == "bytes":
            _read_bytes(buf)
        elif schema == "boolean":
            buf.read(1)
        elif schema == "null":
            pass


def _collect_named(schema, named_types: dict):
    if isinstance(schema, list):
        for s in schema:
            _collect_named(s, named_types)
    elif isinstance(schema, dict):
        if schema.get("type") == "record":
            named_types[schema["name"]] = schema
            for f in schema.get("fields", []):
                _collect_named(f["type"], named_types)
        elif schema.get("type") in ("array", "map"):
            _collect_named(schema.get("items", schema.get("values", {})), named_types)


def _read_record_field(buf: io.BytesIO, field_schema, named_types: dict, want: set):
    """Читаем одно поле записи. Возвращаем значение если имя в want, иначе None."""
    name = field_schema["name"]
    ftype = field_schema["type"]
    if name not in want:
        _skip_value(buf, ftype, named_types)
        return name, None

    # читаем реально
    return name, _read_typed(buf, ftype, named_types)


def _read_typed(buf: io.BytesIO, schema, named_types: dict):
    if isinstance(schema, list):
        idx = _read_long(buf)
        actual = schema[idx]
        if actual == "null":
            return None
        return _read_typed(buf, actual, named_types)
    if isinstance(schema, str):
        if schema in named_types:
            return _read_typed(buf, named_types[schema], named_types)
        if schema == "string":
            return _read_string(buf)
        if schema in ("int", "long"):
            return _read_long(buf)
        if schema == "float":
            return struct.unpack(">f", buf.read(4))[0]
        if schema == "double":
            return struct.unpack(">d", buf.read(8))[0]
        if schema == "bytes":
            return _read_bytes(buf)
        if schema == "boolean":
            return bool(buf.read(1)[0])
        if schema == "null":
            return None
        return None
    if isinstance(schema, dict):
        t = schema.get("type")
        if t == "record":
            result = {}
            for f in schema["fields"]:
                result[f["name"]] = _read_typed(buf, f["type"], named_types)
            return result
        if t == "string":
            return _read_string(buf)
        if t in ("int", "long"):
            return _read_long(buf)
        if t == "float":
            return struct.unpack(">f", buf.read(4))[0]
        if t == "double":
            return struct.unpack(">d", buf.read(8))[0]
        if t in ("bytes",):
            return _read_bytes(buf)
        if t == "fixed":
            return buf.read(schema["size"])
        if t == "boolean":
            return bool(buf.read(1)[0])
        if t == "null":
            return None
        if t == "enum":
            idx = _read_long(buf)
            return schema["symbols"][idx]
        if t == "array":
            items = []
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    items.append(_read_typed(buf, schema["items"], named_types))
            return items
        if t == "map":
            result = {}
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    k = _read_string(buf)
                    result[k] = _read_typed(buf, schema["values"], named_types)
            return result
        if isinstance(t, str) and t in named_types:
            return _read_typed(buf, named_types[t], named_types)
    return None


def parse_avro_manifest(data: bytes, want_fields: set) -> list[dict]:
    """
    Минимальный Avro-парсер: читает только нужные поля (want_fields)
    из каждой записи, остальное пропускает.
    """
    buf = io.BytesIO(data)
    magic = buf.read(4)
    if magic != AVRO_MAGIC:
        raise ValueError("Not an Avro file")

    # Читаем мета-блок
    meta = {}
    while True:
        cnt = _read_long(buf)
        if cnt == 0:
            break
        if cnt < 0:
            _read_long(buf)
            cnt = -cnt
        for _ in range(cnt):
            k = _read_string(buf)
            v = _read_bytes(buf)
            meta[k] = v

    codec = meta.get("avro.codec", b"null").decode()
    schema_str = meta.get("avro.schema", b"{}").decode()
    schema = json.loads(schema_str)
    sync_marker = buf.read(16)

    named_types: dict = {}
    _collect_named(schema, named_types)

    records = []
    while True:
        try:
            obj_count = _read_long(buf)
        except EOFError:
            break
        if obj_count == 0:
            break
        block_size = _read_long(buf)
        block_data = buf.read(block_size)
        block_sync = buf.read(16)
        if block_sync != sync_marker:
            raise ValueError("Avro sync marker mismatch")

        if codec == "deflate":
            block_data = zlib.decompress(block_data, -15)
        elif codec == "null":
            pass
        else:
            raise ValueError(f"Unsupported avro codec: {codec}")

        bbuf = io.BytesIO(block_data)
        for _ in range(obj_count):
            rec = {}
            if isinstance(schema, dict) and schema.get("type") == "record":
                for f in schema["fields"]:
                    name, val = _read_record_field(bbuf, f, named_types, want_fields)
                    if val is not None or name in want_fields:
                        rec[name] = val
            records.append(rec)

    return records


# ──────────────────────────── S3 helpers ────────────────────────────────────

def s3_url_to_parts(url: str) -> tuple[str, str]:
    """s3://bucket/prefix/key → (bucket, prefix/key)"""
    url = url.strip()
    if url.startswith("s3://") or url.startswith("s3a://"):
        without = re.sub(r"^s3a?://", "", url)
        parts = without.split("/", 1)
        bucket = parts[0]
        key = parts[1] if len(parts) > 1 else ""
        return bucket, key
    raise ValueError(f"Not an s3/s3a URL: {url}")


def get_s3_client(endpoint_url: Optional[str], profile: Optional[str]) -> boto3.client:
    session = boto3.Session(profile_name=profile) if profile else boto3.Session()
    kwargs = {"service_name": "s3"}
    if endpoint_url:
        kwargs["endpoint_url"] = endpoint_url
    return session.client(**kwargs)


def s3_get_object_bytes(s3, bucket: str, key: str) -> bytes:
    resp = s3.get_object(Bucket=bucket, Key=key)
    return resp["Body"].read()


def s3_list_all_objects(s3, bucket: str, prefix: str) -> dict[str, int]:
    """Возвращает {key: size_bytes} для всех объектов под prefix."""
    result = {}
    paginator = s3.get_paginator("list_objects_v2")
    for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get("Contents", []):
            result[obj["Key"]] = obj["Size"]
    return result


# ──────────────────────────── Iceberg metadata ──────────────────────────────

def find_metadata_json(s3, bucket: str, metadata_prefix: str) -> str:
    """
    Находим последний version-hint.text или последний v*.metadata.json
    """
    # Пробуем version-hint.text
    hint_key = metadata_prefix.rstrip("/") + "/version-hint.text"
    try:
        data = s3_get_object_bytes(s3, bucket, hint_key)
        version = int(data.strip())
        meta_key = metadata_prefix.rstrip("/") + f"/v{version}.metadata.json"
        print(f"  [hint] version={version} → {meta_key}")
        return meta_key
    except (ClientError, ValueError):
        pass

    # Ищем последний v*.metadata.json по имени
    paginator = s3.get_paginator("list_objects_v2")
    candidates = []
    for page in paginator.paginate(Bucket=bucket, Prefix=metadata_prefix.rstrip("/") + "/"):
        for obj in page.get("Contents", []):
            k = obj["Key"]
            if re.search(r"v\d+\.metadata\.json$", k):
                candidates.append((k, obj["LastModified"]))

    if not candidates:
        raise FileNotFoundError(f"No metadata.json found under s3://{bucket}/{metadata_prefix}")

    candidates.sort(key=lambda x: x[1])
    chosen = candidates[-1][0]
    print(f"  [scan] latest metadata → {chosen}")
    return chosen


@dataclass
class IcebergFile:
    path: str
    size: int
    file_type: str   # DATA or EQUALITY_DELETES / POSITION_DELETES
    status: str      # ADDED / EXISTING / DELETED
    snapshot_id: Optional[int] = None


@dataclass
class SnapshotInfo:
    snapshot_id: int
    parent_snapshot_id: Optional[int]
    timestamp_ms: int
    operation: str
    manifest_list_path: str


@dataclass
class AuditResult:
    table_name: str
    table_location: str
    current_snapshot_id: Optional[int]
    format_version: int
    snapshots: list[SnapshotInfo]

    # Файлы, видимые в ТЕКУЩЕМ снапшоте (live)
    live_data_files: dict[str, IcebergFile] = field(default_factory=dict)
    live_delete_files: dict[str, IcebergFile] = field(default_factory=dict)

    # Файлы из всех снапшотов (включая исторические)
    all_referenced_files: dict[str, IcebergFile] = field(default_factory=dict)

    # Metadata-файлы (manifest lists, manifests, metadata.json)
    metadata_files: set[str] = field(default_factory=set)

    # Реальные объекты в S3 (все под table_location)
    s3_all_objects: dict[str, int] = field(default_factory=dict)


# ────────────────────────────── Core parser ─────────────────────────────────

class IcebergAuditor:

    def __init__(self, s3, bucket: str):
        self.s3 = s3
        self.bucket = bucket

    def _get_json(self, key: str) -> dict:
        data = s3_get_object_bytes(self.s3, self.bucket, key)
        return json.loads(data)

    def _key_from_path(self, path: str) -> str:
        _, key = s3_url_to_parts(path)
        return key

    def audit_table(self, table_prefix: str) -> AuditResult:
        """
        table_prefix — путь в S3 до корня таблицы,
        например: 'warehouse/db/my_table'
        """
        metadata_prefix = table_prefix.rstrip("/") + "/metadata"

        print(f"\n{'='*60}")
        print(f"Auditing s3://{self.bucket}/{table_prefix}")
        print(f"{'='*60}")

        # 1. Находим metadata.json
        meta_key = find_metadata_json(self.s3, self.bucket, metadata_prefix)
        meta = self._get_json(meta_key)

        format_version = meta.get("format-version", 1)
        table_uuid = meta.get("table-uuid", "unknown")
        location = meta.get("location", f"s3://{self.bucket}/{table_prefix}")
        current_snapshot_id = meta.get("current-snapshot-id")

        print(f"  format-version : {format_version}")
        print(f"  table-uuid     : {table_uuid}")
        print(f"  location       : {location}")
        print(f"  current snap   : {current_snapshot_id}")

        result = AuditResult(
            table_name=table_prefix.split("/")[-1],
            table_location=location,
            current_snapshot_id=current_snapshot_id,
            format_version=format_version,
            snapshots=[],
        )

        # Регистрируем metadata.json как metadata-файл
        result.metadata_files.add(meta_key)

        # 2. Собираем все снапшоты
        snapshots_raw = meta.get("snapshots", [])
        snap_map: dict[int, dict] = {s["snapshot-id"]: s for s in snapshots_raw}

        for s in snapshots_raw:
            si = SnapshotInfo(
                snapshot_id=s["snapshot-id"],
                parent_snapshot_id=s.get("parent-snapshot-id"),
                timestamp_ms=s.get("timestamp-ms", 0),
                operation=s.get("summary", {}).get("operation", "unknown"),
                manifest_list_path=s.get("manifest-list", ""),
            )
            result.snapshots.append(si)

        print(f"  snapshots total: {len(result.snapshots)}")

        # 3. Читаем все manifest lists → manifests → files
        for si in result.snapshots:
            if not si.manifest_list_path:
                continue
            is_current = (si.snapshot_id == current_snapshot_id)
            self._process_manifest_list(si, result, is_current)

        # 4. Листаем S3 (все объекты под table_prefix)
        print(f"\n  Listing all S3 objects under prefix...")
        result.s3_all_objects = s3_list_all_objects(
            self.s3, self.bucket, table_prefix.rstrip("/") + "/"
        )
        print(f"  S3 objects found: {len(result.s3_all_objects)}")

        return result

    def _process_manifest_list(self, si: SnapshotInfo, result: AuditResult, is_current: bool):
        try:
            ml_key = self._key_from_path(si.manifest_list_path)
            ml_data = s3_get_object_bytes(self.s3, self.bucket, ml_key)
            result.metadata_files.add(ml_key)
        except ClientError as e:
            print(f"  [WARN] Cannot read manifest list {si.manifest_list_path}: {e}")
            return

        # Manifest list — тоже Avro
        try:
            entries = parse_avro_manifest(
                ml_data,
                want_fields={"manifest_path", "manifest_length", "content"}
            )
        except Exception as e:
            print(f"  [WARN] Failed to parse manifest list {ml_key}: {e}")
            return

        for entry in entries:
            mpath = entry.get("manifest_path", "")
            if not mpath:
                continue
            content = entry.get("content", 0)  # 0=DATA, 1=DELETES
            self._process_manifest(mpath, content, result, is_current, si.snapshot_id)

    def _process_manifest(
        self,
        manifest_path: str,
        content_type: int,
        result: AuditResult,
        is_current: bool,
        snapshot_id: int,
    ):
        try:
            m_key = self._key_from_path(manifest_path)
            m_data = s3_get_object_bytes(self.s3, self.bucket, m_key)
            result.metadata_files.add(m_key)
        except ClientError as e:
            print(f"  [WARN] Cannot read manifest {manifest_path}: {e}")
            return

        try:
            entries = parse_avro_manifest(
                m_data,
                want_fields={"data_file", "status", "file_path", "file_size_in_bytes", "content"}
            )
        except Exception as e:
            print(f"  [WARN] Failed to parse manifest {m_key}: {e}")
            return

        # STATUS: 0=EXISTING, 1=ADDED, 2=DELETED
        STATUS_MAP = {0: "EXISTING", 1: "ADDED", 2: "DELETED"}
        CONTENT_MAP = {0: "DATA", 1: "POSITION_DELETES", 2: "EQUALITY_DELETES"}

        for entry in entries:
            status_code = entry.get("status", 0)
            status = STATUS_MAP.get(status_code, "UNKNOWN")

            data_file = entry.get("data_file")
            if not data_file or not isinstance(data_file, dict):
                continue

            fpath = data_file.get("file_path", "")
            fsize = data_file.get("file_size_in_bytes", 0) or 0
            fcontent = data_file.get("content", content_type)
            ftype = CONTENT_MAP.get(fcontent, "DATA")

            if not fpath:
                continue

            try:
                fkey = self._key_from_path(fpath)
            except ValueError:
                fkey = fpath  # relative path

            icefile = IcebergFile(
                path=fkey,
                size=fsize,
                file_type=ftype,
                status=status,
                snapshot_id=snapshot_id,
            )

            # Регистрируем в all_referenced
            result.all_referenced_files[fkey] = icefile

            # В live только если текущий снапшот и не DELETED
            if is_current and status != "DELETED":
                if ftype == "DATA":
                    result.live_data_files[fkey] = icefile
                else:
                    result.live_delete_files[fkey] = icefile

def fmt_size(b: int) -> str:
    for unit in ["B", "KB", "MB", "GB", "TB"]:
        if b < 1024:
            return f"{b:.2f} {unit}"
        b /= 1024
    return f"{b:.2f} PB"


def print_report(result: AuditResult):
    sep = "─" * 70

    print(f"\n{'═'*70}")
    print(f"  ICEBERG S3 STORAGE AUDIT: {result.table_name}")
    print(f"{'═'*70}")
    print(f"  Table location  : {result.table_location}")
    print(f"  Format version  : {result.format_version}")
    print(f"  Current snapshot: {result.current_snapshot_id}")
    print(f"  Total snapshots : {len(result.snapshots)}")

    # ── Снапшоты ──
    print(f"\n{sep}")
    print("  SNAPSHOTS")
    print(sep)
    for si in sorted(result.snapshots, key=lambda x: x.timestamp_ms):
        ts = datetime.fromtimestamp(si.timestamp_ms / 1000, tz=timezone.utc).strftime("%Y-%m-%d %H:%M:%S")
        marker = " ◄ CURRENT" if si.snapshot_id == result.current_snapshot_id else ""
        print(f"  [{ts}] snap={si.snapshot_id} op={si.operation}{marker}")

    # ── Live файлы (текущий снапшот) ──
    live_data_size = sum(f.size for f in result.live_data_files.values())
    live_del_size = sum(f.size for f in result.live_delete_files.values())
    live_total = live_data_size + live_del_size

    print(f"\n{sep}")
    print("  LIVE FILES (current snapshot)")
    print(sep)
    print(f"  Data files      : {len(result.live_data_files):>6}   {fmt_size(live_data_size):>12}")
    print(f"  Delete files    : {len(result.live_delete_files):>6}   {fmt_size(live_del_size):>12}")
    print(f"  {'':─<40}")
    print(f"  LIVE TOTAL      : {len(result.live_data_files)+len(result.live_delete_files):>6}   {fmt_size(live_total):>12}")

    # ── Все файлы данных (все снапшоты) ──
    all_data_size = sum(f.size for f in result.all_referenced_files.values())
    print(f"\n{sep}")
    print("  ALL REFERENCED FILES (all snapshots, incl. historical)")
    print(sep)
    print(f"  Total files     : {len(result.all_referenced_files):>6}   {fmt_size(all_data_size):>12}")
    historical_files = {
        k: v for k, v in result.all_referenced_files.items()
        if k not in result.live_data_files and k not in result.live_delete_files
    }
    hist_size = sum(f.size for f in historical_files.values())
    print(f"  Historical only : {len(historical_files):>6}   {fmt_size(hist_size):>12}  ← reclaimable after expire_snapshots")

    # ── Metadata-файлы ──
    meta_size = sum(
        result.s3_all_objects.get(k, 0) for k in result.metadata_files
    )
    print(f"\n{sep}")
    print("  METADATA FILES")
    print(sep)
    print(f"  Count           : {len(result.metadata_files):>6}   {fmt_size(meta_size):>12}")
    print(f"  (manifest lists, manifests, metadata.json)")

    # ── Orphan files ──
    # Известные Iceberg-файлы
    known_keys = set(result.all_referenced_files.keys()) | result.metadata_files

    # Все объекты в S3
    s3_total_size = sum(result.s3_all_objects.values())
    s3_data_keys = {
        k for k in result.s3_all_objects
        if k.endswith(".parquet") or k.endswith(".orc") or k.endswith(".avro")
    }

    orphan_files: dict[str, int] = {}
    for k, sz in result.s3_all_objects.items():
        if k not in known_keys:
            orphan_files[k] = sz

    orphan_size = sum(orphan_files.values())

    print(f"\n{sep}")
    print("  ORPHAN FILES (in S3 but not in any snapshot)")
    print(sep)
    print(f"  Count           : {len(orphan_files):>6}   {fmt_size(orphan_size):>12}  ← remove_orphan_files")

    if orphan_files:
        top = sorted(orphan_files.items(), key=lambda x: x[1], reverse=True)[:10]
        print(f"\n  Top-10 by size:")
        for k, sz in top:
            print(f"    {fmt_size(sz):>10}  s3://{result.s3_all_objects and ''}{k}")

    # ── Итоговая таблица ──
    reclaimable = hist_size + orphan_size
    print(f"\n{'═'*70}")
    print("  SUMMARY")
    print(f"{'═'*70}")
    print(f"  Total S3 usage           : {fmt_size(s3_total_size):>12}")
    print(f"  ├─ Live data             : {fmt_size(live_total):>12}  (cannot delete)")
    print(f"  ├─ Metadata              : {fmt_size(meta_size):>12}  (cannot delete)")
    print(f"  ├─ Historical snapshots  : {fmt_size(hist_size):>12}  → expire_snapshots + rewrite_data_files")
    print(f"  └─ Orphan files          : {fmt_size(orphan_size):>12}  → remove_orphan_files")
    print(f"  {'─'*45}")
    print(f"  RECLAIMABLE TOTAL        : {fmt_size(reclaimable):>12}  ({100*reclaimable/s3_total_size:.1f}% of total)" if s3_total_size else "")
    print(f"{'═'*70}\n")

    # ── Рекомендации ──
    print("  RECOMMENDED CLEANUP COMMANDS (Spark SQL):")

# ─────────────────────────── Avro minimal reader ────────────────────────────
# Iceberg manifest files — это Avro. Тащить fastavro не хочется,
# поэтому реализуем минимальный reader для нужных полей.

AVRO_MAGIC = b"Obj\x01"


def _read_long(buf: io.BytesIO) -> int:
    """ZigZag + VarInt decode."""
    b, n, shift = 0, 0, 0
    while True:
        byte = buf.read(1)
        if not byte:
            raise EOFError("unexpected EOF in avro long")
        byte = byte[0]
        n |= (byte & 0x7F) << shift
        shift += 7
        if not (byte & 0x80):
            break
    return (n >> 1) ^ -(n & 1)


def _read_bytes(buf: io.BytesIO) -> bytes:
    n = _read_long(buf)
    return buf.read(n)


def _read_string(buf: io.BytesIO) -> str:
    return _read_bytes(buf).decode("utf-8", errors="replace")


def _skip_value(buf: io.BytesIO, schema, named_types: dict):
    """Пропускаем значение по схеме Avro."""
    if isinstance(schema, list):
        idx = _read_long(buf)
        _skip_value(buf, schema[idx], named_types)
        return
    if isinstance(schema, dict):
        t = schema.get("type")
        if t == "record":
            for f in schema["fields"]:
                _skip_value(buf, f["type"], named_types)
            return
        if t == "array":
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    _skip_value(buf, schema["items"], named_types)
            return
        if t == "map":
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    _read_string(buf)
                    _skip_value(buf, schema["values"], named_types)
            return
        if t in ("bytes", "fixed"):
            if t == "fixed":
                buf.read(schema["size"])
            else:
                _read_bytes(buf)
            return
        if t == "string":
            _read_string(buf)
            return
        if t in ("int", "long"):
            _read_long(buf)
            return
        if t in ("float",):
            buf.read(4)
            return
        if t in ("double",):
            buf.read(8)
            return
        if t in ("boolean",):
            buf.read(1)
            return
        if t == "null":
            return
        if t == "enum":
            _read_long(buf)
            return
        # named type ref
        if isinstance(t, str) and t in named_types:
            _skip_value(buf, named_types[t], named_types)
            return
    if isinstance(schema, str):
        if schema in named_types:
            _skip_value(buf, named_types[schema], named_types)
            return
        if schema == "string":
            _read_string(buf)
        elif schema in ("int", "long"):
            _read_long(buf)
        elif schema == "float":
            buf.read(4)
        elif schema == "double":
            buf.read(8)
        elif schema == "bytes":
            _read_bytes(buf)
        elif schema == "boolean":
            buf.read(1)
        elif schema == "null":
            pass


def _collect_named(schema, named_types: dict):
    if isinstance(schema, list):
        for s in schema:
            _collect_named(s, named_types)
    elif isinstance(schema, dict):
        if schema.get("type") == "record":
            named_types[schema["name"]] = schema
            for f in schema.get("fields", []):
                _collect_named(f["type"], named_types)
        elif schema.get("type") in ("array", "map"):
            _collect_named(schema.get("items", schema.get("values", {})), named_types)


def _read_record_field(buf: io.BytesIO, field_schema, named_types: dict, want: set):
    """Читаем одно поле записи. Возвращаем значение если имя в want, иначе None."""
    name = field_schema["name"]
    ftype = field_schema["type"]
    if name not in want:
        _skip_value(buf, ftype, named_types)
        return name, None

    # читаем реально
    return name, _read_typed(buf, ftype, named_types)


def _read_typed(buf: io.BytesIO, schema, named_types: dict):
    if isinstance(schema, list):
        idx = _read_long(buf)
        actual = schema[idx]
        if actual == "null":
            return None
        return _read_typed(buf, actual, named_types)
    if isinstance(schema, str):
        if schema in named_types:
            return _read_typed(buf, named_types[schema], named_types)
        if schema == "string":
            return _read_string(buf)
        if schema in ("int", "long"):
            return _read_long(buf)
        if schema == "float":
            return struct.unpack(">f", buf.read(4))[0]
        if schema == "double":
            return struct.unpack(">d", buf.read(8))[0]
        if schema == "bytes":
            return _read_bytes(buf)
        if schema == "boolean":
            return bool(buf.read(1)[0])
        if schema == "null":
            return None
        return None
    if isinstance(schema, dict):
        t = schema.get("type")
        if t == "record":
            result = {}
            for f in schema["fields"]:
                result[f["name"]] = _read_typed(buf, f["type"], named_types)
            return result
        if t == "string":
            return _read_string(buf)
        if t in ("int", "long"):
            return _read_long(buf)
        if t == "float":
            return struct.unpack(">f", buf.read(4))[0]
        if t == "double":
            return struct.unpack(">d", buf.read(8))[0]
        if t in ("bytes",):
            return _read_bytes(buf)
        if t == "fixed":
            return buf.read(schema["size"])
        if t == "boolean":
            return bool(buf.read(1)[0])
        if t == "null":
            return None
        if t == "enum":
            idx = _read_long(buf)
            return schema["symbols"][idx]
        if t == "array":
            items = []
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    items.append(_read_typed(buf, schema["items"], named_types))
            return items
        if t == "map":
            result = {}
            while True:
                cnt = _read_long(buf)
                if cnt == 0:
                    break
                if cnt < 0:
                    _read_long(buf)
                    cnt = -cnt
                for _ in range(cnt):
                    k = _read_string(buf)
                    result[k] = _read_typed(buf, schema["values"], named_types)
            return result
        if isinstance(t, str) and t in named_types:
            return _read_typed(buf, named_types[t], named_types)
    return None


def parse_avro_manifest(data: bytes, want_fields: set) -> list[dict]:
    """
    Минимальный Avro-парсер: читает только нужные поля (want_fields)
    из каждой записи, остальное пропускает.
    """
    buf = io.BytesIO(data)
    magic = buf.read(4)
    if magic != AVRO_MAGIC:
        raise ValueError("Not an Avro file")

    # Читаем мета-блок
    meta = {}
    while True:
        cnt = _read_long(buf)
        if cnt == 0:
            break
        if cnt < 0:
            _read_long(buf)
            cnt = -cnt
        for _ in range(cnt):
            k = _read_string(buf)
            v = _read_bytes(buf)
            meta[k] = v

    codec = meta.get("avro.codec", b"null").decode()
    schema_str = meta.get("avro.schema", b"{}").decode()
    schema = json.loads(schema_str)
    sync_marker = buf.read(16)

    named_types: dict = {}
    _collect_named(schema, named_types)

    records = []
    while True:
        try:
            obj_count = _read_long(buf)
        except EOFError:
            break
        if obj_count == 0:
            break
        block_size = _read_long(buf)
        block_data = buf.read(block_size)
        block_sync = buf.read(16)
        if block_sync != sync_marker:
            raise ValueError("Avro sync marker mismatch")

        if codec == "deflate":
            block_data = zlib.decompress(block_data, -15)
        elif codec == "snappy":
            # python-snappy опционален; если не установлен — пробуем cramjam
            try:
                import snappy as _snappy
                # python-snappy возвращает bytes напрямую
                # Snappy в Avro: последние 4 байта — CRC32, остальное — snappy payload
                block_data = _snappy.decompress(block_data[:-4])
            except ImportError:
                try:
                    import cramjam as _cramjam
                    block_data = bytes(_cramjam.snappy.decompress_raw(block_data[:-4]))
                except ImportError:
                    raise ImportError(
                        "Avro codec 'snappy' требует: pip install python-snappy  "
                        "или  pip install cramjam"
                    )
        elif codec == "null":
            pass
        else:
            raise ValueError(f"Unsupported avro codec: {codec}")

        bbuf = io.BytesIO(block_data)
        for _ in range(obj_count):
            rec = {}
            if isinstance(schema, dict) and schema.get("type") == "record":
                for f in schema["fields"]:
                    name, val = _read_record_field(bbuf, f, named_types, want_fields)
                    if val is not None or name in want_fields:
                        rec[name] = val
            records.append(rec)

    return records


# ──────────────────────────── S3 helpers ────────────────────────────────────

def s3_url_to_parts(url: str) -> tuple[str, str]:
    """s3://bucket/prefix/key → (bucket, prefix/key)"""
    url = url.strip()
    if url.startswith("s3://") or url.startswith("s3a://"):
        without = re.sub(r"^s3a?://", "", url)
        parts = without.split("/", 1)
        bucket = parts[0]
        key = parts[1] if len(parts) > 1 else ""
        return bucket, key
    raise ValueError(f"Not an s3/s3a URL: {url}")


def get_s3_client(endpoint_url: Optional[str], profile: Optional[str]) -> boto3.client:
    session = boto3.Session(profile_name=profile) if profile else boto3.Session()
    kwargs = {"service_name": "s3"}
    if endpoint_url:
        kwargs["endpoint_url"] = endpoint_url
    return session.client(**kwargs)


def s3_get_object_bytes(s3, bucket: str, key: str) -> bytes:
    resp = s3.get_object(Bucket=bucket, Key=key)
    return resp["Body"].read()


def s3_list_all_objects(s3, bucket: str, prefix: str) -> dict[str, int]:
    """Возвращает {key: size_bytes} для всех объектов под prefix."""
    result = {}
    paginator = s3.get_paginator("list_objects_v2")
    for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get("Contents", []):
            result[obj["Key"]] = obj["Size"]
    return result


# ──────────────────────────── Iceberg metadata ──────────────────────────────

def find_metadata_json(s3, bucket: str, metadata_prefix: str) -> tuple[str, dict]:
    """
    Ищем последний metadata.json файл. Поддерживаем все варианты именования:

      Вариант A (классический):  v1.metadata.json, v2.metadata.json …
                                  + version-hint.text с номером текущей версии

      Вариант B (UUID-префикс):  <uuid>-v1.metadata.json
                                  <uuid>-v2.metadata.json
                                  <random-uuid>.metadata.json  (некоторые каталоги)

    Возвращает (key, parsed_dict).

    Стратегия:
      1. version-hint.text → читаем номер, ищем v{n}.metadata.json
      2. Листаем *.metadata.json, пробуем извлечь sequence number из имени файла
         (паттерн *-v{N}.metadata.json). Если не получается — сортируем по LastModified.
      3. Берём кандидата с максимальным sequence или самым свежим LastModified.
      4. Финальная проверка: читаем файл и смотрим поле "last-sequence-number" /
         "current-snapshot-id" чтобы убедиться что это валидный metadata.json
         и не промежуточная копия.
    """
    prefix = metadata_prefix.rstrip("/") + "/"

    # ── Шаг 1: version-hint.text ──────────────────────────────────────────
    hint_key = prefix + "version-hint.text"
    try:
        data = s3_get_object_bytes(s3, bucket, hint_key)
        version = int(data.strip())
        # Классическое имя
        for tmpl in [f"v{version}.metadata.json"]:
            candidate = prefix + tmpl
            try:
                s3.head_object(Bucket=bucket, Key=candidate)
                print(f"  [hint] version={version} → {candidate}")
                raw = s3_get_object_bytes(s3, bucket, candidate)
                return candidate, json.loads(raw)
            except ClientError:
                pass
        # UUID-префикс: ищем файл с суффиксом -v{version}.metadata.json
        paginator = s3.get_paginator("list_objects_v2")
        for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
            for obj in page.get("Contents", []):
                k = obj["Key"]
                if re.search(rf"-v{version}\.metadata\.json$", k):
                    print(f"  [hint+uuid] version={version} → {k}")
                    raw = s3_get_object_bytes(s3, bucket, k)
                    return k, json.loads(raw)
    except (ClientError, ValueError):
        pass

    # ── Шаг 2: сканируем все *.metadata.json ──────────────────────────────
    paginator = s3.get_paginator("list_objects_v2")
    candidates: list[tuple[str, int, object]] = []
    # (key, extracted_seq_or_-1, LastModified)

    for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get("Contents", []):
            k = obj["Key"]
            fname = k.rsplit("/", 1)[-1]
            if not fname.endswith(".metadata.json"):
                continue

            # Пробуем вытащить sequence number из имени файла
            # Паттерны:
            #   v42.metadata.json                → seq=42
            #   <uuid>-v42.metadata.json         → seq=42
            #   00042-<uuid>.metadata.json        → seq=42  (некоторые версии Flink writer)
            seq = -1
            m = re.search(r"-v(\d+)\.metadata\.json$", fname)
            if m:
                seq = int(m.group(1))
            else:
                m = re.match(r"v(\d+)\.metadata\.json$", fname)
                if m:
                    seq = int(m.group(1))
                else:
                    m = re.match(r"^(\d+)-", fname)
                    if m:
                        seq = int(m.group(1))

            candidates.append((k, seq, obj["LastModified"]))

    if not candidates:
        raise FileNotFoundError(
            f"No *.metadata.json found under s3://{bucket}/{prefix}\n"
            f"Проверь путь к таблице — он должен указывать на корень таблицы, "
            f"а не на /metadata/ или /data/ поддиректорию."
        )

    # ── Шаг 3: выбираем лучшего кандидата ─────────────────────────────────
    has_seq = [c for c in candidates if c[1] >= 0]
    if has_seq:
        best = max(has_seq, key=lambda c: c[1])
        print(f"  [scan] {len(candidates)} metadata files found, "
              f"latest by seq={best[1]} → {best[0]}")
    else:
        best = max(candidates, key=lambda c: c[2])
        print(f"  [scan] {len(candidates)} metadata files found (no seq in names), "
              f"latest by mtime → {best[0]}")

    chosen = best[0]
    raw = s3_get_object_bytes(s3, bucket, chosen)
    meta = json.loads(raw)

    # ── Шаг 4: сверяем last-sequence-number с именами файлов ──────────────
    if "format-version" not in meta:
        raise ValueError(f"File {chosen} is not a valid Iceberg metadata JSON")

    last_seq_in_file = meta.get("last-sequence-number", -1)
    if has_seq and last_seq_in_file > 0:
        exact = [c for c in has_seq if c[1] == last_seq_in_file]
        if exact and exact[0][0] != chosen:
            chosen = exact[0][0]
            print(f"  [validate] corrected to last-sequence-number={last_seq_in_file} → {chosen}")
            raw = s3_get_object_bytes(s3, bucket, chosen)
            meta = json.loads(raw)

    return chosen, meta


@dataclass
class IcebergFile:
    path: str
    size: int
    file_type: str   # DATA or EQUALITY_DELETES / POSITION_DELETES
    status: str      # ADDED / EXISTING / DELETED
    snapshot_id: Optional[int] = None


@dataclass
class SnapshotInfo:
    snapshot_id: int
    parent_snapshot_id: Optional[int]
    timestamp_ms: int
    operation: str
    manifest_list_path: str


@dataclass
class AuditResult:
    table_name: str
    table_location: str
    current_snapshot_id: Optional[int]
    format_version: int
    snapshots: list[SnapshotInfo]

    # Файлы, видимые в ТЕКУЩЕМ снапшоте (live)
    live_data_files: dict[str, IcebergFile] = field(default_factory=dict)
    live_delete_files: dict[str, IcebergFile] = field(default_factory=dict)

    # Файлы из всех снапшотов (включая исторические)
    all_referenced_files: dict[str, IcebergFile] = field(default_factory=dict)

    # Metadata-файлы (manifest lists, manifests, metadata.json)
    metadata_files: set[str] = field(default_factory=set)

    # Реальные объекты в S3 (все под table_location)
    s3_all_objects: dict[str, int] = field(default_factory=dict)


# ────────────────────────────── Core parser ─────────────────────────────────

class IcebergAuditor:

    def __init__(self, s3, bucket: str):
        self.s3 = s3
        self.bucket = bucket

    def _get_json(self, key: str) -> dict:
        data = s3_get_object_bytes(self.s3, self.bucket, key)
        return json.loads(data)

    def _key_from_path(self, path: str) -> str:
        _, key = s3_url_to_parts(path)
        return key

    def _list_metadata_json_keys(self, metadata_prefix: str) -> list[str]:
        """Возвращает все *.metadata.json ключи под metadata_prefix."""
        prefix = metadata_prefix.rstrip("/") + "/"
        result = []
        paginator = self.s3.get_paginator("list_objects_v2")
        for page in paginator.paginate(Bucket=self.bucket, Prefix=prefix):
            for obj in page.get("Contents", []):
                k = obj["Key"]
                if k.endswith(".metadata.json"):
                    result.append(k)
        return result

    def audit_table(self, table_prefix: str) -> AuditResult:
        """
        table_prefix — путь в S3 до корня таблицы,
        например: 'warehouse/db/my_table'
        """
        metadata_prefix = table_prefix.rstrip("/") + "/metadata"

        print(f"\n{'='*60}")
        print(f"Auditing s3://{self.bucket}/{table_prefix}")
        print(f"{'='*60}")

        # 1. Находим metadata.json (все *.metadata.json регистрируем как metadata-файлы)
        all_meta_keys = self._list_metadata_json_keys(metadata_prefix)
        meta_key, meta = find_metadata_json(self.s3, self.bucket, metadata_prefix)

        format_version = meta.get("format-version", 1)
        table_uuid = meta.get("table-uuid", "unknown")
        location = meta.get("location", f"s3://{self.bucket}/{table_prefix}")
        current_snapshot_id = meta.get("current-snapshot-id")

        print(f"  format-version : {format_version}")
        print(f"  table-uuid     : {table_uuid}")
        print(f"  location       : {location}")
        print(f"  current snap   : {current_snapshot_id}")

        result = AuditResult(
            table_name=table_prefix.split("/")[-1],
            table_location=location,
            current_snapshot_id=current_snapshot_id,
            format_version=format_version,
            snapshots=[],
        )

        # Регистрируем все *.metadata.json как metadata-файлы
        result.metadata_files.update(all_meta_keys)
        result.metadata_files.add(meta_key)

        # 2. Собираем все снапшоты
        snapshots_raw = meta.get("snapshots", [])
        snap_map: dict[int, dict] = {s["snapshot-id"]: s for s in snapshots_raw}

        for s in snapshots_raw:
            si = SnapshotInfo(
                snapshot_id=s["snapshot-id"],
                parent_snapshot_id=s.get("parent-snapshot-id"),
                timestamp_ms=s.get("timestamp-ms", 0),
                operation=s.get("summary", {}).get("operation", "unknown"),
                manifest_list_path=s.get("manifest-list", ""),
            )
            result.snapshots.append(si)

        print(f"  snapshots total: {len(result.snapshots)}")

        # 3. Читаем все manifest lists → manifests → files
        for si in result.snapshots:
            if not si.manifest_list_path:
                continue
            is_current = (si.snapshot_id == current_snapshot_id)
            self._process_manifest_list(si, result, is_current)

        # 4. Листаем S3 (все объекты под table_prefix)
        print(f"\n  Listing all S3 objects under prefix...")
        result.s3_all_objects = s3_list_all_objects(
            self.s3, self.bucket, table_prefix.rstrip("/") + "/"
        )
        print(f"  S3 objects found: {len(result.s3_all_objects)}")

        return result

    def _process_manifest_list(self, si: SnapshotInfo, result: AuditResult, is_current: bool):
        try:
            ml_key = self._key_from_path(si.manifest_list_path)
            ml_data = s3_get_object_bytes(self.s3, self.bucket, ml_key)
            result.metadata_files.add(ml_key)
        except ClientError as e:
            print(f"  [WARN] Cannot read manifest list {si.manifest_list_path}: {e}")
            return

        # Manifest list — тоже Avro
        try:
            entries = parse_avro_manifest(
                ml_data,
                want_fields={"manifest_path", "manifest_length", "content"}
            )
        except Exception as e:
            print(f"  [WARN] Failed to parse manifest list {ml_key}: {e}")
            return

        for entry in entries:
            mpath = entry.get("manifest_path", "")
            if not mpath:
                continue
            content = entry.get("content", 0)  # 0=DATA, 1=DELETES
            self._process_manifest(mpath, content, result, is_current, si.snapshot_id)

    def _process_manifest(
        self,
        manifest_path: str,
        content_type: int,
        result: AuditResult,
        is_current: bool,
        snapshot_id: int,
    ):
        try:
            m_key = self._key_from_path(manifest_path)
            m_data = s3_get_object_bytes(self.s3, self.bucket, m_key)
            result.metadata_files.add(m_key)
        except ClientError as e:
            print(f"  [WARN] Cannot read manifest {manifest_path}: {e}")
            return

        try:
            entries = parse_avro_manifest(
                m_data,
                want_fields={"data_file", "status", "file_path", "file_size_in_bytes", "content"}
            )
        except Exception as e:
            print(f"  [WARN] Failed to parse manifest {m_key}: {e}")
            return

        # STATUS: 0=EXISTING, 1=ADDED, 2=DELETED
        STATUS_MAP = {0: "EXISTING", 1: "ADDED", 2: "DELETED"}
        CONTENT_MAP = {0: "DATA", 1: "POSITION_DELETES", 2: "EQUALITY_DELETES"}

        for entry in entries:
            status_code = entry.get("status", 0)
            status = STATUS_MAP.get(status_code, "UNKNOWN")

            data_file = entry.get("data_file")
            if not data_file or not isinstance(data_file, dict):
                continue

            fpath = data_file.get("file_path", "")
            fsize = data_file.get("file_size_in_bytes", 0) or 0
            fcontent = data_file.get("content", content_type)
            ftype = CONTENT_MAP.get(fcontent, "DATA")

            if not fpath:
                continue

            try:
                fkey = self._key_from_path(fpath)
            except ValueError:
                fkey = fpath  # relative path

            icefile = IcebergFile(
                path=fkey,
                size=fsize,
                file_type=ftype,
                status=status,
                snapshot_id=snapshot_id,
            )

            # Регистрируем в all_referenced
            result.all_referenced_files[fkey] = icefile

            # В live только если текущий снапшот и не DELETED
            if is_current and status != "DELETED":
                if ftype == "DATA":
                    result.live_data_files[fkey] = icefile
                else:
                    result.live_delete_files[fkey] = icefile


# ──────────────────────────── Report ────────────────────────────────────────

def fmt_size(b: int) -> str:
    for unit in ["B", "KB", "MB", "GB", "TB"]:
        if b < 1024:
            return f"{b:.2f} {unit}"
        b /= 1024
    return f"{b:.2f} PB"


def print_report(result: AuditResult):
    sep = "─" * 70

    print(f"\n{'═'*70}")
    print(f"  ICEBERG S3 STORAGE AUDIT: {result.table_name}")
    print(f"{'═'*70}")
    print(f"  Table location  : {result.table_location}")
    print(f"  Format version  : {result.format_version}")
    print(f"  Current snapshot: {result.current_snapshot_id}")
    print(f"  Total snapshots : {len(result.snapshots)}")

    # ── Снапшоты ──
    print(f"\n{sep}")
    print("  SNAPSHOTS")
    print(sep)
    for si in sorted(result.snapshots, key=lambda x: x.timestamp_ms):
        ts = datetime.fromtimestamp(si.timestamp_ms / 1000, tz=timezone.utc).strftime("%Y-%m-%d %H:%M:%S")
        marker = " ◄ CURRENT" if si.snapshot_id == result.current_snapshot_id else ""
        print(f"  [{ts}] snap={si.snapshot_id} op={si.operation}{marker}")

    # ── Live файлы (текущий снапшот) ──
    live_data_size = sum(f.size for f in result.live_data_files.values())
    live_del_size = sum(f.size for f in result.live_delete_files.values())
    live_total = live_data_size + live_del_size

    print(f"\n{sep}")
    print("  LIVE FILES (current snapshot)")
    print(sep)
    print(f"  Data files      : {len(result.live_data_files):>6}   {fmt_size(live_data_size):>12}")
    print(f"  Delete files    : {len(result.live_delete_files):>6}   {fmt_size(live_del_size):>12}")
    print(f"  {'':─<40}")
    print(f"  LIVE TOTAL      : {len(result.live_data_files)+len(result.live_delete_files):>6}   {fmt_size(live_total):>12}")

    # ── Все файлы данных (все снапшоты) ──
    all_data_size = sum(f.size for f in result.all_referenced_files.values())
    print(f"\n{sep}")
    print("  ALL REFERENCED FILES (all snapshots, incl. historical)")
    print(sep)
    print(f"  Total files     : {len(result.all_referenced_files):>6}   {fmt_size(all_data_size):>12}")
    historical_files = {
        k: v for k, v in result.all_referenced_files.items()
        if k not in result.live_data_files and k not in result.live_delete_files
    }
    hist_size = sum(f.size for f in historical_files.values())
    print(f"  Historical only : {len(historical_files):>6}   {fmt_size(hist_size):>12}  ← reclaimable after expire_snapshots")

    # ── Metadata-файлы ──
    meta_size = sum(
        result.s3_all_objects.get(k, 0) for k in result.metadata_files
    )
    print(f"\n{sep}")
    print("  METADATA FILES")
    print(sep)
    print(f"  Count           : {len(result.metadata_files):>6}   {fmt_size(meta_size):>12}")
    print(f"  (manifest lists, manifests, metadata.json)")

    # ── Orphan files ──
    # Известные Iceberg-файлы
    known_keys = set(result.all_referenced_files.keys()) | result.metadata_files

    # Все объекты в S3
    s3_total_size = sum(result.s3_all_objects.values())
    s3_data_keys = {
        k for k in result.s3_all_objects
        if k.endswith(".parquet") or k.endswith(".orc") or k.endswith(".avro")
    }

    orphan_files: dict[str, int] = {}
    for k, sz in result.s3_all_objects.items():
        if k not in known_keys:
            orphan_files[k] = sz

    orphan_size = sum(orphan_files.values())

    print(f"\n{sep}")
    print("  ORPHAN FILES (in S3 but not in any snapshot)")
    print(sep)
    print(f"  Count           : {len(orphan_files):>6}   {fmt_size(orphan_size):>12}  ← remove_orphan_files")

    if orphan_files:
        top = sorted(orphan_files.items(), key=lambda x: x[1], reverse=True)[:10]
        print(f"\n  Top-10 by size:")
        for k, sz in top:
            print(f"    {fmt_size(sz):>10}  s3://{result.s3_all_objects and ''}{k}")

    # ── Итоговая таблица ──
    reclaimable = hist_size + orphan_size
    print(f"\n{'═'*70}")
    print("  SUMMARY")
    print(f"{'═'*70}")
    print(f"  Total S3 usage           : {fmt_size(s3_total_size):>12}")
    print(f"  ├─ Live data             : {fmt_size(live_total):>12}  (cannot delete)")
    print(f"  ├─ Metadata              : {fmt_size(meta_size):>12}  (cannot delete)")
    print(f"  ├─ Historical snapshots  : {fmt_size(hist_size):>12}  → expire_snapshots + rewrite_data_files")
    print(f"  └─ Orphan files          : {fmt_size(orphan_size):>12}  → remove_orphan_files")
    print(f"  {'─'*45}")
    print(f"  RECLAIMABLE TOTAL        : {fmt_size(reclaimable):>12}  ({100*reclaimable/s3_total_size:.1f}% of total)" if s3_total_size else "")
    print(f"{'═'*70}\n")

    # ── Рекомендации ──
    print("  RECOMMENDED CLEANUP COMMANDS (Spark SQL):")
    print()
    if hist_size > 0:
        print("  -- 1. Компактировать data files (убирает delete files, переписывает parquet)")
        print(f"  CALL catalog.system.rewrite_data_files(table => 'db.{result.table_name}');")
        print()
        print("  -- 2. Истечь старые снапшоты")
        print(f"  CALL catalog.system.expire_snapshots(")
        print(f"      table => 'db.{result.table_name}',")
        print(f"      older_than => now() - INTERVAL 1 DAYS,")
        print(f"      retain_last => 1);")
        print()
    if orphan_size > 0:
        print("  -- 3. Удалить orphan-файлы")
        print(f"  CALL catalog.system.remove_orphan_files(")
        print(f"      table => 'db.{result.table_name}',")
        print(f"      older_than => now() - INTERVAL 1 DAYS);")
        print()


# ──────────────────────────── CLI ───────────────────────────────────────────

def main():
    parser = argparse.ArgumentParser(
        description="Iceberg S3 Storage Auditor — 100%% достоверный анализ хранилища"
    )
    parser.add_argument(
        "tables",
        nargs="+",
        metavar="s3://bucket/path/to/table",
        help="S3 URI таблицы (можно несколько)",
    )
    parser.add_argument("--endpoint-url", help="S3-совместимый endpoint (MinIO, Ceph, etc.)")
    parser.add_argument("--profile", help="AWS profile name")
    parser.add_argument("--output-json", help="Сохранить сырые данные в JSON")

    args = parser.parse_args()

    s3 = get_s3_client(args.endpoint_url, args.profile)

    all_results = []
    for table_uri in args.tables:
        try:
            bucket, prefix = s3_url_to_parts(table_uri)
            auditor = IcebergAuditor(s3, bucket)
            result = auditor.audit_table(prefix)
            print_report(result)
            all_results.append(result)
        except Exception as e:
            print(f"\n[ERROR] {table_uri}: {e}", file=sys.stderr)
            import traceback; traceback.print_exc()

    # Итог по всем таблицам если больше одной
    if len(all_results) > 1:
        total_s3 = sum(sum(r.s3_all_objects.values()) for r in all_results)
        total_live = sum(
            sum(f.size for f in r.live_data_files.values()) +
            sum(f.size for f in r.live_delete_files.values())
            for r in all_results
        )
        print(f"\n{'═'*70}")
        print(f"  GRAND TOTAL ({len(all_results)} tables)")
        print(f"{'═'*70}")
        print(f"  Total S3: {fmt_size(total_s3)}   Live: {fmt_size(total_live)}")

    if args.output_json:
        import dataclasses
        def to_dict(obj):
            if dataclasses.is_dataclass(obj):
                return dataclasses.asdict(obj)
            if isinstance(obj, set):
                return list(obj)
            return str(obj)
        with open(args.output_json, "w") as f:
            json.dump(
                [dataclasses.asdict(r) for r in all_results],
                f, indent=2, default=str
            )
        print(f"JSON saved to {args.output_json}")


if __name__ == "__main__":
    main()
