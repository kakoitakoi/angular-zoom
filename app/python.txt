# --- Маппинг типов Oracle -> StarRocks ---
def map_type(data_type, length, precision, scale):
    dt = data_type.upper()

    if dt == 'NUMBER':
        if precision is None:
            return 'BIGINT'
        if scale and scale > 0:
            return f'DECIMAL({precision}, {scale})'
        if precision <= 3:
            return 'TINYINT'
        if precision <= 5:
            return 'SMALLINT'
        if precision <= 10:
            return 'INT'
        if precision <= 18:
            return 'BIGINT'
        return f'DECIMAL({precision}, {scale or 0})'

    if dt in ('VARCHAR2', 'NVARCHAR2', 'CHAR', 'NCHAR'):
        return f'VARCHAR({min(length or 255, 65533)})'

    if dt in ('DATE', 'TIMESTAMP') or dt.startswith('TIMESTAMP'):
        return 'DATETIME'

    if dt in ('CLOB', 'NCLOB', 'LONG', 'XMLTYPE'):
        return 'STRING'

    if dt in ('BLOB', 'RAW', 'LONG RAW'):
        return 'STRING'  # или VARBINARY — зависит от задачи

    if dt in ('FLOAT', 'BINARY_FLOAT'):
        return 'FLOAT'

    if dt == 'BINARY_DOUBLE':
        return 'DOUBLE'

    # fallback
    return 'STRING'

# --- Генерация CREATE TABLE для StarRocks ---
def generate_starrocks_ddl(table_name, columns, pk_columns):
    col_defs = []

    for col_name, data_type, length, precision, scale, nullable in columns:
        sr_type = map_type(data_type, length, precision, scale)
        null_clause = "" if nullable == 'Y' else " NOT NULL"
        col_defs.append(f"    {col_name} {sr_type}{null_clause}")

    # Если PK есть — используем его, иначе берём первую колонку
    key_cols = pk_columns if pk_columns else [columns[0][0]]
    key_type = "PRIMARY KEY" if pk_columns else "DUPLICATE KEY"

    ddl = f"CREATE TABLE {table_name} (\n"
    ddl += ",\n".join(col_defs)
    ddl += f"\n)\n"
    ddl += f"{key_type} ({', '.join(key_cols)})\n"
    ddl += f"DISTRIBUTED BY HASH({key_cols[0]}) BUCKETS 10\n"
    ddl += f"PROPERTIES (\"replication_num\" = \"1\");"

    return ddl

def search(query: str, source_filter: str = None, top_k: int = 5) -> list[dict]:
    # Эмбеддинг запроса
    q_vec = embed([query])[0]
    q_arr = json.dumps(q_vec)

    # Doris не поддерживает ANN нативно через SQL в ARRAY<FLOAT>,
    # поэтому считаем cosine similarity через встроенную функцию
    where = "WHERE 1=1"
    params = []

    if source_filter:
        where += " AND source = %s"
        params.append(source_filter)

    sql = f"""
        SELECT
            id,
            doc_id,
            source,
            chunk_text,
            cosine_distance(embedding, CAST({q_arr} AS ARRAY<FLOAT>)) AS score
        FROM chunks
        {where}
        ORDER BY score ASC          -- cosine_distance: меньше = ближе
        LIMIT %s
    """
    params.append(top_k)

    with conn.cursor(pymysql.cursors.DictCursor) as cur:
        cur.execute(sql, params)
        return cur.fetchall()

def build_cql_exact(term: str) -> str:
    # строгое совпадение через кавычки
    return f'type IN (page, comment) AND text ~ "\\"{term}\\"" ORDER BY lastmodified DESC'


def fetch_all_cql(cql: str):
    url = f"{BASE_URL}/rest/api/content/search"
    params = {
        "cql": cql,
        "limit": 100,
        "expand": "body.storage"
    }

    results = []

    while url:
        resp = requests.get(url, params=params, auth=AUTH, headers=HEADERS)
        resp.raise_for_status()
        data = resp.json()

        results.extend(data.get("results", []))

        next_link = data.get("_links", {}).get("next")
        if next_link:
            url = urljoin(BASE_URL, next_link)
            params = None  # дальше параметры уже внутри next
        else:
            url = None

    return results

def html_to_structured_text(html: str) -> str:
    soup = BeautifulSoup(html, "html.parser")

    # удаляем мусор
    for tag in soup(["script", "style"]):
        tag.decompose()

    # удаляем confluence макросы (ac:, ri:)
    for tag in soup.find_all():
        if ":" in tag.name:
            tag.decompose()

    lines = []

    def process(node, indent=0):
        if node.name in ["h1", "h2", "h3", "h4"]:
            lines.append("\n" + node.get_text(strip=True).upper())

        elif node.name == "p":
            lines.append("\n" + node.get_text(strip=True))

        elif node.name == "li":
            prefix = "  " * indent + "- "
            lines.append(prefix + node.get_text(strip=True))

        elif node.name in ["ul", "ol"]:
            for li in node.find_all("li", recursive=False):
                process(li, indent + 1)

        elif node.name == "pre":
            lines.append("\n[CODE]")
            lines.append(node.get_text())
            lines.append("[/CODE]")

        elif node.name == "code":
            lines.append(f"`{node.get_text(strip=True)}`")

    for elem in soup.find_all(recursive=False):
        process(elem)

    # fallback если структура не поймалась
    if not lines:
        return soup.get_text(separator="\n").strip()

    # чистим пустые строки
    return "\n".join([l for l in lines if l.strip()])

def search_and_extract(term: str):
    cql = build_cql_exact(term)
    results = fetch_all_cql(cql)

    output = []

    for item in results:
        content_type = item["type"]
        content_id = item["id"]
        title = item.get("title", "")

        html = item.get("body", {}).get("storage", {}).get("value", "")
        text = html_to_structured_text(html)

        output.append({
            "type": content_type,
            "id": content_id,
            "title": title,
            "text": text
        })

        # если это page → подтянуть комментарии
        if content_type == "page":
            comments = fetch_comments(content_id)

            for c in comments:
                c_html = c.get("body", {}).get("storage", {}).get("value", "")
                c_text = html_to_structured_text(c_html)

                output.append({
                    "type": "comment",
                    "parent_page_id": content_id,
                    "id": c["id"],
                    "text": c_text
                })

    return output

from bs4 import BeautifulSoup

def find_rows_by_keyword(html: str, keyword: str) -> list[dict]:
    """
    Ищет строки в таблицах HTML по ключевому слову.
    Возвращает список найденных строк с контекстом.
    """
    soup = BeautifulSoup(html, "html.parser")
    tables = soup.find_all("table")

    if not tables:
        print("Таблицы в документе не найдены.")
        return []

    print(f"Найдено таблиц: {len(tables)}")
    results = []

    for table_idx, table in enumerate(tables):
        rows = table.find_all("tr")

        # Извлекаем заголовок (первая строка)
        headers = []
        if rows:
            headers = [th.get_text(strip=True) for th in rows[0].find_all(["th", "td"])]

        for row_idx, row in enumerate(rows):
            cells = row.find_all(["th", "td"])
            cell_texts = [cell.get_text(strip=True) for cell in cells]
            row_text = " ".join(cell_texts)

            if keyword.lower() in row_text.lower():
                results.append({
                    "table_index": table_idx + 1,
                    "row_index": row_idx,
                    "headers": headers,
                    "cells": cell_texts,
                })

    return results


def print_results(results: list[dict], keyword: str):
    if not results:
        print(f'Строки с "{keyword}" не найдены.')
        return

    print(f'\nНайдено строк с "{keyword}": {len(results)}\n')
    print("=" * 60)

    for r in results:
        print(f"Таблица #{r['table_index']}, строка #{r['row_index']}")

        if r["headers"]:
            # Выводим как пары заголовок: значение
            pairs = zip(r["headers"], r["cells"])
            for header, value in pairs:
                marker = " ◄" if keyword.lower() in value.lower() else ""
                print(f"  {header or '—'}: {value}{marker}")
            # Если колонок больше чем заголовков
            for extra in r["cells"][len(r["headers"]):]:
                print(f"  —: {extra}")
        else:
            print("  " + " | ".join(r["cells"]))

        print("-" * 60)

from bs4 import BeautifulSoup, NavigableString, Tag


def split_into_sections(soup: BeautifulSoup) -> list[dict]:
    """
    Разбивает HTML на секции по заголовкам h1-h6.
    Каждая секция содержит заголовок и все элементы до следующего заголовка.
    """
    sections = []
    current_section = {"heading": None, "heading_level": 0, "elements": []}

    for element in soup.body.children if soup.body else soup.children:
        if not isinstance(element, Tag):
            continue

        if element.name in ("h1", "h2", "h3", "h4", "h5", "h6"):
            # Сохраняем предыдущую секцию если она не пустая
            if current_section["elements"] or current_section["heading"]:
                sections.append(current_section)

            current_section = {
                "heading": element.get_text(strip=True),
                "heading_level": int(element.name[1]),
                "elements": [],
            }
        else:
            current_section["elements"].append(element)

    # Последняя секция
    if current_section["elements"] or current_section["heading"]:
        sections.append(current_section)

    return sections


def extract_text_from_element(element: Tag) -> str:
    return element.get_text(separator=" ", strip=True)


def search_in_table(table: Tag, keyword: str, headers: list[str]) -> list[dict]:
    """Ищет keyword в строках таблицы."""
    matches = []
    rows = table.find_all("tr")

    # Определяем заголовки таблицы
    table_headers = headers
    if rows:
        first_row_cells = rows[0].find_all(["th", "td"])
        candidate_headers = [c.get_text(strip=True) for c in first_row_cells]
        # Считаем строку заголовком если есть <th> теги
        if rows[0].find("th"):
            table_headers = candidate_headers
            rows = rows[1:]

    for row_idx, row in enumerate(rows):
        cells = [td.get_text(strip=True) for td in row.find_all(["th", "td"])]
        row_text = " ".join(cells)

        if keyword.lower() in row_text.lower():
            if table_headers:
                row_data = dict(zip(table_headers, cells))
            else:
                row_data = {f"col_{i}": v for i, v in enumerate(cells)}

            matches.append({"row_index": row_idx, "data": row_data, "raw_cells": cells})

    return matches


def search_in_section(section: dict, keyword: str) -> list[dict]:
    """Ищет keyword во всех элементах секции."""
    results = []

    for element in section["elements"]:
        # Случай 1: таблица
        if element.name == "table":
            table_matches = search_in_table(element, keyword, headers=[])
            if table_matches:
                results.append({
                    "type": "table",
                    "matches": table_matches,
                })

        # Случай 2: вложенные таблицы (div/section внутри)
        elif element.name in ("div", "section", "article"):
            for table in element.find_all("table"):
                table_matches = search_in_table(table, keyword, headers=[])
                if table_matches:
                    results.append({
                        "type": "table",
                        "matches": table_matches,
                    })
            # И текст вне таблиц внутри div
            for child in element.children:
                if isinstance(child, Tag) and child.name != "table":
                    text = extract_text_from_element(child)
                    if keyword.lower() in text.lower():
                        results.append({
                            "type": "text",
                            "element": child.name,
                            "text": text,
                        })

        # Случай 3: обычный текстовый элемент
        else:
            text = extract_text_from_element(element)
            if keyword.lower() in text.lower():
                results.append({
                    "type": "text",
                    "element": element.name,
                    "text": text,
                })

    return results


def find_by_keyword(html: str, keyword: str) -> list[dict]:
    soup = BeautifulSoup(html, "html.parser")
    sections = split_into_sections(soup)

    all_results = []

    for section in sections:
        matches = search_in_section(section, keyword)
        if matches:
            all_results.append({
                "section_heading": section["heading"] or "(без заголовка)",
                "heading_level": section["heading_level"],
                "matches": matches,
            })

    return all_results


def print_results(results: list[dict], keyword: str):
    if not results:
        print(f'"{keyword}" не найдено.')
        return

    total = sum(len(r["matches"]) for r in results)
    print(f'Найдено вхождений "{keyword}": {total} в {len(results)} секции(ях)\n')

    for section_result in results:
        heading = section_result["section_heading"]
        level = section_result["heading_level"]
        prefix = "#" * level if level else "~"
        print(f"{prefix} {heading}")
        print("=" * 60)

        for match in section_result["matches"]:
            if match["type"] == "table":
                print("  [ТАБЛИЦА]")
                for row in match["matches"]:
                    print(f"  Строка {row['row_index'] + 1}:")
                    for col, val in row["data"].items():
                        marker = " ◄" if keyword.lower() in val.lower() else ""
                        print(f"    {col}: {val}{marker}")
            else:
                tag = match.get("element", "?")
                text = match["text"]
                # Подсвечиваем вхождение
                highlighted = text.replace(
                    keyword, f"[{keyword.upper()}]"
                )
                print(f"  [{tag.upper()}] {highlighted}")

        print()

"""
SQL Rules Generator for Text2SQL
Парсит SQL-запросы из логов StarRocks, обогащает семантическими описаниями,
кластеризует по доменам и генерирует rules через LLM.
"""

import json
import re
import hashlib
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Optional

import sqlglot
from sqlglot import exp
import anthropic

# ─────────────────────────────────────────────
# 1. ТИПЫ ДАННЫХ
# ─────────────────────────────────────────────

@dataclass
class ColumnMeta:
    name: str
    data_type: str
    description: str = ""
    sample_values: list[str] = field(default_factory=list)

@dataclass
class TableMeta:
    name: str
    description: str = ""
    columns: dict[str, ColumnMeta] = field(default_factory=dict)


@dataclass
class ParsedQuery:
    raw_sql: str
    fingerprint: str                        # хэш нормализованного SQL — для дедупликации
    tables: list[str] = field(default_factory=list)
    joins: list[dict] = field(default_factory=list)       # {left, right, condition}
    filters: list[str] = field(default_factory=list)      # выражения из WHERE/HAVING
    aggregations: list[str] = field(default_factory=list) # COUNT, SUM, AVG …
    window_funcs: list[str] = field(default_factory=list) # OVER (…)
    group_by_cols: list[str] = field(default_factory=list)
    order_by_cols: list[str] = field(default_factory=list)
    selected_cols: list[str] = field(default_factory=list)
    subqueries: int = 0
    ctes: list[str] = field(default_factory=list)
    parse_error: Optional[str] = None


# ─────────────────────────────────────────────
# 2. ПАРСИНГ
# ─────────────────────────────────────────────

def _fingerprint(sql: str) -> str:
    """Нормализуем SQL и считаем хэш для дедупликации."""
    try:
        normalized = sqlglot.transpile(sql, read="starrocks", write="starrocks",
                                       pretty=False, normalize=True)[0]
    except Exception:
        normalized = re.sub(r"\s+", " ", sql.upper().strip())
    return hashlib.md5(normalized.encode()).hexdigest()


def parse_query(sql: str) -> ParsedQuery:
    pq = ParsedQuery(raw_sql=sql, fingerprint=_fingerprint(sql))
    try:
        tree = sqlglot.parse_one(sql, read="starrocks", error_level=sqlglot.ErrorLevel.WARN)
    except Exception as e:
        pq.parse_error = str(e)
        return pq

    # Таблицы
    pq.tables = sorted({
        t.name.lower()
        for t in tree.find_all(exp.Table)
        if t.name
    })

    # Колонки в SELECT
    pq.selected_cols = [
        col.sql(dialect="starrocks")
        for col in tree.find_all(exp.Column)
        if isinstance(col.parent, (exp.Select, exp.Alias))
    ]

    # JOIN-условия
    for join in tree.find_all(exp.Join):
        pq.joins.append({
            "table": join.this.name if hasattr(join.this, "name") else str(join.this),
            "kind":  join.args.get("kind", ""),
            "on":    join.args["on"].sql(dialect="starrocks") if join.args.get("on") else "",
        })

    # WHERE / HAVING
    for node_type in (exp.Where, exp.Having):
        node = tree.find(node_type)
        if node:
            pq.filters.append(node.this.sql(dialect="starrocks"))

    # Агрегатные функции
    AGG_TYPES = (exp.Count, exp.Sum, exp.Avg, exp.Max, exp.Min,
                 exp.ApproxDistinct, exp.GroupConcat)
    pq.aggregations = list({
        a.sql(dialect="starrocks") for a in tree.find_all(*AGG_TYPES)
    })

    # Оконные функции
    pq.window_funcs = list({
        w.sql(dialect="starrocks") for w in tree.find_all(exp.Window)
    })

    # GROUP BY
    gb = tree.find(exp.Group)
    if gb:
        pq.group_by_cols = [e.sql(dialect="starrocks") for e in gb.expressions]

    # ORDER BY
    ob = tree.find(exp.Order)
    if ob:
        pq.order_by_cols = [e.sql(dialect="starrocks") for e in ob.expressions]

    # Подзапросы
    pq.subqueries = sum(1 for _ in tree.find_all(exp.Subquery))

    # CTE
    with_node = tree.find(exp.With)
    if with_node:
        pq.ctes = [cte.alias for cte in with_node.find_all(exp.CTE)]

    return pq


# ─────────────────────────────────────────────
# 3. ОБОГАЩЕНИЕ МЕТАДАННЫМИ
# ─────────────────────────────────────────────

def enrich(pq: ParsedQuery, schema: dict[str, TableMeta]) -> dict:
    """Добавляем к ParsedQuery семантику из схемы."""
    enriched_tables = {}
    for tname in pq.tables:
        meta = schema.get(tname)
        if not meta:
            enriched_tables[tname] = {"description": "unknown table"}
            continue
        enriched_tables[tname] = {
            "description": meta.description,
            "columns_used": {},
        }
        # Какие колонки таблицы реально используются в запросе
        all_sql = " ".join(
            pq.selected_cols + pq.filters + pq.aggregations +
            pq.group_by_cols + pq.order_by_cols +
            [j["on"] for j in pq.joins]
        ).lower()
        for cname, cmeta in meta.columns.items():
            if cname.lower() in all_sql:
                enriched_tables[tname]["columns_used"][cname] = {
                    "type":        cmeta.data_type,
                    "description": cmeta.description,
                    "samples":     cmeta.sample_values,
                }

    return {
        "tables":       enriched_tables,
        "joins":        pq.joins,
        "filters":      pq.filters,
        "aggregations": pq.aggregations,
        "window_funcs": pq.window_funcs,
        "group_by":     pq.group_by_cols,
        "order_by":     pq.order_by_cols,
        "subqueries":   pq.subqueries,
        "ctes":         pq.ctes,
    }

def cluster_by_domain(queries: list[ParsedQuery]) -> dict[str, list[ParsedQuery]]:
    """
    Группируем запросы по набору таблиц (доменный ключ).
    Запросы без таблиц — в кластер '_unparsed'.

    clusters: dict[str, list[ParsedQuery]] = defaultdict(list)
    for pq in queries:

        clusters[key].append(pq)
    return dict(clusters)


def deduplicate(queries: list[ParsedQuery]) -> list[ParsedQuery]:

    result = []
    for pq in queries:
        if pq.fingerprint not in seen:
            seen.add(pq.fingerprint)
            result.append(pq)
    return result

"""
SQL Rules Generator for Text2SQL
Парсит SQL-запросы из логов StarRocks, обогащает семантическими описаниями,
кластеризует по доменам и генерирует rules через LLM.
"""

import json
import re
import hashlib
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Optional

import sqlglot
from sqlglot import exp
import anthropic

# ─────────────────────────────────────────────
# 1. ТИПЫ ДАННЫХ
# ─────────────────────────────────────────────

@dataclass
class ColumnMeta:
    name: str
    data_type: str
    description: str = ""
    sample_values: list[str] = field(default_factory=list)

@dataclass
class TableMeta:
    name: str
    description: str = ""
    columns: dict[str, ColumnMeta] = field(default_factory=dict)


@dataclass
class ParsedQuery:
    raw_sql: str
    fingerprint: str                        # хэш нормализованного SQL — для дедупликации
    tables: list[str] = field(default_factory=list)
    joins: list[dict] = field(default_factory=list)       # {left, right, condition}
    filters: list[str] = field(default_factory=list)      # выражения из WHERE/HAVING
    aggregations: list[str] = field(default_factory=list) # COUNT, SUM, AVG …
    window_funcs: list[str] = field(default_factory=list) # OVER (…)
    group_by_cols: list[str] = field(default_factory=list)
    order_by_cols: list[str] = field(default_factory=list)
    selected_cols: list[str] = field(default_factory=list)
    subqueries: int = 0
    ctes: list[str] = field(default_factory=list)
    parse_error: Optional[str] = None


# ─────────────────────────────────────────────
# 2. ПАРСИНГ
# ─────────────────────────────────────────────

def _fingerprint(sql: str) -> str:
    """Нормализуем SQL и считаем хэш для дедупликации."""
    try:
        normalized = sqlglot.transpile(sql, read="starrocks", write="starrocks",
                                       pretty=False, normalize=True)[0]
    except Exception:
        normalized = re.sub(r"\s+", " ", sql.upper().strip())
    return hashlib.md5(normalized.encode()).hexdigest()


def parse_query(sql: str) -> ParsedQuery:
    pq = ParsedQuery(raw_sql=sql, fingerprint=_fingerprint(sql))
    try:
        tree = sqlglot.parse_one(sql, read="starrocks", error_level=sqlglot.ErrorLevel.WARN)
    except Exception as e:
        pq.parse_error = str(e)
        return pq

    # Таблицы
    pq.tables = sorted({
        t.name.lower()
        for t in tree.find_all(exp.Table)
        if t.name
    })

    # Колонки в SELECT
    pq.selected_cols = [
        col.sql(dialect="starrocks")
        for col in tree.find_all(exp.Column)
        if isinstance(col.parent, (exp.Select, exp.Alias))
    ]

    # JOIN-условия
    for join in tree.find_all(exp.Join):
        pq.joins.append({
            "table": join.this.name if hasattr(join.this, "name") else str(join.this),
            "kind":  join.args.get("kind", ""),
            "on":    join.args["on"].sql(dialect="starrocks") if join.args.get("on") else "",
        })

    # WHERE / HAVING
    for node_type in (exp.Where, exp.Having):
        node = tree.find(node_type)
        if node:
            pq.filters.append(node.this.sql(dialect="starrocks"))

    # Агрегатные функции
    AGG_TYPES = (exp.Count, exp.Sum, exp.Avg, exp.Max, exp.Min,
                 exp.ApproxDistinct, exp.GroupConcat)
    pq.aggregations = list({
        a.sql(dialect="starrocks") for a in tree.find_all(*AGG_TYPES)
    })

    # Оконные функции
    pq.window_funcs = list({
        w.sql(dialect="starrocks") for w in tree.find_all(exp.Window)
    })

    # GROUP BY
    gb = tree.find(exp.Group)
    if gb:
        pq.group_by_cols = [e.sql(dialect="starrocks") for e in gb.expressions]

    # ORDER BY
    ob = tree.find(exp.Order)
    if ob:
        pq.order_by_cols = [e.sql(dialect="starrocks") for e in ob.expressions]

    # Подзапросы
    pq.subqueries = sum(1 for _ in tree.find_all(exp.Subquery))

    # CTE
    with_node = tree.find(exp.With)
    if with_node:
        pq.ctes = [cte.alias for cte in with_node.find_all(exp.CTE)]

    return pq


# ─────────────────────────────────────────────
# 3. ОБОГАЩЕНИЕ МЕТАДАННЫМИ
# ─────────────────────────────────────────────

def enrich(pq: ParsedQuery, schema: dict[str, TableMeta]) -> dict:
    """Добавляем к ParsedQuery семантику из схемы."""
    enriched_tables = {}
    for tname in pq.tables:
        meta = schema.get(tname)
        if not meta:
            enriched_tables[tname] = {"description": "unknown table"}
            continue
        enriched_tables[tname] = {
            "description": meta.description,
            "columns_used": {},
        }
        # Какие колонки таблицы реально используются в запросе
        all_sql = " ".join(
            pq.selected_cols + pq.filters + pq.aggregations +
            pq.group_by_cols + pq.order_by_cols +
            [j["on"] for j in pq.joins]
        ).lower()
        for cname, cmeta in meta.columns.items():
            if cname.lower() in all_sql:
                enriched_tables[tname]["columns_used"][cname] = {
                    "type":        cmeta.data_type,
                    "description": cmeta.description,
                    "samples":     cmeta.sample_values,
                }

    return {
        "tables":       enriched_tables,
        "joins":        pq.joins,
        "filters":      pq.filters,
        "aggregations": pq.aggregations,
        "window_funcs": pq.window_funcs,
        "group_by":     pq.group_by_cols,
        "order_by":     pq.order_by_cols,
        "subqueries":   pq.subqueries,
        "ctes":         pq.ctes,
    }


# ─────────────────────────────────────────────
# 4. КЛАСТЕРИЗАЦИЯ ПО ДОМЕНАМ
# ─────────────────────────────────────────────

def cluster_by_domain(queries: list[ParsedQuery]) -> dict[str, list[ParsedQuery]]:
    """
    Группируем запросы по набору таблиц (доменный ключ).
    Запросы без таблиц — в кластер '_unparsed'.
    """
    clusters: dict[str, list[ParsedQuery]] = defaultdict(list)
    for pq in queries:
        key = "+".join(sorted(pq.tables)) if pq.tables else "_unparsed"
        clusters[key].append(pq)
    return dict(clusters)


def deduplicate(queries: list[ParsedQuery]) -> list[ParsedQuery]:
    seen: set[str] = set()
    result = []
    for pq in queries:
        if pq.fingerprint not in seen:
            seen.add(pq.fingerprint)
            result.append(pq)
    return result

# 5. ГЕНЕРАЦИЯ RULES ЧЕРЕЗ LLM
# ─────────────────────────────────────────────

SYSTEM_PROMPT = """\
Ты — эксперт по SQL и проектированию Text2SQL систем.
Тебе дают кластер реальных SQL-запросов из production-логов StarRocks
вместе с обогащёнными метаданными (описания таблиц, колонок, типы данных).

Твоя задача — сформулировать "SQL rules" для этого кластера таблиц.
Rules — это короткие, однозначные правила, которые LLM будет использовать
при генерации SQL для данных таблиц.

Формат ответа — строго JSON-массив объектов:
[
  {
    "rule_id": "R001",
    "category": "join|filter|aggregation|naming|performance|business_logic",
    "tables":   ["table1", "table2"],
    "rule":     "Краткое правило на русском языке",
    "rationale": "Почему это правило важно (1-2 предложения)",
    "example_sql": "Короткий пример SQL (необязательно)"
  }
]

Правила:
- Выводи только то, что подтверждается паттернами из запросов
- Не придумывай правила без доказательств в данных
- Категория business_logic — для правил о бизнес-смысле фильтров/агрегаций
- Минимум 5, максимум 20 rules на кластер
- Только JSON, без markdown-обёртки
"""


def build_user_message(
    cluster_key: str,
    queries: list[ParsedQuery],
    schema: dict[str, TableMeta],
    max_queries: int = 30,
) -> str:
    samples = queries[:max_queries]
    enriched = [enrich(pq, schema) for pq in samples]

    payload = {
        "domain": cluster_key,
        "total_queries_in_cluster": len(queries),
        "sample_size": len(samples),
        "queries": [
            {
                "sql":      pq.raw_sql[:2000],  # обрезаем монстров
                "metadata": enriched[i],
            }
            for i, pq in enumerate(samples)
        ],
    }
    return json.dumps(payload, ensure_ascii=False, indent=2)


def generate_rules_for_cluster(
    cluster_key: str,
    queries: list[ParsedQuery],
    schema: dict[str, TableMeta],
    client: anthropic.Anthropic,
    model: str = "claude-opus-4-5",
) -> list[dict]:
    user_msg = build_user_message(cluster_key, queries, schema)
    response = client.messages.create(
        model=model,
        max_tokens=4096,
        system=SYSTEM_PROMPT,
        messages=[{"role": "user", "content": user_msg}],
    )
    raw = response.content[0].text.strip()
    # На случай если модель всё же обернула в ```json
    raw = re.sub(r"^```(?:json)?\s*", "", raw)
    raw = re.sub(r"\s*```$", "", raw)
    try:
        return json.loads(raw)
    except json.JSONDecodeError as e:
        print(f"  [!] JSON parse error for cluster '{cluster_key}': {e}")
        return [{"rule_id": "ERR", "raw_response": raw}]

def generate_all_rules(
    sql_queries: list[str],
    schema: dict[str, TableMeta],
    api_key: str,
    model: str = "claude-opus-4-5",
    min_cluster_size: int = 2,
    output_path: str = "sql_rules.json",
) -> dict[str, list[dict]]:
    """
    sql_queries      — список сырых SQL-строк из логов
    schema           — словарь {table_name: TableMeta}
    api_key          — Anthropic API key
    min_cluster_size — кластеры меньше этого размера пропускаем
    output_path      — куда сохранить итоговый JSON
    """
    print(f"[1/4] Парсим {len(sql_queries)} запросов...")
    parsed = [parse_query(sql) for sql in sql_queries]
    errors = [p for p in parsed if p.parse_error]
    print(f"      Ошибок парсинга: {len(errors)}")

    print("[2/4] Дедуплицируем...")
    parsed = deduplicate(parsed)
    print(f"      Уникальных запросов: {len(parsed)}")

    print("[3/4] Кластеризуем по доменам...")
    clusters = cluster_by_domain(parsed)
    for key, qs in clusters.items():
        print(f"      {key}: {len(qs)} запросов")

    print("[4/4] Генерируем rules через LLM...")
    client = anthropic.Anthropic(api_key=api_key)
    all_rules: dict[str, list[dict]] = {}

    for cluster_key, qs in clusters.items():
        if cluster_key == "_unparsed":
            print(f"  [skip] {cluster_key}")
            continue
        if len(qs) < min_cluster_size:
            print(f"  [skip] {cluster_key} (слишком мало запросов: {len(qs)})")
            continue
        print(f"  Обрабатываем '{cluster_key}' ({len(qs)} запросов)...")
        rules = generate_rules_for_cluster(cluster_key, qs, schema, client, model)
        all_rules[cluster_key] = rules
        print(f"    → сгенерировано {len(rules)} rules")

    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(all_rules, f, ensure_ascii=False, indent=2)
    print(f"\nГотово! Результат сохранён в {output_path}")
    return all_rules

from sqlglot import parse_one, exp


# -------------------------------
# helpers
# -------------------------------

def extract_tables(tree):
    """
    alias -> {table, db, type}
    """
    tables = {}

    # --- CTE ---
    with_expr = tree.find(exp.With)
    if with_expr:
        for cte in with_expr.expressions:
            alias = cte.alias
            tables[alias] = {
                "table": alias,
                "db": None,
                "type": "CTE"
            }

    # --- TABLES + SUBQUERY ---
    for node in tree.walk():
        if isinstance(node, exp.Table):
            alias = node.alias_or_name
            tables[alias] = {
                "table": node.name,
                "db": node.db,
                "type": "TABLE"
            }

        elif isinstance(node, exp.Subquery):
            alias = node.alias
            if alias:
                tables[alias] = {
                    "table": None,
                    "db": None,
                    "type": "SUBQUERY"
                }

    return tables


def is_column(node):
    return isinstance(node, exp.Column)


def extract_column(node):
    return node.table, node.name


def extract_conditions(expr):
    """
    рекурсивно достаём только COLUMN = COLUMN
    """
    result = []

    if expr is None:
        return result

    if isinstance(expr, exp.EQ):
        l, r = expr.left, expr.right

        if is_column(l) and is_column(r):
            result.append((extract_column(l), extract_column(r)))

    elif isinstance(expr, (exp.And, exp.Or)):
        result += extract_conditions(expr.left)
        result += extract_conditions(expr.right)

    else:
        for arg in expr.args.values():
            if isinstance(arg, exp.Expression):
                result += extract_conditions(arg)
            elif isinstance(arg, list):
                for item in arg:
                    if isinstance(item, exp.Expression):
                        result += extract_conditions(item)

    return result

# MAIN

def extract_joins(sql: str):
    tree = parse_one(sql, read="mysql")

    alias_map = extract_tables(tree)
    results = []

    for join in tree.find_all(exp.Join):
        on_expr = join.args.get("on")

        conditions = extract_conditions(on_expr)

        if not conditions:
            continue

        # определяем алиасы таблиц из условий
        aliases = set()
        for (l, r) in conditions:
            aliases.add(l[0])
            aliases.add(r[0])

        if len(aliases) != 2:
            # пропускаем сложные случаи (3+ таблицы)
            continue

        a1, a2 = list(aliases)

        t1 = alias_map.get(a1, {})
        t2 = alias_map.get(a2, {})

        # --- формируем связи ---
        links = []
        for i, (l, r) in enumerate(conditions, start=1):
            if l[0] == a1:
                left_col = l[1]
                right_col = r[1]
            else:
                left_col = r[1]
                right_col = l[1]

            links.append({
                f"связь{i}": {
                    "поле1": left_col,
                    "поле2": right_col,
                    "предикат": "ON"
                }
            })

        # --- isOnlyTables ---
        is_only_tables = (
            t1.get("type") == "TABLE" and
            t2.get("type") == "TABLE"
        )

        result = {
            "таблица1": t1.get("table"),
            "алиас1": a1,
            "таблица2": t2.get("table"),
            "алиас2": a2,
            "кол-во связей": len(links),
            "join_side": join.side,
            "join_kind": join.kind,
            "isOnlyTables": is_only_tables
        }

        # добавляем связи внутрь
        for link in links:
            result.update(link)

        results.append(result)

    return results








def analyze_joins(sql: str):
    tree = sqlglot.parse_one(sql)

    # 1. Собираем все алиасы
    sources = [tree.find(exp.From).this] + [j.this for j in tree.find_all(exp.Join)]
    aliases = {}
    for src in sources:
        alias = src.alias or src.name
        full  = f"{src.args.get('db', '')}.{src.name}".lstrip(".")
        aliases[alias] = full

    # 2. Собираем связи
    links = []
    for join in tree.find_all(exp.Join):
        on_clause = join.args.get("on")
        if not on_clause:
            continue
        for eq in on_clause.find_all(exp.EQ):
            links.append((
                f"{eq.left.table}.{eq.left.name}",
                f"{eq.right.table}.{eq.right.name}",
            ))

    print("=== Таблицы ===")

import sqlglot
from sqlglot import exp

sql = """
SELECT *
FROM dwh.fact_sales   AS fs
JOIN dwh.dim_product  AS dp ON fs.product_id = dp.id
JOIN dwh.dim_customer AS dc ON get_json_string(dp.json, "customer_id") = dc.id
"""

tree = sqlglot.parse_one(sql, dialect="starrocks")

def extract_columns_from_expr(node) -> list[exp.Column]:
    """Рекурсивно достаём все exp.Column из выражения (включая внутри функций)."""
    return list(node.find_all(exp.Column))

def analyze_joins(tree):
    links = []

    for join in tree.find_all(exp.Join):
        on_clause = join.args.get("on")
        if not on_clause:
            continue

        for eq in on_clause.find_all(exp.EQ):
            left_cols  = extract_columns_from_expr(eq.left)
            right_cols = extract_columns_from_expr(eq.right)

            # Определяем "сырое" представление каждой стороны
            left_repr  = eq.left.sql()   # читаемо: get_json_string(dp.json, 'customer_id')
            right_repr = eq.right.sql()  # dc.id

            # Уникальные алиасы таблиц, задействованных в каждой стороне
            left_tables  = list(dict.fromkeys(c.table for c in left_cols  if c.table))
            right_tables = list(dict.fromkeys(c.table for c in right_cols if c.table))

            links.append({
                "left_expr":   left_repr,
                "left_tables": left_tables,
                "right_expr":  right_repr,
                "right_tables": right_tables,
                # Пара для графа связей (все комбинации)
                "pairs": [(lt, rt) for lt in left_tables for rt in right_tables],
            })

    return links

for link in analyze_joins(tree):

    result = {}

    """Заменяет алиасы таблиц на полные имена во всём выражении."""
    node = node.copy()

    return node.sql()

import sqlglot
from sqlglot import exp

sql = """
WITH cte_sales AS (
    SELECT * FROM dwh.fact_sales
)
SELECT *
FROM cte_sales AS fs
JOIN dwh.dim_product  AS dp ON fs.product_id = dp.id
JOIN dwh.dim_customer AS dc ON get_json_string(dp.json, "customer_id") = dc.id
JOIN (SELECT id FROM dwh.dim_region) AS dr ON dc.region_id = dr.id
"""

tree = sqlglot.parse_one(sql, dialect="starrocks")

def build_alias_map(tree) -> dict[str, str]:
    # Имена CTE — исключаем
    cte_names = {
        cte.alias for cte in tree.find_all(exp.CTE)
    }

    sources = [tree.find(exp.From).this] + [j.this for j in tree.find_all(exp.Join)]
    result = {}

    for src in sources:
        alias = src.alias or src.name

        # Пропускаем подзапросы (Subquery внутри FROM/JOIN)
        if isinstance(src, exp.Subquery):
            continue

        # Пропускаем если имя таблицы совпадает с CTE
        if src.name in cte_names:
            continue

        db   = src.args.get("db") or ""
        full = f"{db}.{src.name}" if db else src.name
        result[alias] = full

    return result

def resolve_expr(node, alias_map: dict) -> str:
    node = node.copy()
    for col in node.find_all(exp.Column):
        if col.table and col.table in alias_map:
            col.set("table", exp.Identifier(this=alias_map[col.table], quoted=False))
    return node.sql()

def analyze_joins(tree):
    alias_map = build_alias_map(tree)
    links = []

    for join in tree.find_all(exp.Join):
        on_clause = join.args.get("on")
        if not on_clause:
            continue

        for eq in on_clause.find_all(exp.EQ):
            left_cols  = list(eq.left.find_all(exp.Column))
            right_cols = list(eq.right.find_all(exp.Column))

            # Фильтруем алиасы, которых нет в alias_map (CTE / Subquery)
            left_tables = list(dict.fromkeys(
                alias_map[c.table]
                for c in left_cols
                if c.table and c.table in alias_map
            ))
            right_tables = list(dict.fromkeys(
                alias_map[c.table]
                for c in right_cols
                if c.table and c.table in alias_map
            ))

            # Пропускаем связь если обе стороны не разрешились в реальные таблицы
            if not left_tables and not right_tables:
                continue

            links.append({
                "left_expr":   resolve_expr(eq.left,  alias_map),
                "right_expr":  resolve_expr(eq.right, alias_map),
                "left_tables":  left_tables  or ["<cte_or_subquery>"],
                "right_tables": right_tables or ["<cte_or_subquery>"],
                "pairs":        [(lt, rt) for lt in (left_tables  or ["<cte_or_subquery>"])
                                          for rt in (right_tables or ["<cte_or_subquery>"])],
            })

    return links

for link in analyze_joins(tree):
    print(f"Выражение: {link['left_expr']}  =  {link['right_expr']}")
    print(f"Таблицы:   {link['left_tables']} <-> {link['right_tables']}")
    print(f"Связи:     {link['pairs']}")
    print()

CTE-фильтр — собираем все имена через tree.find_all(exp.CTE) и исключаем таблицы, чьё src.name попало в этот список.
Subquery-фильтр — isinstance(src, exp.Subquery) отсекает инлайн-подзапросы в FROM/JOIN.
В analyze_joins — дополнительная проверка c.table in alias_map при формировании left_tables/right_tables: алиасы CTE и Subquery просто не попадают в результат, заменяясь на <cte_or_subquery> для наглядности (можно заменить на None или пропускать совсем).​​​​​​​​​​​​​​​​

# ── 1. Получить одну задачу ──────────────────────────────────────────
def get_issue(issue_key: str) -> dict:
    """
    Возвращает поля задачи, включая description в rendered (HTML) виде.
    """
    url = f"{JIRA_URL}/rest/api/3/issue/{issue_key}"
    params = {
        "fields": "summary,description,status,assignee,comment",
        "expand": "renderedFields",   # ← даёт description как HTML
    }
    resp = requests.get(url, headers=headers, auth=auth, params=params)
    resp.raise_for_status()
    return resp.json()


# ── 2. Поиск задач по JQL ────────────────────────────────────────────
def search_issues(jql: str, fields: list[str] = None, max_results: int = 50) -> list[dict]:
    """
    Поиск через актуальный эндпоинт /rest/api/3/search/jql
    с поддержкой пагинации через nextPageToken.
    """
    url = f"{JIRA_URL}/rest/api/3/search/jql"
    fields = fields or ["summary", "description", "status", "assignee"]

    issues = []
    next_page_token = None

    while True:
        payload = {
            "jql": jql,
            "maxResults": max_results,
            "fields": fields,
            "expand": ["renderedFields"],
        }
        if next_page_token:
            payload["nextPageToken"] = next_page_token

        resp = requests.post(url, headers={**headers, "Content-Type": "application/json"},
                             auth=auth, json=payload)
        resp.raise_for_status()
        data = resp.json()

        issues.extend(data.get("issues", []))
        next_page_token = data.get("nextPageToken")

        if not next_page_token:
            break

    return issues


# ── 3. Извлечь HTML-описание и прогнать через наш парсер ─────────────
def get_issue_html(issue: dict) -> str | None:
    """
    renderedFields.description содержит HTML (если expand=renderedFields).
    """
    rendered = issue.get("renderedFields", {})
    return rendered.get("description")  # HTML-строка или None

def search_issues(keyword: str, max_results: int = 50) -> list[dict]:
    """
    text~ ищет по summary, description, comments, environment
    и всем текстовым custom fields одновременно.
    """
    url = f"{JIRA_URL}/rest/api/2/search"
    issues = []
    start_at = 0

    while True:
        payload = {
            "jql": f'text ~ "{keyword}" ORDER BY updated DESC',
            "startAt": start_at,
            "maxResults": max_results,
            "fields": [
                "summary", "description", "status", "assignee",
                "reporter", "project", "issuetype", "comment",
                "attachment", "labels", "environment",
            ],
            "expand": ["renderedFields"],
        }

        resp = requests.post(url, headers=headers, auth=auth, json=payload)
        resp.raise_for_status()
        data = resp.json()
        batch = data.get("issues", [])
        issues.extend(batch)

        # Пагинация
        start_at += len(batch)
        if start_at >= data.get("total", 0) or not batch:
            break

    return issues


# ═══════════════════════════════════════════════════════════

def search_in_worklogs(issue_key: str, keyword: str) -> list[dict]:

    resp = requests.get(url, headers=headers, auth=auth)
    resp.raise_for_status()

    matches = []
    for wl in resp.json().get("worklogs", []):
        comment = wl.get("comment", "")
        if keyword.lower() in comment.lower():

                "started":    wl.get("started", "")[:10],
                "time_spent": wl.get("timeSpent"),

# 4. ИЗВЛЕЧЕНИЕ ВХОЖДЕНИЙ ИЗ HTML-ПОЛЕЙ
#    (description, environment, comment — приходят как HTML)
# ═══════════════════════════════════════════════════════════

def extract_context_from_html(html: str, keyword: str) -> list[str]:
    """Возвращает список абзацев/ячеек содержащих keyword."""
    if not html:
        return []
    soup = BeautifulSoup(html, "html.parser")
    matches = []

    for el in soup.find_all(["p", "li", "td", "th", "h1", "h2", "h3", "pre"]):
        text = el.get_text(separator=" ", strip=True)
        if keyword.lower() in text.lower():
            highlighted = text.replace(keyword, f"[{keyword.upper()}]")
            tag = el.name.upper()
            matches.append(f"[{tag}] {highlighted}")

    return matches

def analyze_issue(issue: dict, keyword: str, include_worklogs: bool = True) -> dict | None:
    key     = issue["key"]
    fields  = issue["fields"]
    rendered = issue.get("renderedFields", {})

    result = {
        "key":       key,
        "summary":   fields.get("summary", ""),
        "project":   fields.get("project", {}).get("name"),
        "status":    fields.get("status", {}).get("name"),
        "assignee":  (fields.get("assignee") or {}).get("displayName", "—"),
        "reporter":  (fields.get("reporter") or {}).get("displayName", "—"),
        "areas":     {},
    }

    # ── Summary ──────────────────────────────────────────
    if keyword.lower() in result["summary"].lower():
        result["areas"]["summary"] = result["summary"]

    # ── Description ──────────────────────────────────────
    desc_html = rendered.get("description", "")
    desc_hits = extract_context_from_html(desc_html, keyword)
    if desc_hits:
        result["areas"]["description"] = desc_hits

    # ── Environment ──────────────────────────────────────
    env_html = rendered.get("environment", "")
    env_hits = extract_context_from_html(env_html, keyword)
    if env_hits:
        result["areas"]["environment"] = env_hits

    # ── Labels ───────────────────────────────────────────
    label_hits = [l for l in fields.get("labels", []) if keyword.lower() in l.lower()]
    if label_hits:
        result["areas"]["labels"] = label_hits

    # ── Comments ─────────────────────────────────────────
    comment_hits = []
    comments = rendered.get("comment", {}).get("comments", [])
    for c in comments:
        hits = extract_context_from_html(c.get("renderedBody", ""), keyword)
        if hits:
            comment_hits.append({
                "author":  c.get("author", {}).get("displayName"),
                "created": c.get("created", "")[:10],
                "hits":    hits,
            })
    if comment_hits:
        result["areas"]["comments"] = comment_hits

    # ── Attachments ──────────────────────────────────────
    att_hits = search_in_attachments(issue, keyword)
    if att_hits:
        result["areas"]["attachments"] = att_hits

    # ── Worklogs ─────────────────────────────────────────
    if include_worklogs:
        wl_hits = search_in_worklogs(key, keyword)
        if wl_hits:
            result["areas"]["worklogs"] = wl_hits

    # Если нет ни одного совпадения — пропускаем
    if not result["areas"]:
        return None

    return result

import re

def is_exact_match(text: str, keyword: str) -> bool:
    """
    Проверяет что keyword встречается как отдельное слово/идентификатор,
    а не как часть другого слова.
    Граница: пробел, пунктуация, начало/конец строки.
    """
    # \b не работает для snake_case идентификаторов с _ на границе,
    # поэтому используем явные разделители
    pattern = r'(?<![a-zA-Z0-9_])' + re.escape(keyword) + r'(?![a-zA-Z0-9_])'
    return bool(re.search(pattern, text, re.IGNORECASE))

def extract_context_from_html(html: str, keyword: str) -> list[str]:
    if not html:
        return []
    soup = BeautifulSoup(html, "html.parser")
    matches = []

    for el in soup.find_all(["p", "li", "td", "th", "h1", "h2", "h3", "pre"]):
        text = el.get_text(separator=" ", strip=True)
        if is_exact_match(text, keyword):       # ← было: keyword.lower() in text.lower()
            highlighted = re.sub(
                r'(?<![a-zA-Z0-9_])' + re.escape(keyword) + r'(?![a-zA-Z0-9_])',
                f"[{keyword.upper()}]",
                text,
                flags=re.IGNORECASE,
            )
            tag = el.name.upper()
            matches.append(f"[{tag}] {highlighted}")

    return matches

import sqlglot

from collections import Counter
from typing import List

def extract_tables(sql: str, dialect: str = "starrocks") -> List[str]:
    """Извлекает все таблицы из SQL запроса в формате db.table"""
    try:
        statements = sqlglot.parse(sql, dialect=dialect)
    except Exception:
        return []

    tables = []
    for statement in statements:
        if statement is None:
            continue
        for table in statement.find_all(exp.Table):
            # Пропускаем CTE-алиасы (сами определения WITH x AS (...))
            if isinstance(table.parent, exp.CTE):
                continue

            db = table.args.get("db")
            name = table.name

            if not name:
                continue

            if db:
                tables.append(f"{db.name}.{name}")
            else:
                tables.append(name)

    return tables


def count_tables(queries: List[str], dialect: str = "starrocks") -> Counter:
    """Считает сколько раз каждая таблица встречается во всех запросах"""
    counter = Counter()
    for sql in queries:
        tables = extract_tables(sql, dialect=dialect)
        counter.update(tables)
    return counter

def flush_to_doris(conn: pymysql.Connection, counter: Counter, batch_size: int):
    """Сбрасывает Counter в Doris батчами через INSERT ... ON DUPLICATE KEY UPDATE."""
    rows    = [(db, name, cnt) for (db, name), cnt in counter.items()]
    total   = len(rows)
    written = 0

    with conn.cursor() as cur:
        for i in range(0, total, batch_size):
            batch = rows[i : i + batch_size]
            placeholders = ", ".join(["(%s, %s, %s)"] * len(batch))
            flat_values  = [v for row in batch for v in row]

            sql = f"""
                INSERT INTO {TARGET_TABLE} (db, name, cnt)
                VALUES {placeholders}
                ON DUPLICATE KEY UPDATE cnt = cnt + VALUES(cnt)
            """
            cur.execute(sql, flat_values)
            written += len(batch)

    conn.commit()
    log.info(f"Записано в Doris: {written:,} / {total:,} уникальных таблиц")

# ─── Fetch ───────────────────────────────────────────────────────────────────

def get_total(config: dict) -> int:
    conn = pymysql.connect(**config)
    try:
        with conn.cursor() as cur:
            cur.execute(f"""
                SELECT COUNT(*)
                FROM {SOURCE_TABLE}
                WHERE stmt LIKE 'SELECT%'
                  AND state = 'EOF'
                  AND stmt IS NOT NULL
            """)
            return cur.fetchone()[0]
    finally:
        conn.close()


def fetch_page(config: dict, offset: int, limit: int) -> list:
    conn = pymysql.connect(**config)
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(f"""
                SELECT
                    queryId,
                    timestamp,
                    user,
                    db,
                    stmt,
                    scanBytes,
                    scanRows,
                    returnRows,
                    cpuCostNs,
                    memCostBytes
                FROM {SOURCE_TABLE}
                WHERE stmt LIKE 'SELECT%'
                  AND state = 'EOF'
                  AND stmt IS NOT NULL
                ORDER BY queryId
                LIMIT %s OFFSET %s
            """, (limit, offset))
            return cur.fetchall()
    finally:
        conn.close()


def iter_pages(config: dict, total: int, page_size: int) -> Iterator[list]:
    for offset in range(0, total, page_size):
        rows = fetch_page(config, offset, page_size)
        if not rows:
            break
        yield rows

# ─── Score ───────────────────────────────────────────────────────────────────
# Логика оценки:
#   Хороший эталонный запрос — тот который:
#     - читает много данных (высокий scanRows/scanBytes) → сложный, показательный
#     - возвращает мало строк (низкий returnRows/scanRows) → есть фильтрация/агрегация
#     - эффективен по CPU и памяти относительно объёма скана
#
#   Штрафуем за:
#     - огромный mem_cost (неэффективное использование памяти)
#     - returnRows ≈ scanRows (нет фильтрации — скорее всего SELECT *)
#     - нулевой скан (пустой/тривиальный запрос)

def log_scale(value: float, base: float = 1_000_000) -> float:
    """Нормализует большие числа в [0, 1] через логарифм."""
    if value <= 0:
        return 0.0
    return min(1.0, math.log1p(value) / math.log1p(base))


def compute_score(
    scan_bytes: int,
    scan_rows: int,
    return_rows: int,
    cpu_cost_ns: int,
    mem_cost_bytes: int,
) -> int:
    """
    Возвращает score [0, 100].

    Компоненты (веса):
      40% — complexity:   насколько много данных читает запрос (log-шкала по scanRows)
      30% — selectivity:  насколько хорошо фильтрует (1 - returnRows/scanRows)
      20% — cpu_eff:      CPU на строку скана (меньше — лучше, штраф за дорогие запросы)
      10% — mem_eff:      память на строку скана (меньше — лучше)
    """
    if scan_rows == 0:
        return 0  # тривиальный запрос — не эталон

    # 1. Complexity — хотим запросы с реальным сканом (эталоны должны быть нетривиальны)
    complexity = log_scale(scan_rows, base=10_000_000)  # 10M строк → score ~1.0

    # 2. Selectivity — хороший запрос возвращает << чем читает
    selectivity = 1.0 - min(1.0, return_rows / scan_rows)

    # 3. CPU efficiency — нормируем cpu_cost на scanRow, штрафуем дорогие
    cpu_per_row = cpu_cost_ns / scan_rows  # ns на строку
    # считаем что 1000 ns/row — граница нормы; выше — штраф
    cpu_eff = 1.0 - log_scale(cpu_per_row, base=100_000)

    # 4. Memory efficiency
    mem_per_row = mem_cost_bytes / scan_rows
    mem_eff = 1.0 - log_scale(mem_per_row, base=10_000)

    raw = (
        0.40 * complexity  +
        0.30 * selectivity +
        0.20 * cpu_eff     +
        0.10 * mem_eff
    )

    return max(0, min(100, round(raw * 100)))

# ─── SQL parsing ─────────────────────────────────────────────────────────────

def remove_comments(sql: str) -> str:
    """Убирает -- и /* */ комментарии через sqlglot tokenizer."""
    try:
        tokens = sqlglot.tokens.Tokenizer().tokenize(sql)
        clean_parts = []
        prev_end = 0
        for tok in tokens:
            if tok.token_type in (
                sqlglot.tokens.TokenType.COMMENT,
                sqlglot.tokens.TokenType.BLOCK_COMMENT,
            ):
                continue
            clean_parts.append(tok.text)
        return " ".join(clean_parts)
    except Exception:
        return sql


def parse_sql(sql: str) -> dict:
    """
    Возвращает:
      tables     : list[str]  — только реальные таблицы db.name (не CTE, не подзапрос-алиас)
      has_cte    : bool
      has_group_by: bool
      has_limit  : bool
      has_window : bool
    """
    result = {
        "tables":       [],
        "has_cte":      False,
        "has_group_by": False,
        "has_limit":    False,
        "has_window":   False,
    }
    try:
        stmts = sqlglot.parse(sql, dialect="starrocks", error_level=sqlglot.ErrorLevel.IGNORE)
    except Exception:
        return result

    for stmt in stmts:
        if stmt is None:
            continue

        # ── CTE ──────────────────────────────────────────────────────────────
        cte_names = set()
        for cte in stmt.find_all(exp.CTE):
            result["has_cte"] = True
            cte_names.add(cte.alias)  # имя CTE — не считаем реальной таблицей

        # ── GROUP BY ─────────────────────────────────────────────────────────
        if stmt.find(exp.Group):
            result["has_group_by"] = True

        # ── LIMIT ────────────────────────────────────────────────────────────
        if stmt.find(exp.Limit):
            result["has_limit"] = True

        # ── Window functions ─────────────────────────────────────────────────
        if stmt.find(exp.Window):
            result["has_window"] = True

        # ── Tables (только реальные) ─────────────────────────────────────────
        for table in stmt.find_all(exp.Table):
            # пропускаем объявление CTE
            if isinstance(table.parent, exp.CTE):
                continue
            name = table.name
            if not name:
                continue
            # пропускаем ссылки на CTE по имени
            if name in cte_names:
                continue
            db_node = table.args.get("db")
            full    = f"{db_node.name}.{name}" if db_node else name
            result["tables"].append(full)

    # дедупликация с сохранением порядка
    seen = set()
    result["tables"] = [t for t in result["tables"] if not (t in seen or seen.add(t))]
    return result


# ─── Worker ──────────────────────────────────────────────────────────────────

def process_batch(rows: list) -> list[dict]:
    """Обрабатывает батч строк из audit-таблицы, возвращает список готовых записей."""
    results = []
    for row in rows:
        sql_raw = row.get("stmt") or ""

        sql_clean   = remove_comments(sql_raw).strip()
        parsed      = parse_sql(sql_clean)

        scan_bytes     = int(row.get("scanBytes")     or 0)
        scan_rows      = int(row.get("scanRows")      or 0)
        return_rows    = int(row.get("returnRows")    or 0)
        cpu_cost_ns    = int(row.get("cpuCostNs")     or 0)
        mem_cost_bytes = int(row.get("memCostBytes")  or 0)

        score = compute_score(scan_bytes, scan_rows, return_rows, cpu_cost_ns, mem_cost_bytes)

        results.append({
            "query_id":       row.get("queryId", ""),
            "query_time":     str(row.get("timestamp", "")),
            "user":           row.get("user", ""),
            "db":             row.get("db", ""),
            "sql_clean":      sql_clean[:65_000],   # TEXT лимит Doris
            "tables":         json.dumps(parsed["tables"], ensure_ascii=False),
            "has_cte":        int(parsed["has_cte"]),
            "has_group_by":   int(parsed["has_group_by"]),
            "has_limit":      int(parsed["has_limit"]),
            "has_window":     int(parsed["has_window"]),
            "scan_bytes":     scan_bytes,
            "scan_rows":      scan_rows,
            "return_rows":    return_rows,
            "cpu_cost_ns":    cpu_cost_ns,
            "mem_cost_bytes": mem_cost_bytes,
            "score":          score,
        })
    return results

# ─── Write ───────────────────────────────────────────────────────────────────

def flush_to_doris(conn: pymysql.Connection, records: list[dict], batch_size: int):
    if not records:
        return

    columns = [
        "query_id", "query_time", "user", "db", "sql_clean",
        "tables", "has_cte", "has_group_by", "has_limit", "has_window",
        "scan_bytes", "scan_rows", "return_rows", "cpu_cost_ns", "mem_cost_bytes",
        "score",
    ]
    col_str = ", ".join(columns)

    with conn.cursor() as cur:
        for i in range(0, len(records), batch_size):
            batch        = records[i : i + batch_size]
            placeholders = ", ".join([f"({', '.join(['%s'] * len(columns))})"] * len(batch))
            flat_values  = [row[c] for row in batch for c in columns]
            cur.execute(f"INSERT INTO {TARGET_TABLE} ({col_str}) VALUES {placeholders}", flat_values)

    conn.commit()

# ─── Main ────────────────────────────────────────────────────────────────────

def main():
    log.info("Считаю строки в audit-таблице StarRocks...")
    total = get_total(STARROCKS_CONFIG)
    log.info(f"Всего SELECT запросов: {total:,} | воркеров: {N_WORKERS}")

    processed  = 0
    write_conn = pymysql.connect(**DORIS_CONFIG)

    try:
        with mp.Pool(processes=N_WORKERS) as pool:
            for batch_records in pool.imap_unordered(
                process_batch,
                iter_pages(STARROCKS_CONFIG, total, PAGE_SIZE),
                chunksize=1,
            ):
                flush_to_doris(write_conn, batch_records, WRITE_BATCH)
                processed += len(batch_records)

                if processed % LOG_EVERY == 0:
                    log.info(f"Обработано и записано: {processed:,} / {total:,}")
    finally:
        write_conn.close()

    log.info(f"Готово. Всего записано: {processed:,}")


if __name__ == "__main__":
    main()

@dataclass
class LLMConfig:
    url: str
    api_key: str
    max_concurrent: int = 20      # параллельных запросов к LLM
    max_retries: int = 3
    retry_delays: list = field(default_factory=lambda: [5, 15, 30])  # сек
    timeout_sec: int = 120
    retry_on_status: set = field(default_factory=lambda: {429, 500, 502, 503, 504})


# ─── Результат обработки одной записи ────────────────────────────────────────

@dataclass
class RecordResult:
    record_id: Any
    success: bool
    data: dict | None = None
    error: str | None = None
    attempts: int = 0


# ─── Вызов LLM с retry ───────────────────────────────────────────────────────

async def call_llm_with_retry(
    session: aiohttp.ClientSession,
    record: dict,
    config: LLMConfig,
) -> RecordResult:
    record_id = record.get("id")
    last_error = None

    for attempt in range(1, config.max_retries + 1):
        try:
            payload = build_payload(record)  # твоя функция сборки промпта

            async with session.post(
                config.url,
                json=payload,
                timeout=aiohttp.ClientTimeout(total=config.timeout_sec),
            ) as resp:

                # HTTP-ошибки с retry
                if resp.status in config.retry_on_status:
                    last_error = f"HTTP {resp.status}"
                    logger.warning(f"[{record_id}] attempt {attempt}: {last_error}")
                    await _maybe_wait(attempt, config)
                    continue

                # Другие HTTP-ошибки — не ретраим
                if resp.status != 200:
                    return RecordResult(
                        record_id=record_id,
                        success=False,
                        error=f"HTTP {resp.status} (no retry)",
                        attempts=attempt,
                    )

                raw = await resp.json()

        except (aiohttp.ClientError, asyncio.TimeoutError) as e:
            last_error = str(e)
            logger.warning(f"[{record_id}] attempt {attempt}: network error: {e}")
            await _maybe_wait(attempt, config)
            continue

        # Парсим ответ — изолируем ошибки структуры
        try:
            parsed = parse_llm_response(raw)  # твоя функция парсинга
            logger.info(f"[{record_id}] OK after {attempt} attempt(s)")
            return RecordResult(
                record_id=record_id,
                success=True,
                data=parsed,
                attempts=attempt,
            )
        except (KeyError, ValueError, json.JSONDecodeError) as e:
            last_error = f"parse error: {e}"
            logger.warning(f"[{record_id}] attempt {attempt}: {last_error}, raw={str(raw)[:200]}")
            await _maybe_wait(attempt, config)
            continue

    return RecordResult(
        record_id=record_id,
        success=False,
        error=last_error,
        attempts=config.max_retries,
    )


async def _maybe_wait(attempt: int, config: LLMConfig):
    if attempt <= len(config.retry_delays):
        delay = config.retry_delays[attempt - 1]
        logger.info(f"  waiting {delay}s before retry...")
        await asyncio.sleep(delay)


# ─── Оркестратор с семафором ──────────────────────────────────────────────────

async def process_records(
    records: list[dict],
    config: LLMConfig,
) -> list[RecordResult]:
    semaphore = asyncio.Semaphore(config.max_concurrent)
    results: list[RecordResult] = []
    failed: list[RecordResult] = []

    async def bounded_call(record: dict) -> RecordResult:
        async with semaphore:
            return await call_llm_with_retry(session, record, config)

    connector = aiohttp.TCPConnector(limit=config.max_concurrent + 5)
    async with aiohttp.ClientSession(
        headers={"Authorization": f"Bearer {config.api_key}"},
        connector=connector,
    ) as session:
        tasks = [bounded_call(r) for r in records]

        # tqdm-прогресс (опционально: pip install tqdm)
        from tqdm.asyncio import tqdm
        for coro in tqdm.as_completed(tasks, total=len(tasks)):
            result = await coro
            results.append(result)
            if not result.success:
                failed.append(result)

    # Итог
    logger.info(f"Done: {len(results) - len(failed)}/{len(results)} OK, "
                f"{len(failed)} failed")
    for f in failed:
        logger.error(f"FAILED id={f.record_id}: {f.error} ({f.attempts} attempts)")

    return results


# ─── Точка входа ─────────────────────────────────────────────────────────────

def run_pipeline(records: list[dict], config: LLMConfig) -> list[RecordResult]:
    return asyncio.run(process_records(records, config))

@contextmanager
def doris_connection(cfg: DorisConfig):
    conn = pymysql.connect(
        host=cfg.host,
        port=cfg.port,
        user=cfg.user,
        password=cfg.password,
        database=cfg.database,
        charset=cfg.charset,
        connect_timeout=10,
        autocommit=False,
    )
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def fetch_pending_records(cfg: DorisConfig, batch_size: int = 500) -> list[QueryRecord]:
    """
    Забираем записи у которых ещё нет LLM-обогащения.
    Поле llm_processed_at = NULL означает "не обработано".
    """
    sql = """
        SELECT
            query_md5,
            sql_text,
            db_name,
            user_name,
            query_count
        FROM dwh.query_log_enriched
        WHERE llm_processed_at IS NULL
          AND sql_text IS NOT NULL
          AND LENGTH(sql_text) > 10
        ORDER BY query_count DESC          -- приоритет: частые запросы первыми
        LIMIT %(batch_size)s
    """
    with doris_connection(cfg) as conn:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(sql, {"batch_size": batch_size})
            rows = cur.fetchall()

    logger.info(f"Fetched {len(rows)} pending records from Doris")
    return [
        QueryRecord(
            query_md5=r["query_md5"],
            sql_text=r["sql_text"],
            db_name=r["db_name"] or "",
            user_name=r["user_name"] or "",
            query_count=r["query_count"] or 0,
        )
        for r in rows
    ]


def write_results(cfg: DorisConfig, results: list[RecordResult]) -> None:
    """
    Пишем результаты в Doris.
    Успешные — обновляем enrichment-поля.
    Упавшие — пишем ошибку в отдельную колонку, чтобы не терять трек.
    """
    successful = [r for r in results if r.success]
    failed = [r for r in results if not r.success]

    if successful:
        _upsert_successful(cfg, successful)
    if failed:
        _upsert_failed(cfg, failed)

    logger.info(f"Written to Doris: {len(successful)} OK, {len(failed)} failed")


def _upsert_successful(cfg: DorisConfig, results: list[RecordResult]) -> None:
    sql = """
        UPDATE dwh.query_log_enriched
        SET
            llm_domain          = %(domain)s,
            llm_intent          = %(intent)s,
            llm_complexity      = %(complexity)s,
            llm_main_tables     = %(main_tables)s,
            llm_comment         = %(comment)s,
            llm_processed_at    = %(processed_at)s,
            llm_error           = NULL
        WHERE query_md5 = %(query_md5)s
    """
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    params_batch = [
        {
            "domain":       r.llm_result.domain,
            "intent":       r.llm_result.intent,
            "complexity":   r.llm_result.complexity,
            "main_tables":  json.dumps(r.llm_result.main_tables, ensure_ascii=False),
            "comment":      r.llm_result.comment,
            "processed_at": now,
            "query_md5":    r.query_md5,
        }
        for r in results
    ]
    with doris_connection(cfg) as conn:
        with conn.cursor() as cur:
            cur.executemany(sql, params_batch)


def _upsert_failed(cfg: DorisConfig, results: list[RecordResult]) -> None:
    sql = """
        UPDATE dwh.query_log_enriched
        SET
            llm_error        = %(error)s,
            llm_processed_at = %(processed_at)s
        WHERE query_md5 = %(query_md5)s
    """
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    params_batch = [
        {
            "error":        r.error[:1000] if r.error else "unknown",
            "processed_at": now,
            "query_md5":    r.query_md5,
        }
        for r in results
    ]
    with doris_connection(cfg) as conn:
        with conn.cursor() as cur:
            cur.executemany(sql, params_batch)

def build_payload(record: QueryRecord, model: str) -> dict:
    system_prompt = """Ты — аналитик SQL-запросов корпоративного DWH на базе StarRocks.
Тебе дают SQL-запрос. Верни ТОЛЬКО валидный JSON без пояснений и markdown-блоков.

Формат ответа:
{
  "domain": "<домен: finance | hr | sales | logistics | analytics | other>",
  "intent": "<краткое намерение запроса на русском, 5–15 слов>",
  "complexity": "<low | medium | high>",
  "main_tables": ["<таблица1>", "<таблица2>"],
  "comment": "<необязательный комментарий>"
}

Правила сложности:
- low: простой SELECT, 1–2 таблицы, без оконных функций
- medium: JOIN 2–4 таблиц, агрегации, подзапросы
- high: 5+ таблиц, оконные функции, CTE, UNION"""

    user_prompt = f"""База: {record.db_name}
Пользователь: {record.user_name}
Частота выполнения: {record.query_count}

SQL:
{record.sql_text[:3000]}"""   # обрезаем очень длинные запросы

    return {
        "model": model,
        "max_tokens": 512,
        "temperature": 0.1,          # детерминизм важнее креативности
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user",   "content": user_prompt},
        ],
    }


def parse_llm_response(raw: dict) -> LLMResult:
    """
    Парсим ответ LLM. Два уровня защиты:
    1. Вытаскиваем JSON даже если модель обернула его в ```json ... ```
    2. Валидируем обязательные поля и допустимые значения enum
    """
    content = raw["choices"][0]["message"]["content"].strip()

    # Защита от ```json ... ``` обёртки
    json_match = re.search(r"```(?:json)?\s*(\{.*?\})\s*```", content, re.DOTALL)
    json_str = json_match.group(1) if json_match else content

    data = json.loads(json_str)   # ValueError/JSONDecodeError если кривой JSON

    # Валидация enum-полей
    valid_domains = {"finance", "hr", "sales", "logistics", "analytics", "other"}
    valid_complexity = {"low", "medium", "high"}

    domain = str(data.get("domain", "other")).lower()
    if domain not in valid_domains:
        domain = "other"

    complexity = str(data.get("complexity", "medium")).lower()
    if complexity not in valid_complexity:
        complexity = "medium"

    intent = str(data.get("intent", "")).strip()
    if not intent:
        raise ValueError("LLM returned empty intent")

    main_tables = data.get("main_tables", [])
    if not isinstance(main_tables, list):
        main_tables = []

    return LLMResult(
        domain=domain,
        intent=intent,
        complexity=complexity,
        main_tables=[str(t) for t in main_tables[:20]],  # не более 20 таблиц
        comment=str(data.get("comment", ""))[:500],
    )

# ASYNC LLM-ВЫЗОВ С RETRY
# ══════════════════════════════════════════════════════════════════════════════

async def call_llm_with_retry(
    session: aiohttp.ClientSession,
    record: QueryRecord,
    config: LLMConfig,
) -> RecordResult:
    last_error = None

    for attempt in range(1, config.max_retries + 1):
        try:
            payload = build_payload(record, config.model)

            async with session.post(
                config.url,
                json=payload,
                timeout=aiohttp.ClientTimeout(total=config.timeout_sec),
            ) as resp:

                if resp.status in config.retry_on_status:
                    last_error = f"HTTP {resp.status}"
                    logger.warning(
                        f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}"
                    )
                    await _backoff(attempt, config)
                    continue

                if resp.status != 200:
                    body = await resp.text()
                    return RecordResult(
                        query_md5=record.query_md5,
                        success=False,
                        error=f"HTTP {resp.status}: {body[:200]}",
                        attempts=attempt,
                    )

                raw = await resp.json()

        except asyncio.TimeoutError:
            last_error = f"timeout after {config.timeout_sec}s"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

        except aiohttp.ClientError as e:
            last_error = f"network: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

        # Парсинг ответа
        try:
            llm_result = parse_llm_response(raw)
            logger.debug(
                f"[{record.query_md5[:8]}] OK attempt={attempt} "
                f"domain={llm_result.domain} complexity={llm_result.complexity}"
            )
            return RecordResult(
                query_md5=record.query_md5,
                success=True,
                llm_result=llm_result,
                attempts=attempt,
            )

        except (KeyError, ValueError, json.JSONDecodeError) as e:
            last_error = f"parse error: {e}"
            logger.warning(
                f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}"
            )
            await _backoff(attempt, config)
            continue

    return RecordResult(
        query_md5=record.query_md5,
        success=False,
        error=last_error,
        attempts=config.max_retries,
    )


async def _backoff(attempt: int, config: LLMConfig) -> None:
    delay = config.retry_delays[attempt - 1] if attempt <= len(config.retry_delays) else 30
    logger.info(f"  backoff {delay}s...")
    await asyncio.sleep(delay)

# ОРКЕСТРАТОР
# ══════════════════════════════════════════════════════════════════════════════

async def process_batch(
    records: list[QueryRecord],
    llm_cfg: LLMConfig,
) -> list[RecordResult]:
    semaphore = asyncio.Semaphore(llm_cfg.max_concurrent)
    results: list[RecordResult] = []

    async def bounded(record: QueryRecord) -> RecordResult:
        async with semaphore:
            return await call_llm_with_retry(session, record, llm_cfg)

    connector = aiohttp.TCPConnector(limit=llm_cfg.max_concurrent + 10)
    headers = {
        "Authorization": f"Bearer {llm_cfg.api_key}",
        "Content-Type": "application/json",
    }

    async with aiohttp.ClientSession(headers=headers, connector=connector) as session:
        tasks = [bounded(r) for r in records]
        total = len(tasks)
        done = 0

        for coro in asyncio.as_completed(tasks):
            result = await coro
            results.append(result)
            done += 1

            # Прогресс каждые 10 записей
            if done % 10 == 0 or done == total:
                ok = sum(1 for r in results if r.success)
                logger.info(f"Progress: {done}/{total} | OK: {ok} | FAIL: {done - ok}")

    return results

# ══════════════════════════════════════════════════════════════════════════════

def run_enrichment_pipeline(
    doris_cfg: DorisConfig,
    llm_cfg: LLMConfig,
    batch_size: int = 500,
    write_every: int = 50,        # пишем в Doris каждые N обработанных записей
) -> dict:
    start = datetime.now()

    # 1. Читаем из Doris
    records = fetch_pending_records(doris_cfg, batch_size)
    if not records:
        logger.info("No pending records. Exiting.")
        return {"processed": 0, "success": 0, "failed": 0}

    # 2. Async обработка с промежуточной записью
    all_results: list[RecordResult] = []

    async def run_with_flush():
        semaphore = asyncio.Semaphore(llm_cfg.max_concurrent)
        buffer: list[RecordResult] = []

        async def bounded(record):
            async with semaphore:
                return await call_llm_with_retry(session, record, llm_cfg)

        connector = aiohttp.TCPConnector(limit=llm_cfg.max_concurrent + 10)
        headers = {
            "Authorization": f"Bearer {llm_cfg.api_key}",
            "Content-Type": "application/json",
        }
        async with aiohttp.ClientSession(headers=headers, connector=connector) as session:
            tasks = [bounded(r) for r in records]
            total = len(tasks)
            done = 0

            for coro in asyncio.as_completed(tasks):
                result = await coro
                buffer.append(result)
                all_results.append(result)
                done += 1

                # Flush в Doris по достижении порога
                if len(buffer) >= write_every:
                    logger.info(f"Flushing {len(buffer)} results to Doris...")
                    write_results(doris_cfg, buffer)
                    buffer.clear()

                if done % 10 == 0 or done == total:
                    ok = sum(1 for r in all_results if r.success)
                    logger.info(f"Progress: {done}/{total} | OK: {ok} | FAIL: {done-ok}")

            # Финальный flush остатка
            if buffer:
                logger.info(f"Final flush: {len(buffer)} results to Doris...")
                write_results(doris_cfg, buffer)

    asyncio.run(run_with_flush())

    # 3. Итоговая статистика
    elapsed = (datetime.now() - start).total_seconds()
    success = sum(1 for r in all_results if r.success)
    failed = len(all_results) - success

    stats = {
        "processed": len(all_results),
        "success":   success,
        "failed":    failed,
        "elapsed_sec": round(elapsed, 1),
        "rps": round(len(all_results) / elapsed, 2) if elapsed > 0 else 0,
    }
    logger.info(f"Pipeline done: {stats}")
    return stats

from aiohttp import ServerDisconnectedError, ClientError, ClientSession

async def call_llm_with_retry(
    session: ClientSession,
    record: QueryRecord,
    config: LLMConfig,
) -> RecordResult:
    last_error = None
    MAX_RECONNECTS = 5

    for attempt in range(1, config.max_retries + 1):
        raw = None

        # ── Внутренний цикл: только для ServerDisconnected ──────────────────
        for reconnect in range(1, MAX_RECONNECTS + 1):
            try:
                payload = build_payload(record, config.model)
                async with session.post(
                    config.url,
                    json=payload,
                    timeout=aiohttp.ClientTimeout(total=config.timeout_sec),
                ) as resp:

                    if resp.status in config.retry_on_status:
                        last_error = f"HTTP {resp.status}"
                        break  # выходим из reconnect → идём в основной retry

                    if resp.status != 200:
                        body = await resp.text()
                        return RecordResult(
                            query_md5=record.query_md5,
                            success=False,
                            error=f"HTTP {resp.status}: {body[:200]}",
                            attempts=attempt,
                        )

                    raw = await resp.json()
                    break  # успех — выходим из reconnect-цикла

            except ServerDisconnectedError:
                wait = reconnect * 2
                logger.warning(
                    f"[{record.query_md5[:8]}] ServerDisconnected "
                    f"(attempt={attempt}, reconnect={reconnect}/{MAX_RECONNECTS})"
                    f" → retry in {wait}s"
                )
                await asyncio.sleep(wait)
                continue  # следующая reconnect-попытка

            except asyncio.TimeoutError:
                last_error = f"timeout after {config.timeout_sec}s"
                logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
                break  # в основной retry

            except ClientError as e:
                last_error = f"network: {e}"
                logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
                break  # в основной retry

        else:
            # reconnect-цикл исчерпан (все 5 попыток — ServerDisconnected)
            last_error = f"ServerDisconnected after {MAX_RECONNECTS} reconnects"
            logger.error(f"[{record.query_md5[:8]}] {last_error}")
            await _backoff(attempt, config)
            continue  # следующая основная попытка

        # ── Парсинг если получили raw ────────────────────────────────────────
        if raw is None:
            await _backoff(attempt, config)
            continue

        try:
            llm_result = parse_llm_response(raw)
            return RecordResult(
                query_md5=record.query_md5,
                success=True,
                llm_result=llm_result,
                attempts=attempt,
            )
        except (KeyError, ValueError, json.JSONDecodeError) as e:
            last_error = f"parse error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

    return RecordResult(
        query_md5=record.query_md5,
        success=False,
        error=last_error,
        attempts=config.max_retries,
    )

@contextmanager
def doris_connection(cfg: DorisConfig):
    conn = pymysql.connect(
        host=cfg.host,
        port=cfg.port,
        user=cfg.user,
        password=cfg.password,
        database=cfg.database,
        charset=cfg.charset,
        connect_timeout=10,
        autocommit=False,
    )
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def fetch_pending_records(cfg: DorisConfig, batch_size: int = 500) -> list[QueryRecord]:
    sql = """
        SELECT
            query_md5,
            sql_text,
            db_name,
            user_name,
            query_count
        FROM dwh.query_log_enriched
        WHERE llm_processed_at IS NULL
          AND sql_text IS NOT NULL
          AND LENGTH(sql_text) > 10
        ORDER BY query_count DESC
        LIMIT %(batch_size)s
    """
    with doris_connection(cfg) as conn:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(sql, {"batch_size": batch_size})
            rows = cur.fetchall()

    logger.info(f"Fetched {len(rows)} pending records from Doris")
    return [
        QueryRecord(
            query_md5=r["query_md5"],
            sql_text=r["sql_text"],
            db_name=r["db_name"] or "",
            user_name=r["user_name"] or "",
            query_count=r["query_count"] or 0,
        )
        for r in rows
    ]


def write_results(cfg: DorisConfig, results: list[RecordResult]) -> None:
    successful = [r for r in results if r.success]
    failed = [r for r in results if not r.success]
    if successful:
        _upsert_successful(cfg, successful)
    if failed:
        _upsert_failed(cfg, failed)
    logger.info(f"Written to Doris: {len(successful)} OK, {len(failed)} failed")


def _upsert_successful(cfg: DorisConfig, results: list[RecordResult]) -> None:
    sql = """
        UPDATE dwh.query_log_enriched
        SET
            llm_domain       = %(domain)s,
            llm_intent       = %(intent)s,
            llm_complexity   = %(complexity)s,
            llm_main_tables  = %(main_tables)s,
            llm_comment      = %(comment)s,
            llm_processed_at = %(processed_at)s,
            llm_error        = NULL
        WHERE query_md5 = %(query_md5)s
    """
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    params = [
        {
            "domain":       r.llm_result.domain,
            "intent":       r.llm_result.intent,
            "complexity":   r.llm_result.complexity,
            "main_tables":  json.dumps(r.llm_result.main_tables, ensure_ascii=False),
            "comment":      r.llm_result.comment,
            "processed_at": now,
            "query_md5":    r.query_md5,
        }
        for r in results
    ]
    with doris_connection(cfg) as conn:
        with conn.cursor() as cur:
            cur.executemany(sql, params)


def _upsert_failed(cfg: DorisConfig, results: list[RecordResult]) -> None:
    sql = """
        UPDATE dwh.query_log_enriched
        SET
            llm_error        = %(error)s,
            llm_processed_at = %(processed_at)s
        WHERE query_md5 = %(query_md5)s
    """
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    params = [
        {
            "error":        (r.error or "unknown")[:1000],
            "processed_at": now,
            "query_md5":    r.query_md5,
        }
        for r in results
    ]
    with doris_connection(cfg) as conn:
        with conn.cursor() as cur:
            cur.executemany(sql, params)

# ══════════════════════════════════════════════════════════════════════════════
# ASYNC LLM-ВЫЗОВ С RETRY  (httpx)
# ══════════════════════════════════════════════════════════════════════════════

async def call_llm_with_retry(
    client: httpx.AsyncClient,
    record: QueryRecord,
    config: LLMConfig,
) -> RecordResult:
    last_error: str | None = None

    for attempt in range(1, config.max_retries + 1):
        try:
            payload = build_payload(record, config.model)
            resp = await client.post(config.url, json=payload)

            # Retry на временные ошибки сервера
            if resp.status_code in config.retry_on_status:
                last_error = f"HTTP {resp.status_code}"
                logger.warning(
                    f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}"
                )
                await _backoff(attempt, config)
                continue

            # Постоянные ошибки — не ретраим
            if resp.status_code != 200:
                return RecordResult(
                    query_md5=record.query_md5,
                    success=False,
                    error=f"HTTP {resp.status_code}: {resp.text[:200]}",
                    attempts=attempt,
                )

            raw = resp.json()

        except httpx.TimeoutException as e:
            last_error = f"timeout: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

        except httpx.RemoteProtocolError as e:
            # Аналог ServerDisconnected из aiohttp
            last_error = f"remote protocol error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

        except httpx.ConnectError as e:
            last_error = f"connect error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

        except httpx.HTTPError as e:
            # Все остальные httpx-ошибки
            last_error = f"http error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

        # Парсинг ответа
        try:
            llm_result = parse_llm_response(raw)
            logger.debug(
                f"[{record.query_md5[:8]}] OK attempt={attempt} "
                f"domain={llm_result.domain} complexity={llm_result.complexity}"
            )
            return RecordResult(
                query_md5=record.query_md5,
                success=True,
                llm_result=llm_result,
                attempts=attempt,
            )

        except (KeyError, ValueError, json.JSONDecodeError) as e:
            last_error = f"parse error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            await _backoff(attempt, config)
            continue

    return RecordResult(
        query_md5=record.query_md5,
        success=False,
        error=last_error,
        attempts=config.max_retries,
    )


async def _backoff(attempt: int, config: LLMConfig) -> None:
    delay = (
        config.retry_delays[attempt - 1]
        if attempt <= len(config.retry_delays)
        else 30
    )
    logger.info(f"  backoff {delay}s...")
    await asyncio.sleep(delay)

# ══════════════════════════════════════════════════════════════════════════════

async def run_with_flush(
    records: list[QueryRecord],
    doris_cfg: DorisConfig,
    llm_cfg: LLMConfig,
    write_every: int,
) -> list[RecordResult]:
    semaphore = asyncio.Semaphore(llm_cfg.max_concurrent)
    all_results: list[RecordResult] = []
    buffer: list[RecordResult] = []

    timeout = httpx.Timeout(
        total=llm_cfg.timeout_total,
        connect=llm_cfg.timeout_connect,
    )
    headers = {
        "Authorization": f"Bearer {llm_cfg.api_key}",
        "Content-Type": "application/json",
        "Accept": "application/json",
    }

    async def bounded(record: QueryRecord) -> RecordResult:
        async with semaphore:
            return await call_llm_with_retry(client, record, llm_cfg)

    # httpx.AsyncClient — один на весь батч
    async with httpx.AsyncClient(
        headers=headers,
        timeout=timeout,
        http2=False,          # HTTP/1.1 — стабильнее с большинством LLM API
        follow_redirects=True,
        limits=httpx.Limits(
            max_connections=llm_cfg.max_concurrent + 10,
            max_keepalive_connections=llm_cfg.max_concurrent,
            keepalive_expiry=30.0,
        ),
    ) as client:
        tasks = [bounded(r) for r in records]
        total = len(tasks)
        done = 0

        for coro in asyncio.as_completed(tasks):
            result = await coro
            buffer.append(result)
            all_results.append(result)
            done += 1

            # Промежуточный flush в Doris
            if len(buffer) >= write_every:
                logger.info(f"Flushing {len(buffer)} results to Doris...")
                write_results(doris_cfg, buffer)
                buffer.clear()

            if done % 10 == 0 or done == total:
                ok = sum(1 for r in all_results if r.success)
                logger.info(
                    f"Progress: {done}/{total} | OK: {ok} | FAIL: {done - ok}"
                )

        # Финальный flush
        if buffer:
            logger.info(f"Final flush: {len(buffer)} results to Doris...")
            write_results(doris_cfg, buffer)

    return all_results

# ТОЧКА ВХОДА
# ══════════════════════════════════════════════════════════════════════════════

def run_enrichment_pipeline(
    doris_cfg: DorisConfig,
    llm_cfg: LLMConfig,
    batch_size: int = 500,
    write_every: int = 50,
) -> dict:
    start = datetime.now()

    records = fetch_pending_records(doris_cfg, batch_size)
    if not records:
        logger.info("No pending records. Exiting.")
        return {"processed": 0, "success": 0, "failed": 0}

    all_results = asyncio.run(
        run_with_flush(records, doris_cfg, llm_cfg, write_every)
    )

    elapsed = (datetime.now() - start).total_seconds()
    success = sum(1 for r in all_results if r.success)
    failed  = len(all_results) - success

    stats = {
        "processed":   len(all_results),
        "success":     success,
        "failed":      failed,
        "elapsed_sec": round(elapsed, 1),
        "rps":         round(len(all_results) / elapsed, 2) if elapsed > 0 else 0,
    }
    logger.info(f"Pipeline done: {stats}")
    return stats

import threading
_thread_local = threading.local()

def get_session(config: LLMConfig) -> requests.Session:
    """Возвращает Session для текущего потока, создаёт если нет."""
    if not hasattr(_thread_local, "session"):
        session = requests.Session()
        session.headers.update({
            "Authorization": f"Bearer {config.api_key}",
            "Content-Type": "application/json",
            "Accept": "application/json",
        })
        # Retry на уровне urllib3 — для connection-ошибок до HTTP-ответа
        from requests.adapters import HTTPAdapter
        from urllib3.util.retry import Retry
        retry_policy = Retry(
            total=0,               # retry логику ведём сами — здесь отключаем
            connect=3,             # но переподключение при обрыве — разрешаем
            read=0,
            backoff_factor=1,
        )
        adapter = HTTPAdapter(
            max_retries=retry_policy,
            pool_connections=1,    # на поток — одно соединение
            pool_maxsize=1,
        )
        session.mount("https://", adapter)
        session.mount("http://",  adapter)
        _thread_local.session = session

    return _thread_local.session

@contextmanager
def doris_connection(cfg: DorisConfig):
    conn = pymysql.connect(
        host=cfg.host,
        port=cfg.port,
        user=cfg.user,
        password=cfg.password,
        database=cfg.database,
        charset=cfg.charset,
        connect_timeout=10,
        autocommit=False,
    )
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def fetch_pending_records(cfg: DorisConfig, batch_size: int = 500) -> list[QueryRecord]:
    sql = """
        SELECT
            query_md5,
            sql_text,
            db_name,
            user_name,
            query_count
        FROM dwh.query_log_enriched
        WHERE llm_processed_at IS NULL
          AND sql_text IS NOT NULL
          AND LENGTH(sql_text) > 10
        ORDER BY query_count DESC
        LIMIT %(batch_size)s
    """
    with doris_connection(cfg) as conn:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(sql, {"batch_size": batch_size})
            rows = cur.fetchall()

    logger.info(f"Fetched {len(rows)} pending records from Doris")
    return [
        QueryRecord(
            query_md5=r["query_md5"],
            sql_text=r["sql_text"],
            db_name=r["db_name"] or "",
            user_name=r["user_name"] or "",
            query_count=r["query_count"] or 0,
        )
        for r in rows
    ]


def write_results(cfg: DorisConfig, results: list[RecordResult]) -> None:
    successful = [r for r in results if r.success]
    failed     = [r for r in results if not r.success]
    if successful:
        _upsert_successful(cfg, successful)
    if failed:
        _upsert_failed(cfg, failed)
    logger.info(f"Written to Doris: {len(successful)} OK, {len(failed)} failed")


def _upsert_successful(cfg: DorisConfig, results: list[RecordResult]) -> None:
    sql = """
        UPDATE dwh.query_log_enriched
        SET
            llm_domain       = %(domain)s,
            llm_intent       = %(intent)s,
            llm_complexity   = %(complexity)s,
            llm_main_tables  = %(main_tables)s,
            llm_comment      = %(comment)s,
            llm_processed_at = %(processed_at)s,
            llm_error        = NULL
        WHERE query_md5 = %(query_md5)s
    """
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    params = [
        {
            "domain":       r.llm_result.domain,
            "intent":       r.llm_result.intent,
            "complexity":   r.llm_result.complexity,
            "main_tables":  json.dumps(r.llm_result.main_tables, ensure_ascii=False),
            "comment":      r.llm_result.comment,
            "processed_at": now,
            "query_md5":    r.query_md5,
        }
        for r in results
    ]
    with doris_connection(cfg) as conn:
        with conn.cursor() as cur:
            cur.executemany(sql, params)


def _upsert_failed(cfg: DorisConfig, results: list[RecordResult]) -> None:
    sql = """
        UPDATE dwh.query_log_enriched
        SET
            llm_error        = %(error)s,
            llm_processed_at = %(processed_at)s
        WHERE query_md5 = %(query_md5)s
    """
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    params = [
        {
            "error":        (r.error or "unknown")[:1000],
            "processed_at": now,
            "query_md5":    r.query_md5,
        }
        for r in results
    ]
    with doris_connection(cfg) as conn:
        with conn.cursor() as cur:
            cur.executemany(sql, params)

def build_payload(record: QueryRecord, model: str) -> dict:
    system_prompt = """Ты — аналитик SQL-запросов корпоративного DWH на базе StarRocks.
Тебе дают SQL-запрос. Верни ТОЛЬКО валидный JSON без пояснений и markdown-блоков.

Формат ответа:
{
  "domain": "<finance | hr | sales | logistics | analytics | other>",
  "intent": "<краткое намерение запроса на русском, 5–15 слов>",
  "complexity": "<low | medium | high>",
  "main_tables": ["<таблица1>", "<таблица2>"],
  "comment": "<необязательный комментарий>"
}

Правила сложности:
- low: простой SELECT, 1–2 таблицы, без оконных функций
- medium: JOIN 2–4 таблиц, агрегации, подзапросы
- high: 5+ таблиц, оконные функции, CTE, UNION"""

    user_prompt = (
        f"База: {record.db_name}\n"
        f"Пользователь: {record.user_name}\n"
        f"Частота выполнения: {record.query_count}\n\n"
        f"SQL:\n{record.sql_text[:3000]}"
    )

    return {
        "model": model,
        "max_tokens": 512,
        "temperature": 0.1,
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user",   "content": user_prompt},
        ],
    }


def parse_llm_response(raw: dict) -> LLMResult:
    content = raw["choices"][0]["message"]["content"].strip()

    json_match = re.search(r"```(?:json)?\s*(\{.*?\})\s*```", content, re.DOTALL)
    json_str = json_match.group(1) if json_match else content

    data = json.loads(json_str)

    valid_domains    = {"finance", "hr", "sales", "logistics", "analytics", "other"}
    valid_complexity = {"low", "medium", "high"}

    domain = str(data.get("domain", "other")).lower()
    if domain not in valid_domains:
        domain = "other"

    complexity = str(data.get("complexity", "medium")).lower()
    if complexity not in valid_complexity:
        complexity = "medium"

    intent = str(data.get("intent", "")).strip()
    if not intent:
        raise ValueError("LLM returned empty intent")

    main_tables = data.get("main_tables", [])
    if not isinstance(main_tables, list):
        main_tables = []

    return LLMResult(
        domain=domain,
        intent=intent,
        complexity=complexity,
        main_tables=[str(t) for t in main_tables[:20]],
        comment=str(data.get("comment", ""))[:500],
    )

# LLM-ВЫЗОВ С RETRY  (requests, вызывается из потока)
# ══════════════════════════════════════════════════════════════════════════════

def call_llm_with_retry(record: QueryRecord, config: LLMConfig) -> RecordResult:
    """Синхронная функция — запускается в отдельном потоке."""
    session = get_session(config)
    last_error: str | None = None

    for attempt in range(1, config.max_retries + 1):
        try:
            payload = build_payload(record, config.model)
            resp = session.post(
                config.url,
                json=payload,
                timeout=config.timeout_sec,
            )

            if resp.status_code in config.retry_on_status:
                last_error = f"HTTP {resp.status_code}"
                logger.warning(
                    f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}"
                )
                _backoff(attempt, config)
                continue

            if resp.status_code != 200:
                return RecordResult(
                    query_md5=record.query_md5,
                    success=False,
                    error=f"HTTP {resp.status_code}: {resp.text[:200]}",
                    attempts=attempt,
                )

            raw = resp.json()

        except requests.Timeout:
            last_error = f"timeout after {config.timeout_sec}s"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            _backoff(attempt, config)
            continue

        except requests.ConnectionError as e:
            # Покрывает ServerDisconnected, ConnectError и пр.
            last_error = f"connection error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            # Пересоздаём сессию — старое соединение могло протухнуть
            _reset_session()
            _backoff(attempt, config)
            continue

        except requests.RequestException as e:
            last_error = f"request error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            _backoff(attempt, config)
            continue

        # Парсинг
        try:
            llm_result = parse_llm_response(raw)
            logger.debug(
                f"[{record.query_md5[:8]}] OK attempt={attempt} "
                f"domain={llm_result.domain} complexity={llm_result.complexity}"
            )
            return RecordResult(
                query_md5=record.query_md5,
                success=True,
                llm_result=llm_result,
                attempts=attempt,
            )

        except (KeyError, ValueError, json.JSONDecodeError) as e:
            last_error = f"parse error: {e}"
            logger.warning(f"[{record.query_md5[:8]}] attempt {attempt}: {last_error}")
            _backoff(attempt, config)
            continue

    return RecordResult(
        query_md5=record.query_md5,
        success=False,
        error=last_error,
        attempts=config.max_retries,
    )


def _backoff(attempt: int, config: LLMConfig) -> None:
    delay = (
        config.retry_delays[attempt - 1]
        if attempt <= len(config.retry_delays)
        else 30
    )
    logger.info(f"  backoff {delay}s...")
    time.sleep(delay)


def _reset_session() -> None:
    """Удаляем сессию потока — при следующем вызове get_session создаст новую."""
    if hasattr(_thread_local, "session"):
        try:
            _thread_local.session.close()
        except Exception:
            pass
        del _thread_local.session

def run_enrichment_pipeline(
    doris_cfg: DorisConfig,
    llm_cfg: LLMConfig,
    batch_size: int = 500,
    write_every: int = 50,
) -> dict:
    start = datetime.now()

    records = fetch_pending_records(doris_cfg, batch_size)
    if not records:
        logger.info("No pending records. Exiting.")
        return {"processed": 0, "success": 0, "failed": 0}

    all_results: list[RecordResult] = []
    buffer: list[RecordResult] = []
    # Lock нужен т.к. flush в Doris вызывается из главного потока,
    # но счётчики меняются по мере прихода future-результатов
    flush_lock = Lock()

    with ThreadPoolExecutor(max_workers=llm_cfg.max_workers) as executor:
        futures = {
            executor.submit(call_llm_with_retry, record, llm_cfg): record
            for record in records
        }
        total = len(futures)
        done  = 0

        for future in as_completed(futures):
            record = futures[future]
            try:
                result = future.result()
            except Exception as e:
                # Страховка: необработанное исключение из потока
                logger.error(f"[{record.query_md5[:8]}] unhandled exception: {e}")
                result = RecordResult(
                    query_md5=record.query_md5,
                    success=False,
                    error=f"unhandled: {e}",
                    attempts=0,
                )

            with flush_lock:
                buffer.append(result)
                all_results.append(result)
                done += 1

                if len(buffer) >= write_every:
                    logger.info(f"Flushing {len(buffer)} results to Doris...")
                    write_results(doris_cfg, buffer)
                    buffer.clear()

                if done % 10 == 0 or done == total:
                    ok = sum(1 for r in all_results if r.success)
                    logger.info(
                        f"Progress: {done}/{total} | OK: {ok} | FAIL: {done - ok}"
                    )

    # Финальный flush
    if buffer:
        logger.info(f"Final flush: {len(buffer)} results to Doris...")
        write_results(doris_cfg, buffer)

    elapsed = (datetime.now() - start).total_seconds()
    success = sum(1 for r in all_results if r.success)
    failed  = len(all_results) - success

    stats = {
        "processed":   len(all_results),
        "success":     success,
        "failed":      failed,
        "elapsed_sec": round(elapsed, 1),
        "rps":         round(len(all_results) / elapsed, 2) if elapsed > 0 else 0,
    }
    logger.info(f"Pipeline done: {stats}")
    return stats

# Определяем "контекст" колонки — в каком клозе она находится
# ---------------------------------------------------------------------------

def get_column_context(col_node: exp.Column) -> str:
    """
    Поднимается по дереву предков и определяет,
    в каком клозе находится колонка.
    """
    for ancestor in col_node.walk_up():
        if isinstance(ancestor, (exp.Column, exp.Table)):
            continue
        if isinstance(ancestor, exp.Select):
            return "select"
        if isinstance(ancestor, exp.Where):
            return "where"
        if isinstance(ancestor, (exp.Join,)):
            return "join"
        if isinstance(ancestor, exp.Group):
            return "group_by"
        if isinstance(ancestor, exp.Order):
            return "order_by"
        if isinstance(ancestor, exp.Having):
            return "having"
        # CTE-тело
        if isinstance(ancestor, exp.CTE):
            return "cte"
        # Subquery (Subquery — это SELECT внутри выражения)
        if isinstance(ancestor, exp.Subquery):
            return "subquery"
    return "other"


# ---------------------------------------------------------------------------
# Резолв алиасов таблиц через sqlglot qualify
# ---------------------------------------------------------------------------

def try_qualify(statement: exp.Expression, dialect: str) -> exp.Expression:
    """
    Пытается резолвить алиасы (o -> orders).
    Без schema резолвит не всё, но убирает большинство алиасов.
    """
    try:
        return qualify.qualify(
            statement,
            dialect=dialect,
            qualify_columns=True,
            qualify_tables=True,
            expand_alias_refs=True,
            infer_schema=True,
        )
    except Exception:
        return statement


# ---------------------------------------------------------------------------
# Основная функция
# ---------------------------------------------------------------------------

def analyze_column_usage(
    queries: list[str],
    dialect: str = "starrocks",
    use_qualify: bool = True,
) -> dict[tuple, ColumnStat]:
    """
    Парсит все запросы и собирает статистику по каждой колонке.

    Возвращает dict: {(table_or_None, column_name): ColumnStat}
    """
    stats: dict[tuple, ColumnStat] = {}
    errors = []

    for i, sql in enumerate(queries):
        if not sql or not sql.strip():
            continue
        try:
            statements = sqlglot.parse(sql.strip(), dialect=dialect)
        except Exception as e:
            errors.append((i, str(e)))
            continue

        for stmt in statements:
            if stmt is None:
                continue

            if use_qualify:
                stmt = try_qualify(stmt, dialect)

            # find_all рекурсивно обходит ВСЁ дерево:
            # CTE, подзапросы, джойны, CASE WHEN, HAVING и тд.
            for col in stmt.find_all(exp.Column):
                col_name = col.name.lower().strip()
                if not col_name or col_name == "*":
                    continue

                # Таблица/алиас (после qualify — уже резолвленное имя)
                table_name = (col.table or "").lower().strip() or None

                key = (table_name, col_name)
                if key not in stats:
                    stats[key] = ColumnStat(table=table_name, column=col_name)

                stat = stats[key]
                stat.total += 1

                ctx = get_column_context(col)
                if ctx == "select":    stat.in_select += 1
                elif ctx == "where":   stat.in_where += 1
                elif ctx == "join":    stat.in_join += 1
                elif ctx == "group_by": stat.in_group_by += 1
                elif ctx == "order_by": stat.in_order_by += 1
                elif ctx == "having":  stat.in_having += 1
                elif ctx == "cte":     stat.in_cte += 1
                elif ctx == "subquery": stat.in_subquery += 1

    if errors:
        print(f"[WARN] Не удалось распарсить {len(errors)} запросов")

    return stats

def print_report(stats: dict[tuple, ColumnStat], top_n: int = 50):
    sorted_stats = sorted(stats.values(), key=lambda s: s.total, reverse=True)

def _strip_leading_comments(sql: str) -> tuple[str, str]:
    """
    Отделяет ведущий блок комментариев/пробелов от тела запроса.
    Возвращает (leading, body).
    """
    pattern = re.compile(
        r'^(?P<lead>(?:\s|--[^\n]*\n?|/\*.*?\*/)*)',
        re.DOTALL,
    )
    m = pattern.match(sql)
    lead = m.group("lead") if m else ""
    body = sql[len(lead):]
    return lead, body


def _already_has_explain(body: str) -> bool:
    return re.match(r'EXPLAIN\b', body, re.IGNORECASE) is not None


def inject_explain_sqlglot(sql: str) -> tuple[str, str | None]:
    """
    Использует sqlglot для определения структуры запроса.

    Стратегия:
      - Парсим AST через sqlglot (диалект starrocks/mysql).
      - Если корневой узел — With или Select/Insert/Update/Delete →
        EXPLAIN ставится ДО всей конструкции, включая WITH.
      - StarRocks синтаксис: EXPLAIN <query>
        где <query> может начинаться с WITH.
      - Если sqlglot не смог распарсить — fallback на regex.

    Возвращает (explain_sql, warning_or_none).
    """
    lead, body = _strip_leading_comments(sql)

    if _already_has_explain(body):
        return sql, None

    warning = None

    # --- попытка разобрать через sqlglot ---
    try:
        statements = sqlglot.parse(body, dialect="mysql", error_level=sqlglot.ErrorLevel.RAISE)
    except sqlglot.errors.ParseError:
        statements = None

    if statements:
        stmt = statements[0]  # берём первый statement

        # sqlglot корректно распознаёт:
        #   exp.With      → WITH cte AS (...) SELECT ...
        #   exp.Select    → SELECT ...
        #   exp.Insert    → INSERT INTO ...
        #   exp.Update    → UPDATE ...
        #   exp.Delete    → DELETE FROM ...
        #   exp.Union     → SELECT ... UNION SELECT ...
        supported = (exp.With, exp.Select, exp.Insert, exp.Update, exp.Delete, exp.Union)

        if isinstance(stmt, supported):
            # EXPLAIN идёт перед ВСЕМ statement-ом (включая WITH)
            # Просто prepend к body — это и есть правильный StarRocks синтаксис
            return lead + "EXPLAIN " + body, None
        else:
            warning = f"sqlglot: unexpected top-level node type {type(stmt).__name__}, falling back to regex"
    else:
        warning = "sqlglot: parse failed, falling back to regex"

    # --- fallback: regex ---
    # Ищем первое DML-слово (WITH / SELECT / INSERT / UPDATE / DELETE)
    # и ставим EXPLAIN перед ним
    dml_re = re.compile(r'\b(WITH|SELECT|INSERT|UPDATE|DELETE)\b', re.IGNORECASE)
    m = dml_re.search(body)
    if m:
        pos = m.start()
        injected = lead + body[:pos] + "EXPLAIN " + body[pos:]
        return injected, warning

    # Если ничего не нашли — отправляем как есть (StarRocks вернёт ошибку сам)
    return sql, (warning or "Could not determine injection point")

                conn.close()
            except Exception:
                pass

def explain_sql(request: SQLRequest):
    """
    Принимает SQL-запрос, оборачивает его в EXPLAIN и выполняет в StarRocks.

        explain_plan=result if status == "OK" else None,
        error_message=(result if status == "ERROR" else None) or warning,
        injected_sql=injected,
    )

# ─── ТИП 1: Точное совпадение по схеме и имени ───────────────────────────────
def search_exact(cursor, owner: str, table_name: str) -> list[TableMatch]:
    cursor.execute("""
        SELECT owner, table_name, num_rows
        FROM   dba_tables
        WHERE  UPPER(owner)      = UPPER(:owner)
          AND  UPPER(table_name) = UPPER(:table_name)
    """, owner=owner, table_name=table_name)

    return [
        TableMatch(row[0], row[1], row[2], "1 — точное совпадение")
        for row in cursor.fetchall()
    ]

def search_exact_stage(cursor, owner: str, table_name: str) -> list[TableMatch]:
    stage_name = f"STAGE${table_name.upper()}"

    cursor.execute("""
        SELECT owner, table_name, num_rows
        FROM   dba_tables
        WHERE  UPPER(owner)      = UPPER(:owner)
          AND  UPPER(table_name) = :stage_name
    """, owner=owner, stage_name=stage_name)

    return [
        TableMatch(row[0], row[1], row[2], "2 — точное совпадение + stage$")
        for row in cursor.fetchall()
    ]


# ─── ТИП 3: LIKE по имени с префиксом stage$ (любая схема) ───────────────────
def search_like_stage(cursor, table_name: str) -> list[TableMatch]:
    pattern = f"%STAGE${table_name.upper()}%"

    cursor.execute("""
        SELECT owner, table_name, num_rows
        FROM   dba_tables
        WHERE  UPPER(table_name) LIKE :pattern
        ORDER  BY owner, table_name
    """, pattern=pattern)

    return [
        TableMatch(row[0], row[1], row[2], "3 — LIKE + stage$")
        for row in cursor.fetchall()
    ]

batch_size  : int           = BATCH_SIZE,

) -> None:

```

total_source = read_source_count()
log.info(f"Строк в {SOURCE_TABLE}: {total_source:,}")

# 2. Загружаем известные fingerprints (инкрементальный режим)
existing_fps: set[str] = set()
if incremental:
    log.info("Загружаем существующие fingerprints из Doris...")
    existing_fps = load_existing_fingerprints()
    log.info(f"Известных fingerprints: {len(existing_fps):,}")

# 3. Читаем источник чанками
all_rows: list[dict] = []
offsets = range(0, total_source, READ_CHUNK)

log.info("Читаем исходные запросы из Doris...")
for offset in tqdm(offsets, desc="Чтение из Doris", unit="chunk"):
    all_rows.extend(read_source_chunk(offset, READ_CHUNK))

log.info(f"Загружено строк: {len(all_rows):,}")

# 4. Разбиваем на батчи для multiprocessing
batches = [
    all_rows[i : i + batch_size]
    for i in range(0, len(all_rows), batch_size)
]
workers = num_workers or mp.cpu_count()
log.info(
    f"Запускаем multiprocessing: "
    f"{len(batches)} батчей × {batch_size} строк, {workers} воркеров"
)

# 5. Параллельная идентификация
raw_results: list[dict] = []
with mp.Pool(processes=num_workers) as pool:
    for batch_result in tqdm(
        pool.imap_unordered(_process_batch, batches),
        total=len(batches),
        desc="Идентификация",
        unit="batch",
    ):
        raw_results.extend(batch_result)

log.info(f"Обработано записей: {len(raw_results):,}")

# 6. Агрегация дублей
log.info("Агрегируем дубли...")
df_raw           = pd.DataFrame(raw_results)
df_dedup, df_err = aggregate_duplicates(df_raw)

log.info(

)

def _md5(s: str) -> str:
    return hashlib.md5(s.encode("utf-8")).hexdigest()


def _regex_normalize(sql: str) -> str:
    """Fallback-нормализация без AST-парсера."""
    s = re.sub(r"'[^']*'",      "?", sql)
    s = re.sub(r"\b\d+(\.\d+)?\b", "?", s)
    s = re.sub(r"--[^\n]*",     " ", s)
    s = re.sub(r"/\*.*?\*/",    " ", s, flags=re.S)
    s = re.sub(r"\s+",          " ", s).strip().upper()
    return s


def structural_fingerprint(sql: str) -> tuple[str, bool, str]:
    """
    Возвращает (fingerprint, parse_ok, error_msg).

    Алгоритм:
      1. Парсим AST через sqlglot (dialect=starrocks)
      2. Заменяем все Literal-узлы → Placeholder (?)
      3. Нормализуем алиасы таблиц → __t__
      4. MD5 от итогового SQL

    Fallback при ошибке: regex-нормализация, parse_ok=False.
    """
    try:
        tree = sqlglot.parse_one(
            sql, dialect=DIALECT, error_level=sqlglot.ErrorLevel.RAISE
        )
        for node in tree.walk():
            if isinstance(node, exp.Literal):
                node.replace(exp.Placeholder())
        for node in tree.walk():
            if isinstance(node, exp.TableAlias):
                node.set("this", exp.Identifier(this="__t__"))
        return _md5(tree.sql(dialect=DIALECT)), True, ""
    except Exception as e:
        return _md5(_regex_normalize(sql)), False, str(e)[:300]


def semantic_key_and_meta(sql: str) -> tuple[str, str, list[str]]:
    """
    Возвращает (semantic_key, query_type, tables_list).
    Приоритет типов: WINDOW > SUBQUERY > JOIN_HEAVY > AGG > JOIN_SIMPLE > SIMPLE
    """
    try:
        tree = sqlglot.parse_one(
            sql, dialect=DIALECT, error_level=sqlglot.ErrorLevel.RAISE
        )
        tables = sorted({
            t.name.lower()
            for t in tree.find_all(exp.Table)
            if t.name
        })
        has_agg    = bool(list(tree.find_all(exp.AggFunc)))
        has_window = bool(list(tree.find_all(exp.Window)))
        has_sub    = bool(list(tree.find_all(exp.Subquery)))
        join_count = len(list(tree.find_all(exp.Join)))

        key = _md5(f"{tables}|agg={has_agg}|win={has_window}")

        if has_window:        qtype = "WINDOW"
        elif has_sub:         qtype = "SUBQUERY"
        elif join_count >= 3: qtype = "JOIN_HEAVY"
        elif has_agg:         qtype = "AGG"
        elif join_count > 0:  qtype = "JOIN_SIMPLE"
        else:                 qtype = "SIMPLE"

        return key, qtype, tables

    except Exception:
        raw_tables = re.findall(
            r"(?:FROM|JOIN)\s+([a-zA-Z_][a-zA-Z0-9_.]*)",
            sql, flags=re.IGNORECASE,
        )
        tables = sorted({t.lower().split(".")[-1] for t in raw_tables})
        return _md5(f"{tables}|agg=?|win=?"), "UNKNOWN", tables

def run_identification(
    num_workers : Optional[int] = NUM_WORKERS,
    batch_size  : int           = BATCH_SIZE,
    incremental : bool          = INCREMENTAL,
) -> None:

    # 1. Размер источника
    total_source = read_source_count()
    log.info(f"Строк в {SOURCE_TABLE}: {total_source:,}")

    # 2. Загружаем известные fingerprints (инкрементальный режим)
    existing_fps: set[str] = set()
    if incremental:
        log.info("Загружаем существующие fingerprints из Doris...")
        existing_fps = load_existing_fingerprints()
        log.info(f"Известных fingerprints: {len(existing_fps):,}")

    # 3. Читаем источник чанками
    all_rows: list[dict] = []
    offsets = range(0, total_source, READ_CHUNK)

    log.info("Читаем исходные запросы из Doris...")
    for offset in tqdm(offsets, desc="Чтение из Doris", unit="chunk"):
        all_rows.extend(read_source_chunk(offset, READ_CHUNK))

    log.info(f"Загружено строк: {len(all_rows):,}")

    # 4. Разбиваем на батчи для multiprocessing
    batches = [
        all_rows[i : i + batch_size]
        for i in range(0, len(all_rows), batch_size)
    ]
    workers = num_workers or mp.cpu_count()
    log.info(
        f"Запускаем multiprocessing: "
        f"{len(batches)} батчей × {batch_size} строк, {workers} воркеров"
    )

    # 5. Параллельная идентификация
    raw_results: list[dict] = []
    with mp.Pool(processes=num_workers) as pool:
        for batch_result in tqdm(
            pool.imap_unordered(_process_batch, batches),
            total=len(batches),
            desc="Идентификация",
            unit="batch",
        ):
            raw_results.extend(batch_result)

    log.info(f"Обработано записей: {len(raw_results):,}")

    # 6. Агрегация дублей
    log.info("Агрегируем дубли...")
    df_raw           = pd.DataFrame(raw_results)
    df_dedup, df_err = aggregate_duplicates(df_raw)

    log.info(
        f"Уникальных структур: {len(df_dedup):,} "
        f"(сжатие {len(df_dedup) / max(len(all_rows), 1) * 100:.1f}%)  |  "
        f"Ошибок парсинга: {len(df_err):,}"
    )

    # 7. Инкрементальная фильтрация
    if incremental and existing_fps:
        before   = len(df_dedup)
        df_dedup = df_dedup[~df_dedup["structural_fp"].isin(existing_fps)]
        log.info(
            f"Инкрементальная фильтрация: "
            f"{before - len(df_dedup):,} fp уже существуют, "
            f"новых для записи: {len(df_dedup):,}"
        )

    # 8. Запись в Doris через Stream Load
    if not df_dedup.empty:
        log.info(f"Записываем {len(df_dedup):,} строк → {SINK_TABLE}...")
        _stream_load(SINK_TABLE, df_dedup)
    else:
        log.info("Нет новых fingerprints для записи.")

    if not df_err.empty:
        log.info(f"Записываем {len(df_err):,} ошибок → {SINK_ERR_TABLE}...")
        err_cols = [
            "structural_fp", "raw_sql", "query_type",
            "tables_used", "parse_ok", "error_msg",
        ]
        _stream_load(SINK_ERR_TABLE, df_err[err_cols])

    log.info("Готово.")

Шаг 1 из пайплайна обработки SQL-логов StarRocks.

@dataclass
class IdentifiedQuery:
    structural_fp  : str          # MD5 нормализованного AST — основной ключ
    semantic_key   : str          # MD5(sorted_tables | has_agg | has_window)
    raw_sql        : str          # оригинальный SQL (первый встреченный)
    query_type     : str          # SIMPLE/AGG/WINDOW/JOIN_SIMPLE/JOIN_HEAVY/SUBQUERY/UNKNOWN
    tables_used    : str          # pipe-separated: "t1|t2|t3"
    exec_count     : int   = 1
    total_exec_ms  : int   = 0
    avg_exec_ms    : float = 0.0
    parse_ok       : bool  = True
    error_msg      : str   = ""
    llm_processed  : bool  = False
    llm_metadata_id: str   = ""


# ---------------------------------------------------------------------------
# Функции идентификации (top-level — обязательно для pickle в multiprocessing)
# ---------------------------------------------------------------------------

def _md5(s: str) -> str:
    return hashlib.md5(s.encode("utf-8")).hexdigest()


def _regex_normalize(sql: str) -> str:
    """Fallback-нормализация без AST-парсера."""
    s = re.sub(r"'[^']*'",      "?", sql)
    s = re.sub(r"\b\d+(\.\d+)?\b", "?", s)
    s = re.sub(r"--[^\n]*",     " ", s)
    s = re.sub(r"/\*.*?\*/",    " ", s, flags=re.S)
    s = re.sub(r"\s+",          " ", s).strip().upper()
    return s


def structural_fingerprint(sql: str) -> tuple[str, bool, str]:
    """
    Возвращает (fingerprint, parse_ok, error_msg).

    Алгоритм:
      1. Парсим AST через sqlglot (dialect=starrocks)
      2. Заменяем все Literal-узлы → Placeholder (?)
      3. Нормализуем алиасы таблиц → __t__
      4. MD5 от итогового SQL

    Fallback при ошибке: regex-нормализация, parse_ok=False.
    """
    try:
        tree = sqlglot.parse_one(
            sql, dialect=DIALECT, error_level=sqlglot.ErrorLevel.RAISE
        )
        for node in tree.walk():
            if isinstance(node, exp.Literal):
                node.replace(exp.Placeholder())
        for node in tree.walk():
            if isinstance(node, exp.TableAlias):
                node.set("this", exp.Identifier(this="__t__"))
        return _md5(tree.sql(dialect=DIALECT)), True, ""
    except Exception as e:
        return _md5(_regex_normalize(sql)), False, str(e)[:300]


def semantic_key_and_meta(sql: str) -> tuple[str, str, list[str]]:
    """
    Возвращает (semantic_key, query_type, tables_list).
    Приоритет типов: WINDOW > SUBQUERY > JOIN_HEAVY > AGG > JOIN_SIMPLE > SIMPLE
    """
    try:
        tree = sqlglot.parse_one(
            sql, dialect=DIALECT, error_level=sqlglot.ErrorLevel.RAISE
        )
        tables = sorted({
            t.name.lower()
            for t in tree.find_all(exp.Table)
            if t.name
        })
        has_agg    = bool(list(tree.find_all(exp.AggFunc)))
        has_window = bool(list(tree.find_all(exp.Window)))
        has_sub    = bool(list(tree.find_all(exp.Subquery)))
        join_count = len(list(tree.find_all(exp.Join)))

        key = _md5(f"{tables}|agg={has_agg}|win={has_window}")

        if has_window:        qtype = "WINDOW"
        elif has_sub:         qtype = "SUBQUERY"
        elif join_count >= 3: qtype = "JOIN_HEAVY"
        elif has_agg:         qtype = "AGG"
        elif join_count > 0:  qtype = "JOIN_SIMPLE"
        else:                 qtype = "SIMPLE"

        return key, qtype, tables

    except Exception:
        raw_tables = re.findall(
            r"(?:FROM|JOIN)\s+([a-zA-Z_][a-zA-Z0-9_.]*)",
            sql, flags=re.IGNORECASE,
        )
        tables = sorted({t.lower().split(".")[-1] for t in raw_tables})
        return _md5(f"{tables}|agg=?|win=?"), "UNKNOWN", tables


# ---------------------------------------------------------------------------
# Воркер multiprocessing
# ---------------------------------------------------------------------------

def _process_batch(rows: list[dict]) -> list[dict]:
    """
    Принимает список {sql, query_time_ms}.
    Возвращает список dict(IdentifiedQuery).
    Работает в отдельном процессе — не должен держать соединение с БД.
    """
    results = []
    for row in rows:
        sql     = (row.get("sql") or "").strip()
        exec_ms = int(row.get("query_time_ms") or 0)
        if not sql:
            continue

        fp, parse_ok, error_msg = structural_fingerprint(sql)
        sk, qtype, tables       = semantic_key_and_meta(sql)

        results.append(asdict(IdentifiedQuery(
            structural_fp  = fp,
            semantic_key   = sk,
            raw_sql        = sql,
            query_type     = qtype,
            tables_used    = "|".join(tables),
            exec_count     = 1,
            total_exec_ms  = exec_ms,
            avg_exec_ms    = float(exec_ms),
            parse_ok       = parse_ok,
            error_msg      = error_msg,
        )))
    return results


# ---------------------------------------------------------------------------
# Doris: чтение через pymysql
# ---------------------------------------------------------------------------

def _doris_conn() -> pymysql.Connection:
    return pymysql.connect(
        host            = DORIS_HOST,
        port            = DORIS_FE_PORT,
        user            = DORIS_USER,
        password        = DORIS_PASSWORD,
        database        = DORIS_DB,
        charset         = "utf8mb4",
        connect_timeout = 30,
    )


def read_source_count() -> int:
    with _doris_conn() as conn:
        with conn.cursor() as cur:
            cur.execute(f"SELECT COUNT(*) FROM {SOURCE_TABLE}")
            return cur.fetchone()[0]


def read_source_chunk(offset: int, limit: int) -> list[dict]:
    """
    Читаем один чанк из исходной таблицы.
    ORDER BY гарантирует стабильность при OFFSET-пагинации.
    """
    sql = f"""
        SELECT
            {SOURCE_SQL_COL}  AS sql,
            {SOURCE_MS_COL}   AS query_time_ms
        FROM {SOURCE_TABLE}
        ORDER BY {SOURCE_SQL_COL}
        LIMIT {limit} OFFSET {offset}
    """
    with _doris_conn() as conn:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(sql)
            return list(cur.fetchall())


def load_existing_fingerprints() -> set[str]:
    """
    Инкрементальный режим: загружаем structural_fp уже записанных строк.
    При большом SINK_TABLE можно заменить на временную таблицу + JOIN на стороне Doris.
    """
    with _doris_conn() as conn:
        with conn.cursor() as cur:
            cur.execute(f"SELECT structural_fp FROM {SINK_TABLE}")
            return {row[0] for row in cur.fetchall()}


# ---------------------------------------------------------------------------
# Doris: запись через Stream Load
# ---------------------------------------------------------------------------

def _stream_load(table: str, df: pd.DataFrame) -> None:
    """
    Загружает DataFrame в Doris через Stream Load (JSON).
    Разбивает на батчи по STREAM_LOAD_BATCH строк.
    SINK_TABLE использует UNIQUE KEY → повторная загрузка = upsert.
    """
    if df.empty:
        return

    url = (
        f"http://{DORIS_HOST}:{DORIS_HTTP_PORT}"
        f"/api/{DORIS_DB}/{table}/_stream_load"
    )
    headers = {
        "format"            : "json",
        "strip_outer_array" : "true",
        "Expect"            : "100-continue",
        "max_filter_ratio"  : "0.0",
        "timeout"           : str(STREAM_LOAD_TIMEOUT),
    }

    total_loaded = 0
    for start in range(0, len(df), STREAM_LOAD_BATCH):
        chunk   = df.iloc[start : start + STREAM_LOAD_BATCH]
        payload = chunk.to_json(orient="records", force_ascii=False)

        resp = requests.put(
            url,
            data    = payload.encode("utf-8"),
            headers = headers,
            auth    = (DORIS_USER, DORIS_PASSWORD),
            timeout = STREAM_LOAD_TIMEOUT,
        )
        resp.raise_for_status()

        result = resp.json()
        status = result.get("Status", "")
        if status not in ("Success", "Publish Timeout"):
            raise RuntimeError(
                f"Stream Load failed [{table}]: "
                f"{json.dumps(result, ensure_ascii=False)}"
            )

        loaded = result.get("NumberLoadedRows", 0)
        total_loaded += loaded
        log.info(
            f"  [{table}] batch {start}–{start + len(chunk)}: "
            f"loaded={loaded}, txn={result.get('TxnId', '?')}"
        )

    log.info(f"Stream Load [{table}] итого: {total_loaded:,} строк")
