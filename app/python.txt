"""
Автономная цепочка: Kafka(alerts) -> LangGraph agent (Kimi + MCP: db/git) -> Kafka(results)

Архитектура:
  - KafkaConsumer читает алерты, кладёт их в asyncio.Queue
  - пул воркеров (ограничен семафором) разбирает очередь, гоняет LangGraph-граф
  - commit делается вручную ПОСЛЕ успешной обработки (или отправки в DLQ)
  - результат — строго структурированный JSON (Pydantic), не свободный текст

pip install langgraph langchain-mcp-adapters langchain-openai aiokafka pydantic
"""

import asyncio
import json
import logging
from typing import Literal

from aiokafka import AIOKafkaConsumer, AIOKafkaProducer
from langchain_mcp_adapters.client import MultiServerMCPClient
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, MessagesState, START, END
from langgraph.prebuilt import ToolNode, tools_condition
from pydantic import BaseModel, Field

logging.basicConfig(level=logging.INFO)
log = logging.getLogger("flink-debug-agent")

# ---------------------------------------------------------------------------
# 1. Модель результата — агент обязан вернуть это, а не "простыню" текста.
#    Формируется отдельным финальным узлом графа (structured output call),
#    а не парсингом последнего сообщения регулярками.
# ---------------------------------------------------------------------------

class DebugResult(BaseModel):
    alert_id: str
    status: Literal["root_cause_found", "needs_human", "not_reproducible", "error"]
    summary: str = Field(description="1-3 предложения, что произошло")
    root_cause: str | None = None
    suggested_action: str | None = None
    evidence_refs: list[str] = Field(default_factory=list, description="job_id/query_id/commit sha и т.п.")


# ---------------------------------------------------------------------------
# 2. LLM (Kimi через ваш LiteLLM-прокси, OpenAI-совместимый эндпоинт)
# ---------------------------------------------------------------------------

llm = ChatOpenAI(
    model="kimi-k2-5",                       # алиас, как в вашем litellm config.yaml
    base_url="http://litellm-proxy.internal:4000/v1",
    api_key="dummy-not-used-by-proxy",       # реальный auth — на уровне прокси/сети
    temperature=0,
    timeout=120,
)

# structured-output модель для финального узла (function calling под капотом)
llm_structured = llm.with_structured_output(DebugResult)


# ---------------------------------------------------------------------------
# 3. MCP-клиент: ваши существующие database + git серверы.
#    stateless по умолчанию — на каждый tool call новая сессия, это ок для
#    независимых вызовов; если серверу нужен persistent-контекст, см. client.session().
# ---------------------------------------------------------------------------

mcp_client = MultiServerMCPClient(
    {
        "database": {
            "transport": "http",
            "url": "http://mcp-database.internal:8080/mcp",
            "headers": {"Authorization": "Bearer ${MCP_DB_TOKEN}"},  # service-account токен, не личный
        },
        "git": {
            "transport": "http",
            "url": "http://mcp-git.internal:8081/mcp",
            "headers": {"Authorization": "Bearer ${MCP_GIT_TOKEN}"},
        },
    },
    handle_tool_errors=True,   # ошибка тула уходит модели как ToolMessage(status="error") -> агент сам решает, ретраить/сдаться
)


# ---------------------------------------------------------------------------
# 4. Граф. Не берём голый create_react_agent — нужен явный финальный узел
#    со structured output, иначе результат для Kafka придётся вытаскивать
#    регулярками из последнего AIMessage.
# ---------------------------------------------------------------------------

SYSTEM_PROMPT = """Ты — агент дебага Flink job'ов в DWH-платформе (Oracle Exadata -> StarRocks миграция).
Тебе на вход приходит алерт. Используй доступные инструменты (database, git) чтобы:
1. Найти релевантный job/запрос/таблицу
2. Проверить последние изменения схемы/кода (git blame/log)
3. Сформулировать гипотезу root cause
Если для однозначного вывода не хватает данных или требуется доступ, которого у тебя нет — не выдумывай,
пометь needs_human. Не выполняй деструктивные операции (DROP/DELETE/write) ни при каких обстоятельствах."""


async def build_graph():
    tools = await mcp_client.get_tools()
    model_with_tools = llm.bind_tools(tools)

    async def call_model(state: MessagesState):
        response = await model_with_tools.ainvoke(state["messages"])
        return {"messages": [response]}

    async def finalize(state: MessagesState, config) -> dict:
        alert_id = config["configurable"]["alert_id"]
        result: DebugResult = await llm_structured.ainvoke(
            state["messages"]
            + [{"role": "user", "content": "Сформируй итоговый структурированный результат по этому расследованию."}]
        )
        result.alert_id = alert_id
        return {"messages": [{"role": "assistant", "content": result.model_dump_json()}]}

    builder = StateGraph(MessagesState)
    builder.add_node("agent", call_model)
    builder.add_node("tools", ToolNode(tools))
    builder.add_node("finalize", finalize)

    builder.add_edge(START, "agent")
    builder.add_conditional_edges("agent", tools_condition, {"tools": "tools", END: "finalize"})
    builder.add_edge("tools", "agent")
    builder.add_edge("finalize", END)

    return builder.compile()


# ---------------------------------------------------------------------------
# 5. Kafka-обвязка. Это то, чего LangGraph сам по себе не даёт.
#    Ключевые решения:
#      - ручной commit ПОСЛЕ обработки (enable_auto_commit=False)
#      - обработка идёт в отдельных тасках, а poll-луп не блокируется
#        (иначе долгий ран агента триггерит rebalance по session/poll timeout)
#      - ограниченный параллелизм (Semaphore) — LLM+MCP дорогие по латентности
#      - DLQ-топик на любую необработанную ошибку, чтобы не терять алерты
# ---------------------------------------------------------------------------

KAFKA_BOOTSTRAP = "kafka.internal:9092"
TOPIC_IN = "dwh.alerts.flink"
TOPIC_OUT = "dwh.alerts.flink.debug-result"
TOPIC_DLQ = "dwh.alerts.flink.dlq"
GROUP_ID = "flink-debug-agent"
MAX_CONCURRENCY = 4  # сколько параллельных агентских ранов держим одновременно


async def process_one(graph, producer: AIOKafkaProducer, msg) -> None:
    alert = json.loads(msg.value)
    alert_id = alert.get("alert_id", f"{msg.topic}-{msg.partition}-{msg.offset}")

    try:
        final_state = await asyncio.wait_for(
            graph.ainvoke(
                {"messages": [
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": json.dumps(alert, ensure_ascii=False)},
                ]},
                config={"configurable": {"alert_id": alert_id}},
            ),
            timeout=300,  # хард-лимит на весь дебаг одного алерта
        )
        result_json = final_state["messages"][-1].content
        DebugResult.model_validate_json(result_json)  # финальная валидация контракта

        await producer.send_and_wait(TOPIC_OUT, key=alert_id.encode(), value=result_json.encode())
        log.info("alert %s processed ok", alert_id)

    except Exception as e:
        log.exception("alert %s failed: %s", alert_id, e)
        dlq_payload = {"alert_id": alert_id, "error": str(e), "original": alert}
        await producer.send_and_wait(TOPIC_DLQ, key=alert_id.encode(),
                                      value=json.dumps(dlq_payload, ensure_ascii=False).encode())


async def run():
    graph = await build_graph()

    consumer = AIOKafkaConsumer(
        TOPIC_IN,
        bootstrap_servers=KAFKA_BOOTSTRAP,
        group_id=GROUP_ID,
        enable_auto_commit=False,
        max_poll_interval_ms=600_000,   # запас под долгие агентские раны
        auto_offset_reset="earliest",
    )
    producer = AIOKafkaProducer(bootstrap_servers=KAFKA_BOOTSTRAP)

    await consumer.start()
    await producer.start()

    sem = asyncio.Semaphore(MAX_CONCURRENCY)
    inflight: set[asyncio.Task] = set()

    async def guarded(msg):
        async with sem:
            await process_one(graph, producer, msg)

    try:
        async for msg in consumer:
            task = asyncio.create_task(guarded(msg))
            inflight.add(task)
            task.add_done_callback(inflight.discard)

            # коммитим по мере готовности задач, не блокируя consume loop
            done = {t for t in inflight if t.done()}
            if done:
                await consumer.commit()
    finally:
        if inflight:
            await asyncio.gather(*inflight, return_exceptions=True)
            await consumer.commit()
        await consumer.stop()
        await producer.stop()


if __name__ == "__main__":
    asyncio.run(run())






"""
GitLab MCP Server на базе FastMCP (streamable-HTTP transport).
Полностью асинхронный: async def инструменты + httpx.AsyncClient.

Авторизация — per-request через HTTP-заголовки:
    X-GitLab-Url   — базовый URL корпоративного GitLab (без trailing slash)
    X-GitLab-Token — Personal Access Token пользователя (scope: api)

Запуск:
    pip install fastmcp httpx
    python gitlab_mcp_server.py

Переменные окружения (все опциональны):
    MCP_HOST              — хост HTTP-сервера          (default: 0.0.0.0)
    MCP_PORT              — порт                        (default: 8000)
    PROTECTED_BRANCHES    — ветки через запятую         (default: main,master,production)
    PROTECTED_PATHS       — пути через запятую          (default: .gitlab-ci.yml,Dockerfile)
    ALLOWED_BRANCH_PREFIX — префиксы через запятую      (default: feature/,fix/,agent/,chore/)
    MAX_FILE_SIZE_KB      — макс. размер файла в KB      (default: 1024)

Инструменты (11):
    1.  get_file_content       — содержимое файла или листинг директории
    2.  get_repository_tree    — дерево файлов/папок (рекурсивно, все страницы)
    3.  create_or_update_file  — создать/обновить один файл (один коммит)
    4.  commit_multiple_files  — атомарный коммит нескольких файлов
    5.  create_merge_request   — открыть MR
    6.  merge_merge_request    — принять (смёрджить) MR (с проверкой готовности)
    7.  list_commits           — список коммитов ветки
    8.  get_commit_details     — метаданные коммита
    9.  get_commit_diff        — diff коммита
    10. list_branches          — список веток
    11. search_code            — полнотекстовый поиск по коду
"""

import base64
import logging
import os
from typing import Any

import httpx
from fastmcp import FastMCP, Context
from fastmcp.exceptions import ToolError

# ---------------------------------------------------------------------------
# Логирование — отключаем debug-уровень чтобы токены не попадали в логи
# ---------------------------------------------------------------------------

# Uvicorn по умолчанию логирует все заголовки на уровне DEBUG.
# Принудительно поднимаем порог до WARNING для access-лога и httpx.
logging.getLogger("uvicorn.access").setLevel(logging.WARNING)
logging.getLogger("uvicorn").setLevel(logging.WARNING)
logging.getLogger("httpx").setLevel(logging.WARNING)      # httpx логирует заголовки на DEBUG
logging.getLogger("httpcore").setLevel(logging.WARNING)

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s: %(message)s")
logger = logging.getLogger("gitlab-mcp")

# ---------------------------------------------------------------------------
# Конфигурация безопасности (переопределяется через env)
# ---------------------------------------------------------------------------

MCP_HOST = os.environ.get("MCP_HOST", "0.0.0.0")
MCP_PORT = int(os.environ.get("MCP_PORT", "8000"))

# Проверка SSL-сертификата GitLab.
# Отключай только для корпоративных инстансов с самоподписанным сертификатом.
# Установи SSL_VERIFY=false чтобы отключить.
SSL_VERIFY: bool = os.environ.get("SSL_VERIFY", "true").lower() not in ("false", "0", "no")

# Ветки, в которые запрещена прямая запись — только через MR
PROTECTED_BRANCHES: frozenset[str] = frozenset(
    b.strip() for b in
    os.environ.get("PROTECTED_BRANCHES", "main,master,production").split(",")
    if b.strip()
)

# Файлы/пути, которые нельзя удалять через инструменты
PROTECTED_PATHS: frozenset[str] = frozenset(
    p.strip() for p in
    os.environ.get(
        "PROTECTED_PATHS",
        ".gitlab-ci.yml,Dockerfile,.env.example,pyproject.toml,pom.xml"
    ).split(",")
    if p.strip()
)

# Допустимые префиксы имён веток при создании (пустой список = любые)
_raw_prefixes = os.environ.get("ALLOWED_BRANCH_PREFIX", "feature/,fix/,agent/,chore/,docs/")
ALLOWED_BRANCH_PREFIXES: list[str] = [p.strip() for p in _raw_prefixes.split(",") if p.strip()]

# Максимальный размер содержимого файла
MAX_FILE_SIZE_BYTES: int = int(os.environ.get("MAX_FILE_SIZE_KB", "1024")) * 1024

logger.info("Protected branches : %s", PROTECTED_BRANCHES)
logger.info("Protected paths    : %s", PROTECTED_PATHS)
logger.info("Allowed prefixes   : %s", ALLOWED_BRANCH_PREFIXES)
logger.info("Max file size      : %d KB", MAX_FILE_SIZE_BYTES // 1024)
logger.info("SSL verify         : %s", SSL_VERIFY)
if not SSL_VERIFY:
    logger.warning("SSL verification DISABLED — допустимо только для корпоративного GitLab с самоподписанным сертификатом")

# ---------------------------------------------------------------------------
# Гарды безопасности — вызываются из инструментов перед записью
# ---------------------------------------------------------------------------

def _guard_branch_write(branch: str) -> None:
    """
    Запрещает прямую запись в защищённые ветки.
    Агент должен создать feature-ветку и открыть MR.
    """
    if branch in PROTECTED_BRANCHES:
        raise ToolError(
            f"Прямая запись в ветку '{branch}' запрещена. "
            f"Создай отдельную ветку и открой Merge Request. "
            f"Защищённые ветки: {', '.join(sorted(PROTECTED_BRANCHES))}."
        )


def _guard_branch_name(branch: str) -> None:
    """
    Проверяет что имя новой ветки начинается с допустимого префикса.
    Предотвращает засорение репозитория произвольными именами.
    Пропускает проверку если ALLOWED_BRANCH_PREFIXES пуст.
    """
    if not ALLOWED_BRANCH_PREFIXES:
        return
    if not any(branch.startswith(p) for p in ALLOWED_BRANCH_PREFIXES):
        raise ToolError(
            f"Имя ветки '{branch}' не соответствует соглашению. "
            f"Допустимые префиксы: {', '.join(ALLOWED_BRANCH_PREFIXES)}."
        )


def _guard_file_size(content: str) -> None:
    """Запрещает коммит файлов больше MAX_FILE_SIZE_BYTES."""
    size = len(content.encode("utf-8"))
    if size > MAX_FILE_SIZE_BYTES:
        raise ToolError(
            f"Размер файла {size // 1024} KB превышает лимит "
            f"{MAX_FILE_SIZE_BYTES // 1024} KB. "
            "Большие файлы не должны храниться в git — используй artifact storage."
        )


def _guard_delete_path(file_path: str) -> None:
    """Запрещает удаление критически важных файлов."""
    # Проверяем и точное совпадение, и имя файла без пути
    name = file_path.split("/")[-1]
    if file_path in PROTECTED_PATHS or name in PROTECTED_PATHS:
        raise ToolError(
            f"Удаление файла '{file_path}' запрещено — он находится в списке защищённых путей."
        )


# ---------------------------------------------------------------------------
# Извлечение учётных данных из заголовков входящего запроса
# ---------------------------------------------------------------------------

def _credentials(ctx: Context) -> tuple[str, str]:
    """
    Читает X-GitLab-Url и X-GitLab-Token из HTTP-заголовков текущего запроса.
    Токен намеренно нигде не логируется.

    Returns:
        (api_base, token)

    Raises:
        ToolError — если заголовок отсутствует.
    """
    headers = ctx.request_context.request.headers

    url   = headers.get("x-gitlab-url", "").rstrip("/")
    token = headers.get("x-gitlab-token", "")

    if not url:
        raise ToolError("Заголовок X-GitLab-Url обязателен.")
    if not token:
        raise ToolError("Заголовок X-GitLab-Token обязателен.")

    return f"{url}/api/v4", token


def _http_headers(token: str) -> dict:
    return {"PRIVATE-TOKEN": token, "Content-Type": "application/json"}


# ---------------------------------------------------------------------------
# Асинхронные HTTP-хелперы
# ---------------------------------------------------------------------------

async def _get(api_base: str, token: str, path: str, params: dict | None = None) -> Any:
    async with httpx.AsyncClient(headers=_http_headers(token), timeout=30.0) as c:
        r = await c.get(f"{api_base}{path}", params=params)
        r.raise_for_status()
        return r.json()


async def _post(api_base: str, token: str, path: str, json: dict) -> Any:
    async with httpx.AsyncClient(headers=_http_headers(token), timeout=30.0) as c:
        r = await c.post(f"{api_base}{path}", json=json)
        r.raise_for_status()
        return r.json()


async def _put(api_base: str, token: str, path: str, json: dict) -> Any:
    async with httpx.AsyncClient(headers=_http_headers(token), timeout=30.0) as c:
        r = await c.put(f"{api_base}{path}", json=json)
        r.raise_for_status()
        return r.json()


def _encode_id(project_id: str) -> str:
    """namespace/repo → namespace%2Frepo."""
    return project_id.replace("/", "%2F")


# ---------------------------------------------------------------------------
# FastMCP приложение
# ---------------------------------------------------------------------------

mcp = FastMCP(
    name="gitlab",
    description=(
        "Асинхронные инструменты для работы с корпоративным GitLab (API v4). "
        "Требует заголовки X-GitLab-Url и X-GitLab-Token в каждом запросе."
    ),
)

# ---------------------------------------------------------------------------
# Tool 1 — Содержимое файла или листинг директории
# ---------------------------------------------------------------------------

@mcp.tool()
async def get_file_content(
    project_id: str,
    path: str,
    ref: str = "main",
    ctx: Context = None,
) -> dict:
    """
    Возвращает содержимое файла (декодированное из base64) или список
    элементов директории для указанного пути.

    Возвращает поле last_commit_id — сохрани его если планируешь обновлять
    файл через create_or_update_file (защита от конфликтов).

    Args:
        project_id: ID или «namespace/repo» проекта.
        path:       Путь к файлу или директории.
        ref:        Ветка, тег или SHA (default: main).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    try:
        encoded_path = path.replace("/", "%2F")
        data = await _get(api_base, token,
                          f"/projects/{pid}/repository/files/{encoded_path}",
                          {"ref": ref})
        content = base64.b64decode(data["content"]).decode("utf-8", errors="replace")
        return {
            "type": "file",
            "file_name": data["file_name"],
            "file_path": data["file_path"],
            "size": data["size"],
            "ref": data["ref"],
            # last_commit_id нужен как last_commit_id при последующем update —
            # GitLab откажет в записи если файл изменился с этого коммита
            "last_commit_id": data["last_commit_id"],
            "content": content,
        }
    except httpx.HTTPStatusError as e:
        if e.response.status_code != 404:
            raise

    # Fallback: директория
    items = await _get(api_base, token,
                       f"/projects/{pid}/repository/tree",
                       {"path": path, "ref": ref, "per_page": 100})
    return {"type": "directory", "path": path, "ref": ref, "items": items}


# ---------------------------------------------------------------------------
# Tool 2 — Полное дерево файлов и папок
# ---------------------------------------------------------------------------

@mcp.tool()
async def get_repository_tree(
    project_id: str,
    path: str = "",
    ref: str = "main",
    recursive: bool = True,
    ctx: Context = None,
) -> list[dict]:
    """
    Возвращает дерево файлов и директорий репозитория (все страницы пагинации).

    Args:
        project_id: ID или «namespace/repo» проекта.
        path:       Стартовый путь (пустая строка = корень).
        ref:        Ветка/тег/SHA (default: main).
        recursive:  Обходить подпапки рекурсивно (default: True).
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    params: dict = {"ref": ref, "recursive": recursive, "per_page": 100}
    if path:
        params["path"] = path

    results: list[dict] = []
    page = 1
    while True:
        params["page"] = page
        batch = await _get(api_base, token, f"/projects/{pid}/repository/tree", params)
        if not batch:
            break
        results.extend(batch)
        if len(batch) < 100:
            break
        page += 1

    return results


# ---------------------------------------------------------------------------
# Tool 3 — Создать или обновить один файл
# ---------------------------------------------------------------------------

@mcp.tool()
async def create_or_update_file(
    project_id: str,
    file_path: str,
    content: str,
    commit_message: str,
    branch: str,
    last_commit_id: str | None = None,
    start_branch: str | None = None,
    ctx: Context = None,
) -> dict:
    """
    Создаёт новый файл или обновляет существующий одним коммитом.
    Метод (POST/PUT) определяется автоматически.

    Защиты:
    - Запрещена запись в защищённые ветки (main, master, production).
    - Имя ветки должно соответствовать соглашению (feature/, fix/, agent/ и т.д.).
    - Размер файла ограничен MAX_FILE_SIZE_KB.
    - last_commit_id предотвращает слепую перезапись чужих изменений.

    Args:
        project_id:     ID или «namespace/repo» проекта.
        file_path:      Путь к файлу в репозитории.
        content:        Новое содержимое файла (plain text, UTF-8).
        commit_message: Сообщение коммита.
        branch:         Целевая ветка. Не должна быть защищённой.
        last_commit_id: SHA коммита файла полученного через get_file_content.
                        Передавай при обновлении существующего файла —
                        GitLab вернёт ошибку если файл изменился параллельно.
        start_branch:   Создать новую ветку от этой базы (опционально).
    """
    # --- Гарды ---
    _guard_branch_write(branch)
    _guard_branch_name(branch)
    _guard_file_size(content)

    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)
    encoded_fp = file_path.replace("/", "%2F")

    payload: dict = {
        "branch": branch,
        "content": content,
        "commit_message": commit_message,
        "encoding": "text",
    }
    if start_branch:
        payload["start_branch"] = start_branch

    # Проверяем существование файла → выбираем метод
    try:
        await _get(api_base, token,
                   f"/projects/{pid}/repository/files/{encoded_fp}",
                   {"ref": branch})
        method = "update"
        # Передаём last_commit_id если предоставлен — защита от конфликтов
        if last_commit_id:
            payload["last_commit_id"] = last_commit_id
        result = await _put(api_base, token,
                            f"/projects/{pid}/repository/files/{encoded_fp}",
                            payload)
    except httpx.HTTPStatusError as e:
        if e.response.status_code != 404:
            raise
        method = "create"
        result = await _post(api_base, token,
                             f"/projects/{pid}/repository/files/{encoded_fp}",
                             payload)

    return {"action": method, **result}

# Tool 4 — Атомарный коммит нескольких файлов
# ---------------------------------------------------------------------------

@mcp.tool()
async def commit_multiple_files(
    project_id: str,
    branch: str,
    commit_message: str,
    actions: list[dict],
    start_branch: str | None = None,
    ctx: Context = None,
) -> dict:
    """
    Создаёт один атомарный коммит с произвольным набором изменений файлов.
    Используй когда нужно изменить несколько файлов за раз.

    Защиты:
    - Запрещена запись в защищённые ветки.
    - Имя ветки должно соответствовать соглашению.
    - Размер каждого файла ограничен MAX_FILE_SIZE_KB.
    - Запрещено удаление защищённых файлов (.gitlab-ci.yml, Dockerfile и др.).

    Поддерживаемые action:
        "create"  — создать новый файл         (требует: content)
        "update"  — обновить существующий       (требует: content)
        "delete"  — удалить файл
        "move"    — переместить/переименовать   (требует: previous_path)
        "chmod"   — изменить права              (требует: execute_filemode)

    Рекомендуется передавать last_commit_id для action=update — защита от
    параллельных изменений того же файла.

    Args:
        project_id:     ID или «namespace/repo» проекта.
        branch:         Целевая ветка. Не должна быть защищённой.
        commit_message: Сообщение коммита.
        actions:        Список операций над файлами.
        start_branch:   Создать новую ветку от этой базы (опционально).
    """
    # --- Гарды уровня ветки ---
    _guard_branch_write(branch)
    _guard_branch_name(branch)

    if not actions:
        raise ToolError("Список actions не может быть пустым.")

    # --- Валидация и гарды каждого action ---
    for i, action in enumerate(actions):
        act = action.get("action")
        fp  = action.get("file_path")

        if not act:
            raise ToolError(f"actions[{i}]: отсутствует поле 'action'.")
        if not fp:
            raise ToolError(f"actions[{i}]: отсутствует поле 'file_path'.")

        if act in ("create", "update") and "content" not in action:
            raise ToolError(f"actions[{i}]: action='{act}' требует поле 'content'.")
        if act == "move" and "previous_path" not in action:
            raise ToolError(f"actions[{i}]: action='move' требует поле 'previous_path'.")

        # Запрет удаления защищённых файлов
        if act == "delete":
            _guard_delete_path(fp)

        # Ограничение размера для файлов с содержимым
        if act in ("create", "update", "move") and "content" in action:
            _guard_file_size(action["content"])

    payload: dict = {
        "branch": branch,
        "commit_message": commit_message,
        "actions": actions,
    }
    if start_branch:
        payload["start_branch"] = start_branch

    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)
    return await _post(api_base, token, f"/projects/{pid}/repository/commits", payload)

# Tool 6 — Принять (смёрджить) Merge Request
# ---------------------------------------------------------------------------

@mcp.tool()
async def merge_merge_request(
    project_id: str,
    merge_request_iid: int,
    merge_commit_message: str | None = None,
    squash: bool = False,
    should_remove_source_branch: bool = False,
    ctx: Context = None,
) -> dict:
    """
    Принимает открытый Merge Request.

    Перед мёрджем проверяет:
    - MR существует и открыт (state == opened)
    - CI/CD пайплайн прошёл (merge_status == can_be_merged)
    - Нет блокирующих обсуждений

    Если условия не выполнены — возвращает ToolError с причиной.

    Args:
        project_id:                  ID или «namespace/repo» проекта.
        merge_request_iid:           Внутренний IID MR (не глобальный ID).
        merge_commit_message:        Кастомное сообщение merge-коммита.
        squash:                      Схлопнуть коммиты.
        should_remove_source_branch: Удалить source-ветку после мёрджа.
    """
    api_base, token = _credentials(ctx)
    pid = _encode_id(project_id)

    # Получаем актуальное состояние MR перед мёрджем
    mr = await _get(api_base, token, f"/projects/{pid}/merge_requests/{merge_request_iid}")

    # Проверка 1: MR должен быть открыт
    if mr.get("state") != "opened":
        raise ToolError(
            f"MR !{merge_request_iid} нельзя смёрджить: "
            f"текущий статус '{mr.get('state')}' (ожидается 'opened')."
        )

    # Проверка 2: CI/CD должен быть пройден
    merge_status = mr.get("merge_status", "")
    detailed_status = mr.get("detailed_merge_status", "")
    if merge_status != "can_be_merged":
        raise ToolError(
            f"MR !{merge_request_iid} не готов к мёрджу. "
            f"merge_status='{merge_status}', detailed='{detailed_status}'. "
            "Дождись прохождения CI/CD пайплайна и разрешения всех обсуждений."
        )

    # Проверка 3: нет блокирующих (unresolved) тредов
    if mr.get("blocking_discussions_resolved") is False:
        raise ToolError(
            f"MR !{merge_request_iid} содержит неразрешённые обсуждения. "
            "Разреши все треды перед мёрджем."
        )

    payload: dict = {
        "squash": squash,
        "should_remove_source_branch": should_remove_source_branch,
    }
    if merge_commit_message:
        payload["merge_commit_message"] = merge_commit_message

    return await _put(api_base, token,
                      f"/projects/{pid}/merge_requests/{merge_request_iid}/merge",
                      payload)

@mcp.tool()
async def list_commits(
    project_id: str,
    ref_name: str = "main",
    since: str | None = None,
    until: str | None = None,
    path: str | None = None,
    per_page: int = 20,
    ctx: Context = None,
) -> list[dict]:
    """
    Возвращает список коммитов репозитория.

    Args:
        project_id: ID или «namespace/repo» проекта.
        ref_name:   Ветка/тег/SHA (default: main).
        since:      ISO 8601 дата начала (опционально).

    if since: params["since"] = since
    if until: params["until"] = until
    if path:  params["path"]  = path

    return await _get(api_base, token, f"/projects/{pid}/repository/commits", params)

from fastmcp import FastMCP
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import JSONResponse
import asyncpg  # или aiosqlite

mcp = FastMCP("my-server")

class TokenAuthMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        # health-check пропускаем без токена
        if request.url.path == "/health":
            return await call_next(request)
        
        auth = request.headers.get("Authorization", "")
        if not auth.startswith("Bearer "):
            return JSONResponse({"error": "Unauthorized"}, status_code=401)
        
        token = auth.removeprefix("Bearer ").strip()
        
        if not await is_valid_token(token):
            return JSONResponse({"error": "Invalid token"}, status_code=401)
        
        return await call_next(request)

async def is_valid_token(token: str) -> bool:
    conn = await asyncpg.connect(DATABASE_URL)
    try:
        row = await conn.fetchrow(
            "SELECT 1 FROM mcp_tokens WHERE token = $1 AND is_active = true",
            token
        )
        return row is not None
    finally:
        await conn.close()

# Регистрируем middleware
mcp.app.add_middleware(TokenAuthMiddleware)
