SELECT
    -- Таблица
    t.owner,
    t.table_name,
    tc.comments AS table_comment,
    -- Колонки
    c.column_name,
    c.data_type,
    c.data_length,
    c.data_precision,
    c.data_scale,
    c.nullable,
    c.column_id,
    cc.comments AS column_comment
FROM
    all_tables t
    JOIN all_tab_comments tc ON tc.owner      = t.owner
                             AND tc.table_name = t.table_name
    JOIN all_tab_columns c  ON c.owner        = t.owner
                             AND c.table_name  = t.table_name
    JOIN all_col_comments cc ON cc.owner       = t.owner
                             AND cc.table_name  = c.table_name
                             AND cc.column_name = c.column_name
WHERE
    t.owner = 'SCHEMA_NAME'   -- обязательно указать схему
-- AND t.table_name = 'MY_TABLE'
ORDER BY
    t.owner,
    t.table_name,
    c.column_id;


SELECT
    -- Таблица
    t.table_schema        AS owner,
    t.table_name,
    t.table_comment,
    -- Колонки
    c.ordinal_position    AS column_id,
    c.column_name,
    c.data_type,
    c.character_maximum_length AS data_length,
    c.numeric_precision,
    c.numeric_scale,
    c.is_nullable         AS nullable,
    c.column_comment
FROM
    information_schema.tables t
    JOIN information_schema.columns c
        ON  c.table_schema = t.table_schema
        AND c.table_name   = t.table_name
WHERE
    t.table_schema = 'your_database'   -- название БД/схемы
--  AND t.table_name = 'my_table'
    AND t.table_type = 'BASE TABLE'    -- исключить views
ORDER BY
    t.table_schema,
    t.table_name,
    c.ordinal_position;


WITH allowed AS (
    -- Список разрешённых таблиц в формате SCHEMA.TABLE
    SELECT 'S004LSHOP.GOODS_AUTOFIL_METADATA' AS tbl UNION ALL
    SELECT 'S004LSHOP.ORDERS'                       UNION ALL
    SELECT 'S004LSHOP.ORDERS_MASTER'
    -- ... добавляй свои
),

-- Проверка: все таблицы запроса входят в allowed
table_check AS (
    SELECT
        t.query_id,
        COUNT(DISTINCT v.name)                                         AS total_cnt,
        COUNT(DISTINCT CASE WHEN a.tbl IS NOT NULL THEN v.name END)    AS allowed_cnt
    FROM ODS.TMP_SHOP_STARBUCKS_AUDIT_TBL_SELECT t
    CROSS JOIN LATERAL (
        SELECT value AS name
        FROM JSON_TABLE(
            t.tables, '$[*]'
            COLUMNS (value VARCHAR(255) PATH '$')
        )
    ) v
    LEFT JOIN allowed a ON v.name = a.tbl
    WHERE t.tables IS NOT NULL AND t.tables != '[]'
    GROUP BY t.query_id
),

valid_queries AS (
    SELECT query_id
    FROM table_check
    WHERE total_cnt = allowed_cnt AND total_cnt > 0
),

quality AS (
    SELECT
        t.query_id,
        t.query_time,
        t.sql_clean,
        t.tables,
        t.has_cte,
        t.has_group_by,
        t.has_limit,
        t.has_window,
        t.scan_bytes,
        t.scan_rows,
        t.return_rows,
        t.cpu_cost_ns,
        t.score,
        -- Сигнатура структурного разнообразия
        CONCAT_WS('_',
            CAST(t.has_cte      AS VARCHAR),
            CAST(t.has_group_by AS VARCHAR),
            CAST(t.has_window   AS VARCHAR),
            CAST(t.has_limit    AS VARCHAR),
            CAST(JSON_LENGTH(t.tables) AS VARCHAR),
            CAST(t.cols_cnt     AS VARCHAR)  -- уникальная деталь из твоей таблицы
        ) AS signature,
        -- Quality score
        t.score
        + 0.5 * t.has_cte
        + 0.3 * t.has_group_by
        + 0.4 * t.has_window
        + 0.1 * t.has_limit
        - LOG(1 + t.scan_bytes) / 40.0
        + 0.2 * (t.return_rows > 0)
        AS quality_score
    FROM ODS.TMP_SHOP_STARBUCKS_AUDIT_TBL_SELECT t
    JOIN valid_queries v ON t.query_id = v.query_id
    WHERE t.sql_clean IS NOT NULL
      AND t.sql_clean != ''
      AND t.return_rows > 0
      AND t.cpu_cost_ns < 3e10
),

ranked AS (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY signature
            ORDER BY quality_score DESC
        ) AS rn_sig,
        ROW_NUMBER() OVER (
            ORDER BY quality_score DESC
        ) AS rn_global
    FROM quality
)

SELECT
    query_id,
    query_time,
    sql_clean,
    tables,
    signature,
    has_cte, has_group_by, has_window, has_limit,
    cols_cnt,
    scan_bytes, return_rows,
    ROUND(quality_score, 3) AS quality_score,
    rn_sig,
    rn_global
FROM ranked
WHERE rn_sig <= 2      -- до 2 эталонов на структурную сигнатуру
   OR rn_global <= 5   -- + безусловный глобальный топ-5
ORDER BY quality_score DESC
LIMIT 30;



WITH normalized AS (
    SELECT
        *,
        MD5(
            -- 1. lowercase
            REGEXP_REPLACE(
            -- 2. строковые литералы → ?
            REGEXP_REPLACE(
            -- 3. числа → ?
            REGEXP_REPLACE(
            -- 4. IN (...) → IN (?)  
            REGEXP_REPLACE(
            -- 5. лишние пробелы
            REGEXP_REPLACE(
            -- 6. даты
            REGEXP_REPLACE(
                LOWER(TRIM(sql_clean)),
                '\\b\\d{4}-\\d{2}-\\d{2}(\\s\\d{2}:\\d{2}:\\d{2})?\\b', '?'
            ), '\\s+', ' '
            ), 'in\\s*\\([^)]+\\)', 'in (?)'
            ), '\\b\\d+\\.?\\d*\\b', '?'
            ), "'[^']*'", '?'
            ), '"[^"]*"', '?'
            )
        ) AS sql_fingerprint,

        -- Длина SQL как грубая мера сложности (для мягкой группировки)
        LENGTH(sql_clean) AS sql_len,

        -- Первые 200 символов нормализованного SQL — для визуальной проверки
        LEFT(
            REGEXP_REPLACE(
            REGEXP_REPLACE(
                LOWER(TRIM(sql_clean)),
                "'[^']*'", '?'
            ), '\\b\\d+\\.?\\d*\\b', '?'
            ),
            200
        ) AS sql_preview
    FROM quality
),

  typeof(json_extract(obj, '$.linked_rule_ids'))
FROM ODS.llm_raw___metrics_rules_scenarios___new d
LATERAL VIEW explode_json_array_json(json_extract(`data`, '$.scenarios')) t as obj
LIMIT 10;

cast(
  array_join(
    cast(
      nullif(json_extract(obj, '$.linked_rule_ids'), 'null') 
      as array<varchar>
    ), ','
  ) as varchar
) as related_rules


, CASE 
    WHEN json_extract(obj, '$.linked_rule_ids') IS NULL 
      OR json_extract(obj, '$.linked_rule_ids') = 'null'
      OR LEFT(TRIM(json_extract(obj, '$.linked_rule_ids')), 1) != '['
    THEN NULL
    ELSE array_join(
           cast(json_extract(obj, '$.linked_rule_ids') as array<varchar>),
         ',')
  END as related_rules

  AND json_extract(obj, '$.linked_rule_ids') IS NOT NULL
LIMIT 20;

WITH pairs AS (
    SELECT
        t1.id                                           AS id_1,
        t1.name                                         AS name_1,
        t2.id                                           AS id_2,
        t2.name                                         AS name_2,

        -- убираем дубликаты перед сравнением
        array_distinct(t1.arr)                          AS arr_1,
        array_distinct(t2.arr)                          AS arr_2,

        array_distinct(array_intersect(
            array_distinct(t1.arr),
            array_distinct(t2.arr)
        ))                                              AS intersection_arr,

        array_distinct(array_union(
            array_distinct(t1.arr),
            array_distinct(t2.arr)
        ))                                              AS union_arr

    FROM test_arrays t1
    JOIN test_arrays t2 ON t1.id < t2.id   -- только уникальные пары, без (A,B) и (B,A)
)
SELECT
    id_1,
    name_1,
    id_2,
    name_2,
    arr_1,
    arr_2,
    intersection_arr,
    array_size(intersection_arr)                        AS common_count,
    array_size(union_arr)                               AS union_count,

    round(
        array_size(intersection_arr) * 100.0
        / nullif(array_size(union_arr), 0),
        2
    )                                                   AS jaccard_pct

FROM pairs
WHERE
    array_size(intersection_arr) * 1.0
    / nullif(array_size(union_arr), 0) >= 0.8   -- порог 80%

ORDER BY jaccard_pct DESC;
